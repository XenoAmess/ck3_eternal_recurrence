# Player transfer guards and free defensive war calls.

xqol_refresh_no_vassal_transfer_flags_effect = {
	every_vassal_or_below = {
		xqol_reconcile_character_transfer_guard_effect = yes
	}
}

xqol_clear_no_vassal_transfer_flags_effect = {
	every_vassal_or_below = {
		limit = { has_character_flag = xqol_no_vassal_transfer_guard }
		remove_character_flag = xqol_no_vassal_transfer_guard
		remove_character_flag = ai_should_not_transfer
	}
}

xqol_reconcile_character_transfer_guard_effect = {
	if = {
		limit = { xqol_should_guard_vassal_transfer_trigger = yes }
		if = {
			limit = { has_character_flag = xqol_no_vassal_transfer_guard }
			# Repair the vanilla guard if another effect cleared it while ours is active.
			add_character_flag = ai_should_not_transfer
		}
		else_if = {
			limit = { NOT = { has_character_flag = ai_should_not_transfer } }
			# Claim ownership only when the vanilla flag was not already present.
			add_character_flag = xqol_no_vassal_transfer_guard
			add_character_flag = ai_should_not_transfer
		}
	}
	else_if = {
		limit = { has_character_flag = xqol_no_vassal_transfer_guard }
		remove_character_flag = xqol_no_vassal_transfer_guard
		remove_character_flag = ai_should_not_transfer
	}
}

xqol_try_call_free_ally_effect = {
	$CANDIDATE$ = { save_scope_as = recipient }
	if = {
		limit = {
			xqol_free_ally_relationship_trigger = yes
			xqol_free_call_target_valid_trigger = yes
		}
		call_ally_interaction_event_effect = yes
		change_variable = { name = xqol_last_auto_called_count add = 1 }
	}
}

xqol_try_call_free_house_member_effect = {
	$CANDIDATE$ = { save_scope_as = recipient }
	if = {
		limit = {
			xqol_free_house_relationship_trigger = yes
			xqol_free_house_call_target_valid_trigger = yes
		}
		call_ally_interaction_event_effect = yes
		change_variable = { name = xqol_last_auto_called_count add = 1 }
	}
}

xqol_auto_call_free_defenders_effect = {
	set_variable = { name = xqol_last_auto_called_count value = 0 }

	every_ally = {
		save_temporary_scope_as = xqol_call_candidate
		scope:actor = {
			xqol_try_call_free_ally_effect = { CANDIDATE = scope:xqol_call_candidate }
		}
	}
	if = {
		limit = { exists = liege }
		liege = { save_temporary_scope_as = xqol_call_candidate }
		xqol_try_call_free_ally_effect = { CANDIDATE = scope:xqol_call_candidate }
	}
	if = {
		limit = { exists = diarch }
		diarch = { save_temporary_scope_as = xqol_call_candidate }
		xqol_try_call_free_ally_effect = { CANDIDATE = scope:xqol_call_candidate }
	}
	if = {
		limit = { exists = suzerain }
		suzerain = { save_temporary_scope_as = xqol_call_candidate }
		xqol_try_call_free_ally_effect = { CANDIDATE = scope:xqol_call_candidate }
	}
	every_tributary = {
		save_temporary_scope_as = xqol_call_candidate
		scope:actor = {
			xqol_try_call_free_ally_effect = { CANDIDATE = scope:xqol_call_candidate }
		}
	}
	if = {
		limit = {
			is_house_head = yes
			exists = house
		}
		house = {
			every_house_member = {
				save_temporary_scope_as = xqol_call_candidate
				scope:actor = {
					xqol_try_call_free_house_member_effect = { CANDIDATE = scope:xqol_call_candidate }
				}
			}
		}
	}
	if = {
		limit = { var:xqol_last_auto_called_count > 0 }
		send_interface_toast = {
			type = event_war_good
			title = xqol_auto_call_defenders_result_title
			desc = xqol_auto_call_defenders_result_desc
		}
	}
}
