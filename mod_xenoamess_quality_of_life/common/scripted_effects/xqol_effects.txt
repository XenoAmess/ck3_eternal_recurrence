xqol_refresh_no_vassal_transfer_flags_effect = {
	every_vassal_or_below = {
		xqol_reconcile_character_transfer_guard_effect = yes
	}
}

xqol_clear_no_vassal_transfer_flags_effect = {
	every_vassal_or_below = {
		limit = { has_character_flag = xqol_no_vassal_transfer_guard }
		remove_character_flag = xqol_no_vassal_transfer_guard
		remove_character_flag = ai_should_not_transfer
	}
}

xqol_reconcile_character_transfer_guard_effect = {
	if = {
		limit = { xqol_should_guard_vassal_transfer_trigger = yes }
		if = {
			limit = { has_character_flag = xqol_no_vassal_transfer_guard }
			# Repair the vanilla guard if another effect cleared it while ours is active.
			add_character_flag = ai_should_not_transfer
		}
		else_if = {
			limit = { NOT = { has_character_flag = ai_should_not_transfer } }
			# Claim ownership only when the vanilla flag was not already present.
			add_character_flag = xqol_no_vassal_transfer_guard
			add_character_flag = ai_should_not_transfer
		}
	}
	else_if = {
		limit = { has_character_flag = xqol_no_vassal_transfer_guard }
		remove_character_flag = xqol_no_vassal_transfer_guard
		remove_character_flag = ai_should_not_transfer
	}
}

xqol_try_call_free_ally_effect = {
	$CANDIDATE$ = { save_scope_as = recipient }

		limit = {

				is_ruler = yes

		call_ally_interaction_effect = yes
		change_variable = { name = xqol_last_auto_called_count add = 1 }
	}
}

xqol_auto_call_free_defenders_effect = {
	set_variable = { name = xqol_last_auto_called_count value = 0 }

	every_ally = {
		save_temporary_scope_as = xqol_call_candidate
		scope:actor = {
			xqol_try_call_free_ally_effect = { CANDIDATE = scope:xqol_call_candidate }
		}
	}
	if = {
		limit = { exists = liege }
		liege = { save_temporary_scope_as = xqol_call_candidate }
		xqol_try_call_free_ally_effect = { CANDIDATE = scope:xqol_call_candidate }
	}
	if = {
		limit = { exists = diarch }
		diarch = { save_temporary_scope_as = xqol_call_candidate }
		xqol_try_call_free_ally_effect = { CANDIDATE = scope:xqol_call_candidate }
	}
	if = {
		limit = { exists = suzerain }
		suzerain = { save_temporary_scope_as = xqol_call_candidate }
		xqol_try_call_free_ally_effect = { CANDIDATE = scope:xqol_call_candidate }
	}
	every_tributary = {
		save_temporary_scope_as = xqol_call_candidate
		scope:actor = {
			xqol_try_call_free_ally_effect = { CANDIDATE = scope:xqol_call_candidate }
		}
	}
	if = {
		limit = {
			is_house_head = yes
			exists = house
		}
		house = {
			every_house_member = {
				save_temporary_scope_as = xqol_call_candidate
				scope:actor = {
					xqol_try_call_free_house_member_effect = { CANDIDATE = scope:xqol_call_candidate }
				}
			}
		}
	}
	if = {
		limit = { var:xqol_last_auto_called_count > 0 }
		send_interface_toast = {
			type = event_war_good
			title = xqol_auto_call_defenders_result_title
			desc = xqol_auto_call_defenders_result_desc
		}
	}
}

xqol_bulk_conversion_threshold_effect = {
	save_scope_as = actor
	set_variable = { name = xqol_mass_conversion_accepted value = 0 }
	set_variable = { name = xqol_mass_conversion_refused value = 0 }
	set_variable = { name = xqol_mass_conversion_pending value = 0 }

	every_courtier = {
		limit = {
			is_ai = yes
			scope:actor = {
				is_character_interaction_potentially_accepted = {
					recipient = prev
					interaction = ask_for_conversion_courtier_interaction
					ai_accept = $THRESHOLD$
				}
			}
		}
		add_character_flag = { flag = xqol_conversion_courtier_candidate days = 365 }
		scope:actor = { change_variable = { name = xqol_mass_conversion_pending add = 1 } }
	}

	every_vassal_or_below = {
		limit = {
			is_ai = yes
			scope:actor = {
				is_character_interaction_potentially_accepted = {
					recipient = prev
					interaction = demand_conversion_vassal_ruler_interaction
					ai_accept = $THRESHOLD$
				}
			}
		}
		add_character_flag = { flag = xqol_conversion_ruler_candidate days = 365 }
		scope:actor = { change_variable = { name = xqol_mass_conversion_pending add = 1 } }
	}

	every_tributary = {
		limit = {
			is_ai = yes
			NOT = { has_character_flag = xqol_conversion_ruler_candidate }
			scope:actor = {
				is_character_interaction_potentially_accepted = {
					recipient = prev
					interaction = demand_conversion_vassal_ruler_interaction
					ai_accept = $THRESHOLD$
				}
			}
		}
		add_character_flag = { flag = xqol_conversion_ruler_candidate days = 365 }
		scope:actor = { change_variable = { name = xqol_mass_conversion_pending add = 1 } }
	}

	if = {
		limit = { NOT = { has_variable = xqol_mass_conversion_pending } }
		xqol_mass_conversion_finish_effect = yes
	}
	else_if = {
		limit = { var:xqol_mass_conversion_pending <= 0 }
		xqol_mass_conversion_finish_effect = yes
	}
	else = {
		xqol_mass_conversion_dispatch_next_effect = yes
	}
}

xqol_mass_conversion_dispatch_next_effect = {
	if = {
		limit = { NOT = { has_variable = xqol_mass_conversion_pending } }
		xqol_mass_conversion_finish_effect = yes
	}
	else_if = {
		limit = { var:xqol_mass_conversion_pending <= 0 }
		xqol_mass_conversion_finish_effect = yes
	}
	else_if = {
		limit = {
			any_courtier = { has_character_flag = xqol_conversion_courtier_candidate }
		}
		random_courtier = {
			limit = { has_character_flag = xqol_conversion_courtier_candidate }
			remove_character_flag = xqol_conversion_courtier_candidate
			save_temporary_scope_as = xqol_conversion_candidate
		}
		if = {
			limit = {
				is_character_interaction_valid = {
					recipient = scope:xqol_conversion_candidate
					interaction = xqol_mass_conversion_courtier_interaction
				}
			}
			run_interaction = {
				interaction = xqol_mass_conversion_courtier_interaction
				actor = scope:actor
				recipient = scope:xqol_conversion_candidate
				send_threshold = decline
			}
		}
		else = {
			change_variable = { name = xqol_mass_conversion_pending add = -1 }
			xqol_mass_conversion_dispatch_next_effect = yes
		}
	}
	else_if = {
		limit = {
			any_vassal_or_below = { has_character_flag = xqol_conversion_ruler_candidate }
		}
		random_vassal_or_below = {
			limit = { has_character_flag = xqol_conversion_ruler_candidate }
			remove_character_flag = xqol_conversion_ruler_candidate
			save_temporary_scope_as = xqol_conversion_candidate
		}
		if = {
			limit = {
				is_character_interaction_valid = {
					recipient = scope:xqol_conversion_candidate
					interaction = xqol_mass_conversion_ruler_interaction
				}
			}
			run_interaction = {
				interaction = xqol_mass_conversion_ruler_interaction
				actor = scope:actor
				recipient = scope:xqol_conversion_candidate
				send_threshold = decline
			}
		}
		else = {
			change_variable = { name = xqol_mass_conversion_pending add = -1 }
			xqol_mass_conversion_dispatch_next_effect = yes
		}
	}
	else_if = {
		limit = {
			any_tributary = { has_character_flag = xqol_conversion_ruler_candidate }
		}
		random_tributary = {
			limit = { has_character_flag = xqol_conversion_ruler_candidate }
			remove_character_flag = xqol_conversion_ruler_candidate
			save_temporary_scope_as = xqol_conversion_candidate
		}
		if = {
			limit = {
				is_character_interaction_valid = {
					recipient = scope:xqol_conversion_candidate
					interaction = xqol_mass_conversion_ruler_interaction
				}
			}
			run_interaction = {
				interaction = xqol_mass_conversion_ruler_interaction
				actor = scope:actor
				recipient = scope:xqol_conversion_candidate
				send_threshold = decline
			}
		}
		else = {
			change_variable = { name = xqol_mass_conversion_pending add = -1 }
			xqol_mass_conversion_dispatch_next_effect = yes
		}
	}
	else = {
		# Any candidate who left the actor's domain is no longer requestable.
		set_variable = { name = xqol_mass_conversion_pending value = 0 }
		xqol_mass_conversion_finish_effect = yes
	}
}

xqol_release_prisoner_terms_effect = {
	if = {
		limit = { scope:recipient = { has_character_modifier = allowed_to_go_outside } }
		scope:recipient = { remove_character_modifier = allowed_to_go_outside }
	}
	if = {
		limit = { scope:recipient = { has_character_modifier = moldy_gruel_diet } }
		scope:recipient = { remove_character_modifier = moldy_gruel_diet }
	}
	scope:recipient = { release_from_prison = yes }

	if = {
		limit = { always = $CONVERSION$ }
		scope:recipient = {
			demand_conversion_interaction_effect = yes
			add_opinion = {
				modifier = demanded_my_conversion_opinion
				target = scope:actor
			}
		}
	}
	if = {
		limit = { always = $RECRUIT$ }
		if = {
			limit = {
				scope:actor = {
					culture = { has_cultural_parameter = can_recruit_prisoners_easily }
				}
			}
			scope:recipient = {
				add_opinion = { modifier = loyal_servant target = scope:actor }
			}
			scope:actor = {
				if = {
					limit = {
						can_add_hook = { target = scope:recipient type = loyalty_hook }
					}
					add_hook = { type = loyalty_hook target = scope:recipient }
				}
			}
		}
		else = {
			scope:recipient = {
				add_opinion = { modifier = demanded_recruitment target = scope:actor }
			}
		}
		if = {
			limit = {
				scope:actor.faith = { has_doctrine = tenet_communal_possessions }
				scope:recipient = { has_trait = peasant_leader }
			}
			scope:actor = { add_piety = medium_piety_gain }
		}
		scope:actor = { add_courtier = scope:recipient }
	}
	if = {
		limit = { always = $HOOK$ }
		scope:recipient = {
			add_opinion = { modifier = demanded_hook target = scope:actor }
		}
		scope:actor = {
			add_hook = { target = scope:recipient type = favor_hook }
		}
	}
}
