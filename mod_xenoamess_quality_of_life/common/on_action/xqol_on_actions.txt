on_game_start_after_lobby = { on_actions = { xqol_on_game_start } }
on_vassal_change = { on_actions = { xqol_on_vassal_change } }
yearly_playable_pulse = { on_actions = { xqol_yearly_player_maintenance } }

xqol_on_game_start = {
	effect = {
		every_player = {
			limit = { is_ai = no }
			xqol_refresh_no_vassal_transfer_flags_effect = yes
		}
	}
}
xqol_on_vassal_change = {
	effect = { root = { xqol_reconcile_character_transfer_guard_effect = yes } }
}

xqol_yearly_player_maintenance = {
	effect = {
		if = {
			limit = { is_ai = no }
			xqol_refresh_no_vassal_transfer_flags_effect = yes
		}
	}
}
