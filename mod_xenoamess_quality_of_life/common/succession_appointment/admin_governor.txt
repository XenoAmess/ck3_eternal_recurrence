admin_governor = {
	default_candidates = { holder_close_family }

	candidate_score = {
		value = {
			# Base - Based on skills
			add = {
				value = martial
				add = diplomacy
				add = stewardship
				add = intrigue
				add = learning
				multiply = 1
				desc = sum_of_skills_desc
			}

			add = appointment_score_base
			add = appointment_score_non_military
			add = appointment_score_non_merit
			add = appointment_score_governor

			# PIETY LEVEL (from admin family attribute)
			if = {
				limit = {
					piety_level >= 1
					house ?= {
						OR = {
							has_house_aspiration_parameter = candidate_score_per_piety_level_1
							has_house_aspiration_parameter = candidate_score_per_piety_level_2
							has_house_aspiration_parameter = candidate_score_per_piety_level_3
						}
					}
				}
				add = {
					value = 0
					add = {
						if = {
							limit = {
								house ?= { has_house_aspiration_parameter = candidate_score_per_piety_level_3 }
							}
							add = candidate_score_per_piety_level_3_value
						}
						else_if = {
							limit = {
								house ?= { has_house_aspiration_parameter = candidate_score_per_piety_level_2 }
							}
							add = candidate_score_per_piety_level_2_value
						}
						else = {
							add = candidate_score_per_piety_level_1_value
						}
					}
					multiply = piety_level
					desc = "family_attribute_desc"
				}
			}
			add = appointment_score_final_factors

			# XQOL_AUTO_APPOINTMENT_BEGIN
			if = {
				limit = {
					top_liege = this
					is_ai = no
					has_variable = xqol_auto_appoint_successors_enabled
				}
				subtract = {
					value = 1000000
					desc = xqol_auto_appointment_score_penalty_desc
				}
			}
			# XQOL_AUTO_APPOINTMENT_END
		}
	}
}
