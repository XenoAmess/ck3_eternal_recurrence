# Character scope. A PRIMARY reference must be private, usable and legally owned by this actor.
xfm_futures_ai_primary_account_trigger = {
 exists = var:xfm_account
 var:xfm_account = {
  xfm_finance_account_usable_trigger = yes
  var:xfm_system_account = 0 var:xfm_holder = prev
  NOT = { has_variable = xfm_legal_entity }
 }
}
xfm_futures_ai_funding_allowed_trigger = {
 is_ai = yes is_alive = yes is_adult = yes
 xfm_futures_ai_primary_account_trigger = yes
 scope:xfm_fut_account = var:xfm_account
 var:xfm_futures_reserve_cents > 0 var:xfm_futures_reserve_cents <= 100000000
 var:xfm_futures_reserve_cents <= xfm_futures_ai_budget_cents_value
 gold >= { value = xfm_futures_ai_gap_cents_value divide = 100 }
}
# Contract scope. This is a discovery index, not a replacement for permanent risk/claim indexes.
xfm_futures_discoverable_trigger = {
 story_type = xfm_futures_contract exists = var:xfm_coin var:xfm_status = 0
 OR = { var:xfm_perpetual = 1 var:xfm_expiry_day > global_var:xfm_exchange.var:xfm_day }
}
# Order scope; bounded contract-local open-order index. Full related-cross validation remains in submit.
xfm_futures_ai_target_order_trigger = {
 story_type = xfm_futures_order xfm_futures_order_open_trigger = yes
 exists = var:xfm_account exists = var:xfm_reservation
 var:xfm_account = {
  xfm_finance_account_usable_trigger = yes
  var:xfm_holder != scope:xfm_fut_ai_actor
  OR = { NOT = { exists = var:xfm_beneficial_owner } var:xfm_beneficial_owner != scope:xfm_fut_ai_actor }
 }
 var:xfm_reservation = { xfm_finance_reservation_usable_trigger = yes }
}

# Asset scope; only the bounded discovery directory and each nonempty contract's open book.
# The current actor alias is set by the normal AI entry; no account or cash is changed here.
xfm_futures_ai_demand_asset_trigger = {
 trigger_if = {
  limit = { var:xfm_max_leverage >= 1 var:xfm_oracle_day = global_var:xfm_exchange.var:xfm_day var:xfm_derivatives_reduce_only = 0 }
  any_in_list = {
   variable = xfm_futures_discovery_contracts
   trigger_if = {
    limit = { xfm_futures_discoverable_trigger = yes var:xfm_coin = prev has_variable = xfm_open_orders_count }
    trigger_if = {
     limit = { var:xfm_open_orders_count > 0 var:xfm_open_orders_count <= xfm_config_book_service_capacity_value xfm_futures_live_oracle_trigger = yes }
     any_in_list = {
      variable = xfm_orders
      trigger_if = {
       limit = { var:xfm_contract = prev xfm_futures_ai_target_order_trigger = yes }
       xfm_futures_order_collateralized_trigger = yes
      }
      trigger_else = { always = no }
     }
    }
    trigger_else = { always = no }
   }
   trigger_else = { always = no }
  }
 }
 trigger_else = { always = no }
}
