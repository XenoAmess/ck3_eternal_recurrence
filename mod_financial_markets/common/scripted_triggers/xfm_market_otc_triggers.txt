# One legal PRIMARY private account per living adult holder avoids duplicate or system buyers.

    story_type = xfm_finance_position
    var:xfm_quantity_units > 0
    var:xfm_coin = root.var:xfm_selected_coin
    exists = var:xfm_account
    var:xfm_account = {
        story_type = xfm_finance_account
        xfm_finance_account_usable_trigger = yes
        var:xfm_system_account = 0
        exists = var:xfm_holder
        var:xfm_holder = {
            is_alive = yes is_adult = yes
            NOT = { this = root }
            exists = var:xfm_account
            var:xfm_account = prev
        }
    }
}
