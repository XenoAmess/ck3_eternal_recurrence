 global_var:xfm_exchange = { var:xfm_active_orders_count < xfm_config_global_order_capacity_value }

xfm_spot_admission_trigger = {
 global_var:xfm_exchange = { var:xfm_order_stories_count < 24576 }
 OR = {
  $IOC$ = 1
  xfm_spot_book_slot_trigger = { ACCOUNT = $ACCOUNT$ COIN = $COIN$ }
  $COIN$ = {
   any_in_list = {
    variable = xfm_spot_orders
    xfm_spot_independent_crossing_trigger = { ACCOUNT = $ACCOUNT$ COIN = $COIN$ PRICE = $PRICE$ SIDE = $SIDE$ }
   }
  }
 }
}

xfm_spot_order_executable_trigger = {
 story_type = xfm_spot_order
 var:xfm_status = 1 var:xfm_remaining_lots > 0 var:xfm_limit_cents > 0
 exists = var:xfm_coin
 var:xfm_coin = { story_type = xfm_finance_coin var:xfm_trading_open = 1 var:xfm_listing_state < 5 }
 exists = var:xfm_account
 var:xfm_account = { xfm_finance_account_usable_trigger = yes }
 OR = {
  AND = {
   var:xfm_side = 1 exists = var:xfm_reservation
   var:xfm_reservation = {
    xfm_finance_reservation_usable_trigger = yes
    var:xfm_account = prev.var:xfm_account
    var:xfm_source_spot_order = prev
   }
  }
  AND = {
   var:xfm_side = 2 exists = var:xfm_sell_position
   var:xfm_sell_position = {
    story_type = xfm_finance_position
    var:xfm_account = prev.var:xfm_account var:xfm_coin = prev.var:xfm_coin
    var:xfm_quantity_units > 0 var:xfm_reserved_units > 0
   }
  }
 }
