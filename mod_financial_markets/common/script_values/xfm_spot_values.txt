# Current order. Reserve the whole original-order future fee before counting lots.
# This deliberately undercounts malformed/partially depleted collateral rather than
# advertising cash which cannot execute. CK3 max caps; min supplies a lower bound.
xfm_spot_remaining_fee_value = {
 value = var:xfm_remaining_lots multiply = var:xfm_limit_cents
 add = var:xfm_filled_notional_cents multiply = 0.003 ceiling = yes min = 1
 subtract = var:xfm_fee_paid_cents min = 0
}
xfm_spot_executable_lots_value = {
 value = 0
 if = {
  limit = { var:xfm_side = 1 exists = var:xfm_reservation var:xfm_limit_cents > 0 }
  value = var:xfm_reservation.var:xfm_remaining_cents
  max = var:xfm_account.var:xfm_reserved_cents
  subtract = xfm_spot_remaining_fee_value min = 0
  divide = var:xfm_limit_cents floor = yes max = var:xfm_remaining_lots
 }
 else_if = {
  limit = { var:xfm_side = 2 exists = var:xfm_sell_position }
  value = var:xfm_sell_position.var:xfm_quantity_units
  max = var:xfm_sell_position.var:xfm_reserved_units
  divide = 100 floor = yes min = 0 max = var:xfm_remaining_lots
 }
}
xfm_spot_executable_bid_cents_value = { value = xfm_spot_executable_lots_value multiply = var:xfm_limit_cents }
