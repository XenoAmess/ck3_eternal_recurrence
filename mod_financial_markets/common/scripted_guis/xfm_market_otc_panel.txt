   limit = { xfm_market_otc_select_buyer_gui_row_valid_trigger = yes }

    limit = { var:xfm_selected_coin = { is_target_in_variable_list = { name = xfm_holders target = scope:xfm_ui_item } }

        exists = scope:xfm_ui_item

     }

   }
  }
  clear_saved_scope = xfm_ui_item
 }
