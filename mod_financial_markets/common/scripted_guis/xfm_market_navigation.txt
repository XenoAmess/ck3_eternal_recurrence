xfm_market_select_coin_gui = {
 scope = character
 saved_scopes = { xfm_ui_row_id }
 is_valid = { xfm_market_select_coin_gui_row_valid_trigger = yes }
 effect = {
  clear_saved_scope = xfm_ui_item
  if = {
   limit = { xfm_market_select_coin_gui_row_valid_trigger = yes }
   xfm_market_row_resolve_coin_effect = yes
   if = {
     limit = {
        is_ai = no exists = scope:xfm_ui_item
        is_target_in_global_variable_list = { name = xfm_coin_registry target = scope:xfm_ui_item }
     }
     xfm_market_ui_select_coin_effect = yes
   }
  }
  clear_saved_scope = xfm_ui_item
 }
}

     limit = { var:xfm_active_account = { is_target_in_variable_list = { name = xfm_positions target = scope:xfm_ui_item } }  is_ai = no exists = scope:xfm_ui_item scope:xfm_ui_item = { story_type = xfm_finance_position var:xfm_account = root.var:xfm_active_account exists = var:xfm_coin }  }

        remove_variable = xfm_futures_selected_position

 }
