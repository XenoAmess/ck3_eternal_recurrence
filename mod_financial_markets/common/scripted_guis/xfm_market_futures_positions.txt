xfm_market_select_futures_position_gui = {
    scope = character
    saved_scopes = { xfm_ui_item }
    is_valid = { is_ai = no exists = scope:xfm_ui_item scope:xfm_ui_item = { story_type = xfm_futures_position OR = { var:xfm_long_account = root.var:xfm_active_account var:xfm_short_account = root.var:xfm_active_account } } }
    effect = {
        set_variable = { name = xfm_futures_selected_position value = scope:xfm_ui_item }
        set_variable = { name = xfm_futures_contract value = scope:xfm_ui_item.var:xfm_contract }
        set_variable = { name = xfm_selected_coin value = scope:xfm_ui_item.var:xfm_contract.var:xfm_coin }
        set_variable = { name = xfm_futures_limit_cents value = scope:xfm_ui_item.var:xfm_mark_cents }
        set_variable = { name = xfm_futures_request_cents value = 100 }
        set_variable = { name = xfm_futures_reduce_lots value = 1 }
        if = {
            limit = { scope:xfm_ui_item.var:xfm_long_account = var:xfm_active_account }
            set_variable = { name = xfm_futures_holding_side value = 1 }
            set_variable = { name = xfm_futures_stop_cents value = scope:xfm_ui_item.var:xfm_long_stop_cents }
            set_variable = { name = xfm_futures_take_profit_cents value = scope:xfm_ui_item.var:xfm_long_take_profit_cents }
        }
        else = {
            set_variable = { name = xfm_futures_holding_side value = 2 }
            set_variable = { name = xfm_futures_stop_cents value = scope:xfm_ui_item.var:xfm_short_stop_cents }
            set_variable = { name = xfm_futures_take_profit_cents value = scope:xfm_ui_item.var:xfm_short_take_profit_cents }
        }
        xfm_market_ui_futures_book_effect = yes
    }
}
