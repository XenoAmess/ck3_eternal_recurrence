xfm_market_select_debt_gui = {
 scope = character
 saved_scopes = { xfm_ui_row_id }
 is_valid = { xfm_market_select_debt_gui_row_valid_trigger = yes }
 effect = {
  clear_saved_scope = xfm_ui_item
  if = {
   limit = { xfm_market_select_debt_gui_row_valid_trigger = yes }
   xfm_market_social_resolve_debt_effect = yes
   if = {
    limit = { var:xfm_active_account = { is_target_in_variable_list = { name = xfm_debts target = scope:xfm_ui_item } }  is_ai = no exists = scope:xfm_ui_item scope:xfm_ui_item = { story_type = xfm_debt_claim }  }
     set_variable = { name = xfm_selected_debt value = scope:xfm_ui_item } set_variable = { name = xfm_ui_debt_pay_cents value = 1000 }
   }
  }
  clear_saved_scope = xfm_ui_item
 }
}
