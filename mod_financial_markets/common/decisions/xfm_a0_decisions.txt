    picture = { reference = "gfx/interface/illustrations/decisions/decision_misc.dds" }
    desc = xfm_a0_issue_decision_desc
    selection_tooltip = xfm_a0_issue_decision_tooltip
    confirm_text = xfm_a0_issue_decision_confirm
    ai_check_interval = 30
    is_shown = { xfm_a0_actor_eligible_trigger = yes NOT = { has_variable = xfm_personal_coin } }
    is_valid = { xfm_a0_can_issue_trigger = yes }

        xfm_a0_issue_effect = yes
        if = {
            limit = { is_ai = no exists = var:xfm_personal_coin }

        }
    }
    ai_potential = { xfm_a0_can_issue_trigger = yes gold >= 12 }
    ai_will_do = { base = 5 }
}

xfm_a0_portfolio_decision = {
    picture = { reference = "gfx/interface/illustrations/decisions/decision_misc.dds" }
    desc = xfm_a0_portfolio_decision_desc
    selection_tooltip = xfm_a0_portfolio_decision_tooltip
    confirm_text = xfm_a0_open
    ai_check_interval = 0
    is_shown = {
        is_ai = no
        any_owned_story = {
            story_type = xfm_a0_position
            NOT = { has_variable = xfm_frozen_beneficiary }
            var:xfm_quantity > 0
        }
    }
    is_valid = { xfm_a0_actor_eligible_trigger = yes }
    effect = {
        save_scope_as = xfm_actor
        random_owned_story = {
            limit = {
                story_type = xfm_a0_position
                NOT = { has_variable = xfm_frozen_beneficiary }
                var:xfm_quantity > 0
            }
            scope:xfm_actor = { set_variable = { name = xfm_selected_coin value = prev.var:xfm_coin } }
        }
        trigger_event = xfm_a0.100
    }
    ai_will_do = { base = 0 }
}
