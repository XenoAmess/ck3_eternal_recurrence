# AI makes a funded order decision on one of at most six watched markets.
xfm_spot_ai_effect = {
 save_scope_as = xfm_spot_ai_actor
 if = {
  limit = { is_ai = yes is_adult = yes is_alive = yes OR = { exists = var:xfm_account gold >= 1 } }
  xfm_finance_select_account_effect = yes
  scope:xfm_account = { save_scope_as = xfm_spot_ai_account }
  xfm_spot_refresh_watchlist_effect = yes
  # Bounded exploration prevents the first six markets from owning attention forever.
  random = {
   chance = 25
   clear_saved_scope = xfm_spot_rotate_coin
   random_in_list = { variable = xfm_finance_watchlist save_scope_as = xfm_spot_rotate_coin }
   if = {
    limit = {
     exists = scope:xfm_spot_rotate_coin
     scope:xfm_spot_ai_account = {
      NOT = { any_in_list = { variable = xfm_positions story_type = xfm_finance_position var:xfm_coin = scope:xfm_spot_rotate_coin var:xfm_quantity_units > 0 } }
      NOT = { any_in_list = { variable = xfm_positions story_type = xfm_futures_position var:xfm_coin = scope:xfm_spot_rotate_coin var:xfm_status = 0 var:xfm_lots > 0 } }
      NOT = { any_in_list = { variable = xfm_orders story_type = xfm_spot_order var:xfm_coin = scope:xfm_spot_rotate_coin var:xfm_status = 1 } }
      NOT = { any_in_list = { variable = xfm_orders story_type = xfm_futures_order OR = { var:xfm_status = 0 var:xfm_status = 1 } var:xfm_remaining_lots > 0 var:xfm_contract = { var:xfm_coin = scope:xfm_spot_rotate_coin } } }
     }
    }
    remove_list_variable = { name = xfm_finance_watchlist target = scope:xfm_spot_rotate_coin }
    change_variable = { name = xfm_watch_count subtract = 1 }
   }
  }
  clear_saved_scope = xfm_spot_ai_coin
  if = {
   limit = { NOT = { has_variable = xfm_watch_count } }
   set_variable = { name = xfm_watch_count value = 0 }
  }
  if = {
   limit = { var:xfm_watch_count < xfm_config_ai_watchlist_capacity_value }
   random_in_global_list = {
    variable = xfm_coin_registry
    # Reuse this one discovery draw for a real residual redemption service, never for closed-venue orders.
    limit = {
     trigger_if = {
      limit = {
       story_type = xfm_finance_coin
       has_variable = xfm_trading_open has_variable = xfm_listing_state
      }
      trigger_if = {
       limit = { var:xfm_trading_open = 1 var:xfm_listing_state < 5 }
       always = yes
      }
      trigger_else = {
       trigger_if = {
        limit = {
         OR = { var:xfm_listing_state = 5 var:xfm_listing_state = 6 }
        }
        trigger_if = {
         limit = {
          exists = var:xfm_redemption_window
          exists = var:xfm_redemption_account
          exists = var:xfm_issuer_account
          exists = global_var:xfm_exchange
         }
         trigger_if = {
          limit = {
           var:xfm_redemption_window = {
            story_type = xfm_finance_redemption_window
            has_variable = xfm_status has_variable = xfm_price_cents has_variable = xfm_expiry_day has_variable = xfm_remaining_cents
            exists = var:xfm_coin exists = var:xfm_redemption_account
           }
           var:xfm_redemption_account = {
            story_type = xfm_finance_account
            has_variable = xfm_account_frozen has_variable = xfm_free_cents has_variable = xfm_reserved_cents
            has_variable = xfm_system_account has_variable = xfm_redemption_only
            exists = var:xfm_legal_entity
           }
           var:xfm_issuer_account = { story_type = xfm_finance_account }
           global_var:xfm_exchange = { has_variable = xfm_day }
          }
          var:xfm_redemption_window = {
           trigger_if = {
            limit = {
             var:xfm_coin = prev
             var:xfm_redemption_account = prev.var:xfm_redemption_account
            }
            xfm_redemption_window_open_trigger = yes
           }
           trigger_else = { always = no }
          }
         }
         trigger_else = { always = no }
        }
        trigger_else = { always = no }
       }
       trigger_else = { always = no }
      }
     }
     trigger_else = { always = no }
    }
    save_scope_as = xfm_spot_ai_coin
   }
   if = {
    limit = { exists = scope:xfm_spot_ai_coin }
    set_variable = { name = xfm_selected_coin value = scope:xfm_spot_ai_coin }
    xfm_spot_watch_coin_effect = yes
   }
  }
  clear_saved_scope = xfm_spot_ai_coin
  if = {
   limit = { has_variable = xfm_promotion_end_day var:xfm_promotion_end_day <= global_var:xfm_exchange.var:xfm_day }
   remove_variable = xfm_promoted_coin
   remove_variable = xfm_promotion_end_day
   set_variable = { name = xfm_promotion_attention_days value = 0 }
  }
  random_in_list = {
   variable = xfm_finance_watchlist
   limit = { var:xfm_trading_open = 1 var:xfm_listing_state < 5 }
   save_scope_as = xfm_spot_ai_coin
  }
  if = {
   limit = {
    exists = var:xfm_promoted_coin
    has_variable = xfm_promotion_end_day
    var:xfm_promotion_end_day > global_var:xfm_exchange.var:xfm_day
    var:xfm_promoted_coin = { var:xfm_trading_open = 1 var:xfm_listing_state < 5 }
    any_in_list = { variable = xfm_finance_watchlist this = scope:xfm_spot_ai_actor.var:xfm_promoted_coin }
   }
   random = { chance = 50 var:xfm_promoted_coin = { save_scope_as = xfm_spot_ai_coin } }
  }
  if = {
   limit = { exists = scope:xfm_spot_ai_coin exists = scope:xfm_spot_ai_account scope:xfm_spot_ai_account = { xfm_finance_account_usable_trigger = yes } }
   scope:xfm_spot_ai_coin = { xfm_spot_maker_effect = yes }
   set_variable = { name = xfm_selected_coin value = scope:xfm_spot_ai_coin }
   set_variable = { name = xfm_order_lots value = 1 }
   set_variable = { name = xfm_order_ioc value = 0 }
   set_variable = { name = xfm_order_side value = 1 }
   if = {
    limit = { scope:xfm_spot_ai_coin = { var:xfm_best_ask_cents > 0 } }
    set_variable = { name = xfm_order_limit_cents value = scope:xfm_spot_ai_coin.var:xfm_best_ask_cents }
   }
   else = { set_variable = { name = xfm_order_limit_cents value = scope:xfm_spot_ai_coin.var:xfm_last_price_cents } }
   random = {
    chance = 50
    scope:xfm_spot_ai_account = {
     set_variable = { name = xfm_request_coin value = scope:xfm_spot_ai_coin }
     xfm_finance_lookup_position_effect = yes
    }
    if = {
     limit = { exists = scope:xfm_position scope:xfm_position = { xfm_finance_position_available_value >= 100 } }
     set_variable = { name = xfm_order_side value = 2 }
     if = {
      limit = { scope:xfm_spot_ai_coin = { var:xfm_best_bid_cents > 0 } }
      set_variable = { name = xfm_order_limit_cents value = scope:xfm_spot_ai_coin.var:xfm_best_bid_cents }
     }
    }
   }
   if = {
    limit = {
     var:xfm_order_limit_cents > 0
     NOT = {
      scope:xfm_spot_ai_account = {
       any_in_list = {
        variable = xfm_orders
        story_type = xfm_spot_order var:xfm_status = 1 var:xfm_coin = scope:xfm_spot_ai_coin
        var:xfm_side = scope:xfm_spot_ai_actor.var:xfm_order_side
       }
      }
     }
    }
    xfm_spot_submit_effect = yes
   }
  }
 }
}
