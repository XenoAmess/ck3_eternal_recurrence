# Character callback inside the EXISTING ordinary/paid attention decision budget.
# One watched coin and one primary-private redemption attempt; no additional world/holder scan.
xfm_redemption_ai_effect = {
 if = {
  limit = {
   is_ai = yes is_alive = yes is_adult = yes exists = var:xfm_account
   var:xfm_account = { xfm_finance_account_usable_trigger = yes var:xfm_system_account = 0 var:xfm_holder = prev NOT = { has_variable = xfm_legal_entity } }
   OR = { NOT = { has_variable = xfm_last_redemption_ai_day } var:xfm_last_redemption_ai_day < global_var:xfm_exchange.var:xfm_day }
  }
  set_variable = { name = xfm_last_redemption_ai_day value = global_var:xfm_exchange.var:xfm_day }
  clear_saved_scope = xfm_red_ai_previous_coin
  clear_saved_scope = xfm_red_ai_previous_active
  clear_saved_scope = xfm_red_ai_coin
  if = { limit = { exists = var:xfm_selected_coin } var:xfm_selected_coin = { save_scope_as = xfm_red_ai_previous_coin } }
  if = { limit = { exists = var:xfm_active_account } var:xfm_active_account = { save_scope_as = xfm_red_ai_previous_active } }
  random_in_list = {
   variable = xfm_finance_watchlist
   limit = {
    story_type = xfm_finance_coin exists = var:xfm_redemption_window
    var:xfm_redemption_window = { xfm_redemption_window_open_trigger = yes }
    OR = { var:xfm_best_bid_cents <= 0 var:xfm_redemption_window.var:xfm_price_cents >= xfm_redemption_ai_alternative_cents_value }
   }
   save_scope_as = xfm_red_ai_coin
  }
  if = {
   limit = { exists = scope:xfm_red_ai_coin }
   set_variable = { name = xfm_selected_coin value = scope:xfm_red_ai_coin }
   set_variable = { name = xfm_active_account value = var:xfm_account }
   set_variable = { name = xfm_redemption_lots value = 1 }
   xfm_redemption_prepare_quote_effect = yes
   if = { limit = { var:xfm_redemption_result = 1 } xfm_redemption_execute_quote_effect = yes }
  }
  if = { limit = { exists = scope:xfm_red_ai_previous_coin } set_variable = { name = xfm_selected_coin value = scope:xfm_red_ai_previous_coin } }
  else = { remove_variable = xfm_selected_coin }
  if = { limit = { exists = scope:xfm_red_ai_previous_active } set_variable = { name = xfm_active_account value = scope:xfm_red_ai_previous_active } }
  else = { remove_variable = xfm_active_account }
  clear_saved_scope = xfm_red_ai_previous_coin
  clear_saved_scope = xfm_red_ai_previous_active
  clear_saved_scope = xfm_red_ai_coin
 }
}
