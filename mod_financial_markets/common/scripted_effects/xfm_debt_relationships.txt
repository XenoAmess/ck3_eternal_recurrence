# 3 effects. Real cash/consent/refusal first; no repeated one-cent opinion farming.
xfm_debt_aid_relation_effect = {
    if = {
        limit = {
            var:xfm_payment_actual_cents >= 100
            scope:xfm_debt_payer.var:xfm_system_account = 0
            exists = scope:xfm_debt_payer.var:xfm_holder
            exists = var:xfm_debtor_account.var:xfm_holder
            scope:xfm_debt_payer.var:xfm_holder = { is_alive = yes }
            var:xfm_debtor_account.var:xfm_holder = { is_alive = yes }
            scope:xfm_debt_payer.var:xfm_holder != var:xfm_debtor_account.var:xfm_holder
        }
        scope:xfm_debt_payer.var:xfm_holder = { save_scope_as = xfm_debt_relationship_helper }
        var:xfm_debtor_account = {
            if = {
                limit = {
                    NOT = { any_in_list = { variable = xfm_aid_thanked_helpers this = scope:xfm_debt_relationship_helper } }
                }
                add_to_variable_list = { name = xfm_aid_thanked_helpers target = scope:xfm_debt_relationship_helper }
                var:xfm_holder = {
                    add_opinion = { target = scope:xfm_debt_relationship_helper modifier = xfm_debt_aid_opinion opinion = 10 years = 3 }
                }
            }
        }
    }
}
