# A 30-record all-flat array is an exact fixed point of the original shift.
# Missing fields, a new price, a trade, suspension/relisting/live state reset this proof.
xfm_spot_track_idle_bars_effect = {
 if = {
  limit = { xfm_spot_idle_bars_eligible_trigger = yes }
  if = {
   limit = {
    OR = {
     NOT = { has_variable = xfm_idle_bar_close_count }
     NOT = { has_variable = xfm_idle_bar_price_cents }
     var:xfm_idle_bar_price_cents != var:xfm_last_price_cents
    }
   }
   set_variable = { name = xfm_idle_bar_close_count value = 1 }
   set_variable = { name = xfm_idle_bar_price_cents value = var:xfm_last_price_cents }
  }
  else_if = {
   limit = { var:xfm_idle_bar_close_count < 30 }
   change_variable = { name = xfm_idle_bar_close_count add = 1 }
  }
 }
 else = {
  if = { limit = { has_variable = xfm_idle_bar_close_count } remove_variable = xfm_idle_bar_close_count }
  if = { limit = { has_variable = xfm_idle_bar_price_cents } remove_variable = xfm_idle_bar_price_cents }
 }
}
