# A close transfers exposure to a funded replacement. Opposing close orders can net.
xfm_futures_reduce_effect = {
 xfm_futures_select_position_side_effect = yes
 if = {
  limit = {
   exists = scope:xfm_fut_margin_account
   var:xfm_futures_reduce_lots >= 1
   var:xfm_futures_reduce_lots <= scope:xfm_fut_selected_position.var:xfm_lots
   var:xfm_futures_reduce_lots >= xfm_futures_reduce_lots_floor_value
   var:xfm_futures_reduce_lots <= xfm_futures_reduce_lots_floor_value
   var:xfm_futures_limit_cents >= 1 var:xfm_futures_limit_cents <= 1000000
   var:xfm_futures_limit_cents >= xfm_futures_limit_floor_value
   var:xfm_futures_limit_cents <= xfm_futures_limit_floor_value
   scope:xfm_fut_selected_position = { NOT = { has_variable = xfm_pending_payment } }
  }
  set_variable = { name = xfm_futures_duplicate_close value = 0 }
  scope:xfm_fut_selected_position.var:xfm_contract = {
   save_scope_as = xfm_fut_contract
   every_in_list = {
    variable = xfm_orders
    limit = { xfm_futures_order_open_trigger = yes var:xfm_close_position = scope:xfm_fut_selected_position var:xfm_account = scope:xfm_fut_margin_account }
    scope:xfm_fut_actor = { set_variable = { name = xfm_futures_duplicate_close value = 1 } }
   }
  }
  if = {
   limit = { var:xfm_futures_duplicate_close = 0 }
   set_variable = { name = xfm_futures_fee_gross value = { value = var:xfm_futures_reduce_lots multiply = var:xfm_futures_limit_cents floor = yes } }
   # Short-close buys have a maximum price; long-close sells reserve to the maximum resting bid.
   scope:xfm_fut_contract = {
    every_in_list = {
     variable = xfm_orders
     limit = { xfm_futures_order_open_trigger = yes var:xfm_side = 1 scope:xfm_fut_actor.var:xfm_futures_holding_side = 1 }
     save_scope_as = xfm_fut_reduce_candidate
     scope:xfm_fut_actor = {
      set_variable = { name = xfm_futures_fee_gross value = { value = var:xfm_futures_fee_gross min = { value = scope:xfm_fut_reduce_candidate.var:xfm_limit_cents multiply = var:xfm_futures_reduce_lots } } }
     }
    }
   }
   set_variable = { name = xfm_futures_fee_reserve value = xfm_futures_ceiling_fee_value }
   clear_saved_scope = xfm_fut_incoming
   global_var:xfm_exchange.story_owner = { create_story = { type = xfm_futures_order save_scope_as = xfm_fut_incoming } }
   if = {
    limit = { exists = scope:xfm_fut_incoming }
    scope:xfm_fut_margin_account = {
     set_variable = { name = xfm_request_cents value = scope:xfm_fut_actor.var:xfm_futures_fee_reserve }
     xfm_finance_reserve_effect = yes
    }
    if = {
     limit = { scope:xfm_fut_margin_account.var:xfm_result = 1 }
     global_var:xfm_exchange = { change_variable = { name = xfm_next_futures_order_id add = 1 } }
     scope:xfm_fut_incoming = {
      set_variable = { name = xfm_kind value = 2 }
      set_variable = { name = xfm_order_id value = global_var:xfm_exchange.var:xfm_next_futures_order_id }
      set_variable = { name = xfm_contract value = scope:xfm_fut_contract }
      set_variable = { name = xfm_account value = scope:xfm_fut_margin_account }
      set_variable = { name = xfm_reservation value = scope:xfm_fut_margin_account.var:xfm_result_reservation }
      set_variable = { name = xfm_side value = 1 }
      if = { limit = { scope:xfm_fut_actor.var:xfm_futures_holding_side = 1 } set_variable = { name = xfm_side value = 2 } }
      set_variable = { name = xfm_leverage value = 1 }
      set_variable = { name = xfm_close_position value = scope:xfm_fut_selected_position }
      set_variable = { name = xfm_reduce_order value = 1 }
      set_variable = { name = xfm_remaining_lots value = scope:xfm_fut_actor.var:xfm_futures_reduce_lots }
      set_variable = { name = xfm_limit_cents value = scope:xfm_fut_actor.var:xfm_futures_limit_cents }
      set_variable = { name = xfm_reserve_price_cents value = { value = scope:xfm_fut_actor.var:xfm_futures_fee_gross divide = var:xfm_remaining_lots ceiling = yes } }
      set_variable = { name = xfm_filled_gross_cents value = 0 }
      set_variable = { name = xfm_fee_paid_cents value = 0 }
      set_variable = { name = xfm_status value = 0 }
      set_variable = { name = xfm_indexed value = 0 }
     }
     scope:xfm_fut_incoming = { xfm_futures_match_effect = yes }
     set_variable = { name = xfm_futures_result value = 1 }
     set_variable = { name = xfm_futures_reason value = 0 }
    }
    else = { scope:xfm_fut_incoming = { end_story = yes } clear_saved_scope = xfm_fut_incoming }
   }
  }
 }
}

xfm_futures_prepare_slice_effect = {
 clear_saved_scope = xfm_fut_$ORDER$_slice
 clear_saved_scope = xfm_fut_$ORDER$_slice_long
 clear_saved_scope = xfm_fut_$ORDER$_slice_short
 if = {
  limit = { exists = scope:xfm_fut_$ORDER$.var:xfm_close_position }
  scope:xfm_fut_$ORDER$.var:xfm_close_position = { save_scope_as = xfm_fut_slice_source }
  if = {
   limit = {
    scope:xfm_fut_slice_source = {
     xfm_futures_position_open_trigger = yes
     NOT = { has_variable = xfm_pending_payment }
     var:xfm_lots >= scope:xfm_fut_fill.var:xfm_lots
     OR = {
      var:xfm_lots = scope:xfm_fut_fill.var:xfm_lots
      AND = {
       global_var:xfm_exchange.var:xfm_futures_open_positions_count < xfm_futures_global_position_capacity_value
       var:xfm_long_account.var:xfm_open_positions_count < xfm_futures_account_position_capacity_value
       var:xfm_short_account.var:xfm_open_positions_count < xfm_futures_account_position_capacity_value
      }
     }
    }
   }
   global_var:xfm_exchange.story_owner = {
    create_story = { type = xfm_futures_position save_scope_as = xfm_fut_$ORDER$_slice }
    create_story = { type = xfm_finance_reservation save_scope_as = xfm_fut_$ORDER$_slice_long }
    create_story = { type = xfm_finance_reservation save_scope_as = xfm_fut_$ORDER$_slice_short }
   }
   if = {
    limit = { exists = scope:xfm_fut_$ORDER$_slice exists = scope:xfm_fut_$ORDER$_slice_long exists = scope:xfm_fut_$ORDER$_slice_short }
    scope:xfm_fut_$ORDER$_slice = {
     set_variable = { name = xfm_source_position value = scope:xfm_fut_slice_source }
     set_variable = { name = xfm_lots value = scope:xfm_fut_fill.var:xfm_lots }
     set_variable = { name = xfm_long_margin_cents value = { value = var:xfm_source_position.var:xfm_long_reservation.var:xfm_remaining_cents multiply = var:xfm_lots divide = var:xfm_source_position.var:xfm_lots floor = yes } }
     set_variable = { name = xfm_short_margin_cents value = { value = var:xfm_source_position.var:xfm_short_reservation.var:xfm_remaining_cents multiply = var:xfm_lots divide = var:xfm_source_position.var:xfm_lots floor = yes } }
     set_variable = { name = xfm_slice_variation value = { value = scope:xfm_fut_fill.var:xfm_mark_cents subtract = var:xfm_source_position.var:xfm_mark_cents multiply = var:xfm_lots floor = yes } }
     set_variable = { name = xfm_long_reservation value = scope:xfm_fut_$ORDER$_slice_long }
     set_variable = { name = xfm_short_reservation value = scope:xfm_fut_$ORDER$_slice_short }
     if = {
      limit = {
       var:xfm_slice_variation >= -100000000 var:xfm_slice_variation <= 100000000
       OR = {
        AND = { var:xfm_slice_variation >= 0 var:xfm_short_margin_cents >= var:xfm_slice_variation }
        AND = { var:xfm_slice_variation < 0 var:xfm_long_margin_cents >= { value = var:xfm_slice_variation multiply = -1 } }
       }
      }
      set_variable = { name = xfm_slice_ready value = 1 }
     }
     else = { scope:xfm_fut_incoming = { set_variable = { name = xfm_reduction_ready value = 0 } } }
    }
   }
   else = { set_variable = { name = xfm_reduction_ready value = 0 } }
  }
  else = { set_variable = { name = xfm_reduction_ready value = 0 } }
 }
}

# Slice-scope. Prepared storage and collateral were checked before either side is changed.
xfm_futures_allocate_slice_effect = {
 save_scope_as = xfm_fut_allocating_slice
 set_variable = { name = xfm_kind value = 2 }
 set_variable = { name = xfm_contract value = var:xfm_source_position.var:xfm_contract }
 xfm_futures_assign_position_id_effect = yes
 set_variable = { name = xfm_coin value = var:xfm_source_position.var:xfm_coin }
 set_variable = { name = xfm_long_account value = var:xfm_source_position.var:xfm_long_account }
 set_variable = { name = xfm_short_account value = var:xfm_source_position.var:xfm_short_account }
 set_variable = { name = xfm_long_leverage value = var:xfm_source_position.var:xfm_long_leverage }
 set_variable = { name = xfm_short_leverage value = var:xfm_source_position.var:xfm_short_leverage }
 set_variable = { name = xfm_mark_cents value = var:xfm_source_position.var:xfm_mark_cents }
 set_variable = { name = xfm_last_settled_day value = var:xfm_source_position.var:xfm_last_settled_day }
 set_variable = { name = xfm_status value = 0 }
 global_var:xfm_exchange = { change_variable = { name = xfm_next_reservation_id add = 1 } }
 var:xfm_long_reservation = {
  set_variable = { name = xfm_account value = scope:xfm_fut_allocating_slice.var:xfm_long_account }
  set_variable = { name = xfm_futures_position value = scope:xfm_fut_allocating_slice }
  set_variable = { name = xfm_reservation_id value = global_var:xfm_exchange.var:xfm_next_reservation_id }
  set_variable = { name = xfm_remaining_cents value = scope:xfm_fut_allocating_slice.var:xfm_long_margin_cents }
  set_variable = { name = xfm_status value = 1 }
 }
 global_var:xfm_exchange = { change_variable = { name = xfm_next_reservation_id add = 1 } }
 var:xfm_short_reservation = {
  set_variable = { name = xfm_account value = scope:xfm_fut_allocating_slice.var:xfm_short_account }
  set_variable = { name = xfm_futures_position value = scope:xfm_fut_allocating_slice }
  set_variable = { name = xfm_reservation_id value = global_var:xfm_exchange.var:xfm_next_reservation_id }
  set_variable = { name = xfm_remaining_cents value = scope:xfm_fut_allocating_slice.var:xfm_short_margin_cents }
  set_variable = { name = xfm_status value = 1 }
 }
 var:xfm_source_position.var:xfm_long_reservation = { change_variable = { name = xfm_remaining_cents subtract = scope:xfm_fut_allocating_slice.var:xfm_long_margin_cents } }
 var:xfm_source_position.var:xfm_short_reservation = { change_variable = { name = xfm_remaining_cents subtract = scope:xfm_fut_allocating_slice.var:xfm_short_margin_cents } }
 var:xfm_long_account = { add_to_variable_list = { name = xfm_reservations target = scope:xfm_fut_allocating_slice.var:xfm_long_reservation } }
 var:xfm_short_account = { add_to_variable_list = { name = xfm_reservations target = scope:xfm_fut_allocating_slice.var:xfm_short_reservation } }
 var:xfm_source_position = {
  change_variable = { name = xfm_lots subtract = scope:xfm_fut_allocating_slice.var:xfm_lots }
  if = {
   limit = { var:xfm_lots = 0 }
   set_variable = { name = xfm_status value = 2 }
   var:xfm_long_account = { change_variable = { name = xfm_open_positions_count subtract = 1 } }
   var:xfm_short_account = { change_variable = { name = xfm_open_positions_count subtract = 1 } }
   xfm_futures_archive_position_effect = yes
  }
 }
 set_variable = { name = xfm_new_mark_cents value = scope:xfm_fut_fill.var:xfm_mark_cents }
 set_variable = { name = xfm_settlement_kind value = 3 }
 set_variable = { name = xfm_variation_cents value = var:xfm_slice_variation }
 xfm_futures_begin_settlement_effect = yes
}

xfm_futures_execute_reduction_effect = {
 set_variable = { name = xfm_reduction_ready value = 1 }
 xfm_futures_prepare_slice_effect = { ORDER = buy }
 if = {
  limit = { exists = scope:xfm_fut_buy.var:xfm_close_position exists = scope:xfm_fut_sell.var:xfm_close_position scope:xfm_fut_buy.var:xfm_close_position = scope:xfm_fut_sell.var:xfm_close_position }
  scope:xfm_fut_buy_slice = { save_scope_as = xfm_fut_sell_slice }
 }
 else = { xfm_futures_prepare_slice_effect = { ORDER = sell } }
 if = {
  limit = { var:xfm_reduction_ready = 1 }
  # Reject a novation that would leave the same beneficiary on both sides.
  if = {
   limit = { exists = scope:xfm_fut_buy_slice }
   scope:xfm_fut_fill = {
    set_variable = { name = xfm_long_account value = scope:xfm_fut_buy_slice.var:xfm_source_position.var:xfm_long_account }
    set_variable = { name = xfm_long_leverage value = scope:xfm_fut_buy_slice.var:xfm_source_position.var:xfm_long_leverage }
   }
  }
  else = { scope:xfm_fut_fill = { set_variable = { name = xfm_long_account value = scope:xfm_fut_buy.var:xfm_account } set_variable = { name = xfm_long_leverage value = scope:xfm_fut_buy.var:xfm_leverage } } }
  if = {
   limit = { exists = scope:xfm_fut_sell_slice }
   scope:xfm_fut_fill = {
    set_variable = { name = xfm_short_account value = scope:xfm_fut_sell_slice.var:xfm_source_position.var:xfm_short_account }
    set_variable = { name = xfm_short_leverage value = scope:xfm_fut_sell_slice.var:xfm_source_position.var:xfm_short_leverage }
   }
  }
  else = { scope:xfm_fut_fill = { set_variable = { name = xfm_short_account value = scope:xfm_fut_sell.var:xfm_account } set_variable = { name = xfm_short_leverage value = scope:xfm_fut_sell.var:xfm_leverage } } }
  scope:xfm_fut_fill.var:xfm_long_account = { xfm_finance_refresh_beneficiary_effect = yes }
  scope:xfm_fut_fill.var:xfm_short_account = { xfm_finance_refresh_beneficiary_effect = yes }
  if = {
   limit = {
    scope:xfm_fut_fill.var:xfm_long_account.var:xfm_holder != scope:xfm_fut_fill.var:xfm_short_account.var:xfm_holder
    NOT = { AND = {
     exists = scope:xfm_fut_fill.var:xfm_long_account.var:xfm_beneficial_owner
     exists = scope:xfm_fut_fill.var:xfm_short_account.var:xfm_beneficial_owner
     scope:xfm_fut_fill.var:xfm_long_account.var:xfm_beneficial_owner = scope:xfm_fut_fill.var:xfm_short_account.var:xfm_beneficial_owner
    } }
   }
   if = { limit = { exists = scope:xfm_fut_buy_slice } scope:xfm_fut_buy_slice = { xfm_futures_allocate_slice_effect = yes } }
   if = {
    limit = { exists = scope:xfm_fut_sell_slice }
    # A single SELL close has no buy_slice. Never compare an existing scope with a missing one.
    if = { limit = { NOT = { exists = scope:xfm_fut_buy_slice } } scope:xfm_fut_sell_slice = { xfm_futures_allocate_slice_effect = yes } }
    else_if = { limit = { scope:xfm_fut_sell_slice != scope:xfm_fut_buy_slice } scope:xfm_fut_sell_slice = { xfm_futures_allocate_slice_effect = yes } }
   }
   xfm_futures_finish_reduction_effect = yes
  }
  else = { set_variable = { name = xfm_match_continue value = 0 } }
 }
 else = { set_variable = { name = xfm_match_continue value = 0 } }
}

xfm_futures_finish_reduction_effect = {
 if = {
  limit = { exists = scope:xfm_fut_buy_slice scope:xfm_fut_buy_slice.var:xfm_short_reservation.var:xfm_remaining_cents > 0 }
  scope:xfm_fut_buy_slice.var:xfm_short_reservation = {
   set_variable = { name = xfm_request_cents value = var:xfm_remaining_cents }
   xfm_finance_release_effect = yes
  }
 }
 if = {
  limit = { exists = scope:xfm_fut_sell_slice scope:xfm_fut_sell_slice.var:xfm_long_reservation.var:xfm_remaining_cents > 0 }
  scope:xfm_fut_sell_slice.var:xfm_long_reservation = {
   set_variable = { name = xfm_request_cents value = var:xfm_remaining_cents }
   xfm_finance_release_effect = yes
  }
 }
 if = {
  limit = { exists = scope:xfm_fut_buy_slice exists = scope:xfm_fut_sell_slice scope:xfm_fut_buy_slice = scope:xfm_fut_sell_slice }
  # The actual old long and short both exited the same matched slice.
  scope:xfm_fut_fill = { set_variable = { name = xfm_status value = 2 } }
 }
 else = {
  scope:xfm_fut_fill = {
   set_variable = { name = xfm_kind value = 2 }
   set_variable = { name = xfm_contract value = scope:xfm_fut_incoming.var:xfm_contract }
   xfm_futures_assign_position_id_effect = yes
   set_variable = { name = xfm_coin value = var:xfm_contract.var:xfm_coin }
   set_variable = { name = xfm_long_reservation value = scope:xfm_fut_long_reserve }
   set_variable = { name = xfm_short_reservation value = scope:xfm_fut_short_reserve }
   set_variable = { name = xfm_last_settled_day value = global_var:xfm_exchange.var:xfm_day }
   set_variable = { name = xfm_created_day value = global_var:xfm_exchange.var:xfm_day }
   set_variable = { name = xfm_status value = 0 }
   set_variable = { name = xfm_long_stop_cents value = 0 }
   set_variable = { name = xfm_long_take_profit_cents value = 0 }
   set_variable = { name = xfm_short_stop_cents value = 0 }
   set_variable = { name = xfm_short_take_profit_cents value = 0 }
   set_variable = { name = xfm_long_exit_requested value = 0 }
   set_variable = { name = xfm_short_exit_requested value = 0 }
   if = {
    limit = { exists = scope:xfm_fut_buy_slice }
    set_variable = { name = xfm_long_reservation value = scope:xfm_fut_buy_slice.var:xfm_long_reservation }
   }
   else = {
    global_var:xfm_exchange = { change_variable = { name = xfm_next_reservation_id add = 1 } }
    scope:xfm_fut_long_reserve = {
     set_variable = { name = xfm_reservation_id value = global_var:xfm_exchange.var:xfm_next_reservation_id }
     set_variable = { name = xfm_account value = scope:xfm_fut_fill.var:xfm_long_account }
     set_variable = { name = xfm_remaining_cents value = scope:xfm_fut_fill.var:xfm_long_margin_cents }
     set_variable = { name = xfm_status value = 1 }
    }
    scope:xfm_fut_buy.var:xfm_reservation = { change_variable = { name = xfm_remaining_cents subtract = scope:xfm_fut_fill.var:xfm_long_margin_cents } }
    var:xfm_long_account = { add_to_variable_list = { name = xfm_reservations target = scope:xfm_fut_long_reserve } }
   }
   if = {
    limit = { exists = scope:xfm_fut_sell_slice }
    set_variable = { name = xfm_short_reservation value = scope:xfm_fut_sell_slice.var:xfm_short_reservation }
   }
   else = {
    global_var:xfm_exchange = { change_variable = { name = xfm_next_reservation_id add = 1 } }
    scope:xfm_fut_short_reserve = {
     set_variable = { name = xfm_reservation_id value = global_var:xfm_exchange.var:xfm_next_reservation_id }
     set_variable = { name = xfm_account value = scope:xfm_fut_fill.var:xfm_short_account }
     set_variable = { name = xfm_remaining_cents value = scope:xfm_fut_fill.var:xfm_short_margin_cents }
     set_variable = { name = xfm_status value = 1 }
    }
    scope:xfm_fut_sell.var:xfm_reservation = { change_variable = { name = xfm_remaining_cents subtract = scope:xfm_fut_fill.var:xfm_short_margin_cents } }
    var:xfm_short_account = { add_to_variable_list = { name = xfm_reservations target = scope:xfm_fut_short_reserve } }
   }
   var:xfm_contract = { add_to_variable_list = { name = xfm_positions target = scope:xfm_fut_fill } }
   var:xfm_long_reservation = { set_variable = { name = xfm_futures_position value = scope:xfm_fut_fill } }
   var:xfm_short_reservation = { set_variable = { name = xfm_futures_position value = scope:xfm_fut_fill } }
   var:xfm_long_account = {
    add_to_variable_list = { name = xfm_positions target = scope:xfm_fut_fill }
    change_variable = { name = xfm_open_positions_count add = 1 }
   }
   var:xfm_short_account = {
    add_to_variable_list = { name = xfm_positions target = scope:xfm_fut_fill }
    change_variable = { name = xfm_open_positions_count add = 1 }
   }
   global_var:xfm_exchange = { change_variable = { name = xfm_futures_open_positions_count add = 1 } }
  }
 }
 if = { limit = { exists = scope:xfm_fut_buy_slice } scope:xfm_fut_buy_slice = { set_variable = { name = xfm_status value = 2 } set_variable = { name = xfm_archived_day value = global_var:xfm_exchange.var:xfm_day } xfm_futures_archive_enqueue_effect = yes } }
 if = {
  limit = { exists = scope:xfm_fut_sell_slice }
  if = { limit = { NOT = { exists = scope:xfm_fut_buy_slice } } scope:xfm_fut_sell_slice = { set_variable = { name = xfm_status value = 2 } set_variable = { name = xfm_archived_day value = global_var:xfm_exchange.var:xfm_day } xfm_futures_archive_enqueue_effect = yes } }
  else_if = { limit = { scope:xfm_fut_sell_slice != scope:xfm_fut_buy_slice } scope:xfm_fut_sell_slice = { set_variable = { name = xfm_status value = 2 } set_variable = { name = xfm_archived_day value = global_var:xfm_exchange.var:xfm_day } xfm_futures_archive_enqueue_effect = yes } }
 }
 scope:xfm_fut_buy = { xfm_futures_fill_order_effect = yes }
 scope:xfm_fut_sell = { xfm_futures_fill_order_effect = yes }
 scope:xfm_fut_fill.var:xfm_contract = {
  if = {
   limit = { var:xfm_execution_day != global_var:xfm_exchange.var:xfm_day }
   set_variable = { name = xfm_execution_day value = global_var:xfm_exchange.var:xfm_day }
   set_variable = { name = xfm_execution_volume_cents value = 0 }
   set_variable = { name = xfm_execution_lots value = 0 }
  }
