# 6 effects. Ordinary and attention work share one total AI budget. Source-only scheduler; the composed queue requires CK3 acceptance.
# Current scope contracts: seed/daily/pop = exchange story; enqueue/register/ensure = character.
# Scripted effects preserve ROOT. Consumers must use current THIS, not assume a new ROOT.
# Pointer variables contain character scopes; ownership of financial rights is independent.
# Death-to-dequeue pointer survival, save/reload and callback composition remain unverified.

xfm_finance_seed_ai_effect = {
    xfm_finance_ai_queue_ensure_effect = yes
}

xfm_finance_enqueue_ai_effect = {
    xfm_finance_ai_queue_push_effect = yes
}

xfm_finance_register_character_effect = {
    # Character-creation hooks can fire before the player bootstraps the exchange.
    # In that case the later initial seed includes the character once.
    if = {
        limit = {
            exists = global_var:xfm_exchange
            global_var:xfm_exchange = { has_variable = xfm_ai_seeded }
        }
        xfm_finance_enqueue_ai_effect = yes
    }
}

xfm_finance_pop_ai_effect = {
    xfm_finance_ai_queue_pop_effect = yes
}

xfm_finance_daily_effect = {
    xfm_finance_daily_attempt_begin_effect = yes
    if = {
        limit = {
            exists = global_var:xfm_exchange
            this = global_var:xfm_exchange
            has_variable = xfm_last_date
            var:xfm_last_date < current_date
        }
        set_variable = { name = xfm_daily_attempt_entered_dispatch value = 1 }
        xfm_finance_daily_calendar_effect = yes
    }
    xfm_finance_daily_attempt_finish_effect = yes
}

xfm_finance_ensure_effect = {
    # Public character-scope entry for decisions/interactions when enabling in an old save.
    # bootstrap owns the exchange singleton and initializes xfm_day/xfm_last_date.
    if = {
        limit = { NOT = { exists = global_var:xfm_exchange } }
        xfm_finance_bootstrap_effect = yes
    }
    else_if = {
        limit = {
            global_var:xfm_exchange = { NOT = { has_variable = xfm_ai_seeded } }
        }
        global_var:xfm_exchange = { xfm_finance_seed_ai_effect = yes }
    }
    if = {
        limit = { exists = global_var:xfm_exchange }
        global_var:xfm_exchange = { xfm_finance_daily_event_schedule_effect = yes }
    }
}

# Independent callback diagnostics. These fields never settle or replay a financial day.
# This block is appended to xfm_finance_schedule.txt; no new timer or caller is installed.
xfm_finance_daily_attempt_begin_effect = {
    if = {
        limit = { exists = global_var:xfm_exchange this = global_var:xfm_exchange }
        if = {
            limit = { NOT = { has_variable = xfm_daily_attempt_count } }
            set_variable = { name = xfm_daily_attempt_count value = 0 }
            set_variable = { name = xfm_daily_attempt_count_saturated value = 0 }
            set_variable = { name = xfm_daily_rejected_count value = 0 }
            set_variable = { name = xfm_daily_rejected_count_saturated value = 0 }
        }
        xfm_finance_daily_attempt_total_effect = yes
        xfm_finance_daily_attempt_date_effect = yes
        set_variable = { name = xfm_daily_attempt_script_date value = current_date }
        set_variable = { name = xfm_daily_attempt_entered_dispatch value = 0 }
        set_variable = { name = xfm_daily_attempt_prev_date_present value = 0 }
        if = {
            limit = { has_variable = xfm_last_date }
            set_variable = { name = xfm_daily_attempt_prev_date value = var:xfm_last_date }
            set_variable = { name = xfm_daily_attempt_prev_date_present value = 1 }
        }
        else = { remove_variable = xfm_daily_attempt_prev_date }
    }
}

xfm_finance_daily_attempt_total_effect = {
    if = {
        limit = { var:xfm_daily_attempt_count < 1000000000 }
        change_variable = { name = xfm_daily_attempt_count add = 1 }
    }
    else = { set_variable = { name = xfm_daily_attempt_count_saturated value = 1 } }
}

xfm_finance_daily_attempt_date_effect = {
    # Compare the previous real helper-entry date before the caller overwrites it.
    # New gap counters start only on a newly observed forward gap; missing old history stays unknown.
    if = {
        limit = { has_variable = xfm_daily_attempt_script_date }
        set_variable = { name = xfm_daily_attempt_script_date_delta value = { value = current_date subtract = var:xfm_daily_attempt_script_date } }
        if = { limit = { var:xfm_daily_attempt_script_date_delta > 1 } xfm_finance_daily_attempt_gap_effect = yes }
    }
    else = { remove_variable = xfm_daily_attempt_script_date_delta }
    if = {
        limit = {
            has_variable = xfm_daily_attempt_script_date
            var:xfm_daily_attempt_script_date = current_date
        }
        if = {
            limit = { var:xfm_daily_attempts_on_date < 1024 }
            change_variable = { name = xfm_daily_attempts_on_date add = 1 }
        }
        else = { set_variable = { name = xfm_daily_attempts_on_date_saturated value = 1 } }
    }
    else = {
        set_variable = { name = xfm_daily_attempts_on_date value = 1 }
        set_variable = { name = xfm_daily_attempts_on_date_saturated value = 0 }
    }
}

xfm_finance_daily_attempt_finish_effect = {
    if = {
        limit = {
            exists = global_var:xfm_exchange
            this = global_var:xfm_exchange
            has_variable = xfm_daily_attempt_entered_dispatch
            var:xfm_daily_attempt_entered_dispatch = 0
        }
        if = {
            limit = { var:xfm_daily_rejected_count < 1000000000 }
            change_variable = { name = xfm_daily_rejected_count add = 1 }
        }
        else = { set_variable = { name = xfm_daily_rejected_count_saturated value = 1 } }
        set_variable = { name = xfm_daily_last_rejected_date value = var:xfm_daily_attempt_script_date }
        set_variable = { name = xfm_daily_last_rejected_prev_date_present value = var:xfm_daily_attempt_prev_date_present }
        if = {
            limit = { var:xfm_daily_attempt_prev_date_present = 1 }
            set_variable = { name = xfm_daily_last_rejected_prev_date value = var:xfm_daily_attempt_prev_date }
        }
        else = { remove_variable = xfm_daily_last_rejected_prev_date }
    }
}

# Raw timer-entry diagnostics, before the authorized daily helper.
# These private aliases never replace scope:story or any financial caller alias.
xfm_finance_daily_raw_entry_effect = {
    clear_saved_scope = xfm_daily_raw_entry_caller
    clear_saved_scope = xfm_daily_raw_entry_story
    save_scope_as = xfm_daily_raw_entry_caller
    if = {
        limit = { exists = scope:story }
        scope:story = { save_scope_as = xfm_daily_raw_entry_story }
    }
    if = {
        limit = {
            exists = global_var:xfm_exchange
            global_var:xfm_exchange = { story_type = xfm_finance_exchange }
        }
        global_var:xfm_exchange = { xfm_finance_daily_raw_record_effect = yes }
    }
    clear_saved_scope = xfm_daily_raw_entry_caller
    clear_saved_scope = xfm_daily_raw_entry_story
}

xfm_finance_daily_raw_record_effect = {
    if = {
        limit = { NOT = { has_variable = xfm_daily_raw_attempt_count } }
        set_variable = { name = xfm_daily_raw_attempt_count value = 0 }
        set_variable = { name = xfm_daily_raw_attempt_count_saturated value = 0 }
        set_variable = { name = xfm_daily_raw_mismatch_count value = 0 }
        set_variable = { name = xfm_daily_raw_mismatch_count_saturated value = 0 }
    }
    if = {
        limit = { var:xfm_daily_raw_attempt_count < 1000000000 }
        change_variable = { name = xfm_daily_raw_attempt_count add = 1 }
    }
    else = { set_variable = { name = xfm_daily_raw_attempt_count_saturated value = 1 } }
    xfm_finance_daily_raw_dates_effect = yes
    set_variable = { name = xfm_daily_raw_script_date value = current_date }
    xfm_finance_daily_raw_capture_effect = yes
    if = {
        limit = {
            OR = {
                var:xfm_daily_raw_caller_matches_exchange = 0
                var:xfm_daily_raw_story_matches_exchange = 0
            }
        }
        xfm_finance_daily_raw_mismatch_effect = yes
    }
}

xfm_finance_daily_raw_capture_effect = {
    set_variable = { name = xfm_daily_raw_caller_present value = 0 }
    set_variable = { name = xfm_daily_raw_caller_matches_exchange value = 0 }
    if = {
        limit = { exists = scope:xfm_daily_raw_entry_caller }
        set_variable = { name = xfm_daily_raw_caller value = scope:xfm_daily_raw_entry_caller }
        set_variable = { name = xfm_daily_raw_caller_present value = 1 }
        if = {
            limit = { this = scope:xfm_daily_raw_entry_caller }
            set_variable = { name = xfm_daily_raw_caller_matches_exchange value = 1 }
        }
    }
    else = { remove_variable = xfm_daily_raw_caller }
    set_variable = { name = xfm_daily_raw_story_present value = 0 }
    set_variable = { name = xfm_daily_raw_story_matches_exchange value = 0 }
    if = {
        limit = { exists = scope:xfm_daily_raw_entry_story }
        set_variable = { name = xfm_daily_raw_story value = scope:xfm_daily_raw_entry_story }
        set_variable = { name = xfm_daily_raw_story_present value = 1 }
        if = {
            limit = { this = scope:xfm_daily_raw_entry_story }
            set_variable = { name = xfm_daily_raw_story_matches_exchange value = 1 }
        }
    }
    else = { remove_variable = xfm_daily_raw_story }
}

xfm_finance_daily_raw_mismatch_effect = {
    if = {
        limit = { var:xfm_daily_raw_mismatch_count < 1000000000 }
        change_variable = { name = xfm_daily_raw_mismatch_count add = 1 }
    }
    else = { set_variable = { name = xfm_daily_raw_mismatch_count_saturated value = 1 } }
    set_variable = { name = xfm_daily_raw_last_mismatch_date value = var:xfm_daily_raw_script_date }
    set_variable = { name = xfm_daily_raw_last_caller_present value = var:xfm_daily_raw_caller_present }
    set_variable = { name = xfm_daily_raw_last_caller_matches_exchange value = var:xfm_daily_raw_caller_matches_exchange }
    if = {
        limit = { var:xfm_daily_raw_caller_present = 1 }
        set_variable = { name = xfm_daily_raw_last_caller value = var:xfm_daily_raw_caller }
    }
    else = { remove_variable = xfm_daily_raw_last_caller }
    set_variable = { name = xfm_daily_raw_last_story_present value = var:xfm_daily_raw_story_present }
    set_variable = { name = xfm_daily_raw_last_story_matches_exchange value = var:xfm_daily_raw_story_matches_exchange }
    if = {
        limit = { var:xfm_daily_raw_story_present = 1 }
        set_variable = { name = xfm_daily_raw_last_story value = var:xfm_daily_raw_story }
    }
    else = { remove_variable = xfm_daily_raw_last_story }
}

# Recorded raw dates are script calendar-day ordinals, not Native raw ticks.
# Keep only the latest pair and the first observed forward gap; never replay it.
xfm_finance_daily_raw_dates_effect = {
    set_variable = { name = xfm_daily_raw_prev_script_date_present value = 0 }
    if = {
        limit = { has_variable = xfm_daily_raw_script_date }
        set_variable = { name = xfm_daily_raw_prev_script_date value = var:xfm_daily_raw_script_date }
        set_variable = { name = xfm_daily_raw_prev_script_date_present value = 1 }
        set_variable = { name = xfm_daily_raw_script_date_delta value = { value = current_date subtract = var:xfm_daily_raw_script_date } }
        if = { limit = { var:xfm_daily_raw_script_date_delta > 1 } xfm_finance_daily_raw_gap_effect = yes }
        if = {
            limit = { var:xfm_daily_raw_script_date_delta > 1 NOT = { has_variable = xfm_daily_raw_first_gap_date } }
            set_variable = { name = xfm_daily_raw_first_gap_prev_date value = var:xfm_daily_raw_script_date }
            set_variable = { name = xfm_daily_raw_first_gap_missing_dates value = { value = var:xfm_daily_raw_script_date_delta subtract = 1 } }
            set_variable = { name = xfm_daily_raw_first_gap_attempt_count value = var:xfm_daily_raw_attempt_count }
            set_variable = { name = xfm_daily_raw_first_gap_date value = current_date }
        }
    }
    else = {
        remove_variable = xfm_daily_raw_prev_script_date
        remove_variable = xfm_daily_raw_script_date_delta
    }
}


# The first real callback adopts an origin. It never replays pre-adoption missing days.
# xfm_last_date is a durable admission cursor, not proof that every callee succeeded.
# A leftover in-flight marker is UNKNOWN: it is not retried by another callback.
xfm_finance_daily_calendar_effect = {
    if = {
        limit = { NOT = { has_variable = xfm_calendar_protocol } }
        set_variable = { name = xfm_calendar_protocol value = 1 }
        set_variable = { name = xfm_calendar_origin_date value = current_date }
        set_variable = { name = xfm_calendar_origin_financial_day value = var:xfm_day }
        set_variable = { name = xfm_calendar_preorigin_unprocessed_dates value = { value = current_date subtract = var:xfm_last_date subtract = 1 min = 0 } }
        set_variable = { name = xfm_last_date value = { value = current_date subtract = 1 } }
    }
    if = {
        limit = {
            var:xfm_calendar_protocol = 1
            OR = { NOT = { has_variable = xfm_calendar_in_flight } var:xfm_calendar_in_flight = 0 }
        }
        set_variable = { name = xfm_calendar_observed_date value = current_date }
        set_variable = { name = xfm_calendar_processed_in_callback value = 0 }
        while = {
            limit = { var:xfm_last_date < var:xfm_calendar_observed_date var:xfm_calendar_processed_in_callback < 7 }
            set_variable = { name = xfm_calendar_due_date value = { value = var:xfm_last_date add = 1 } }
            set_variable = { name = xfm_calendar_recovery value = 0 }
            if = {
                limit = { var:xfm_calendar_due_date < var:xfm_calendar_observed_date }
                set_variable = { name = xfm_calendar_recovery value = 1 }
            }
            xfm_finance_daily_tick_effect = yes
            change_variable = { name = xfm_calendar_processed_in_callback add = 1 }
        }
        set_variable = { name = xfm_calendar_backlog value = { value = var:xfm_calendar_observed_date subtract = var:xfm_last_date min = 0 } }
        set_variable = { name = xfm_calendar_recovery value = 0 }
    }
}

xfm_finance_daily_tick_effect = {
    set_variable = { name = xfm_calendar_in_flight value = 1 }
    set_variable = { name = xfm_last_date value = var:xfm_calendar_due_date }
    change_variable = { name = xfm_day add = 1 }
    set_variable = { name = xfm_new_listings_today value = 0 }
    if = { limit = { NOT = { var:xfm_a0_migration_scanned = 1 } } xfm_finance_migrate_a0_effect = yes }
    # Each overdue period invokes the real obligation chain. Current world state is
    # observed NOW; recovery does not certify historical price or AI opportunities.
    xfm_spot_daily_effect = yes
    xfm_oracle_daily_effect = yes
    xfm_finance_capitalize_default_fund_effect = yes
    xfm_futures_daily_effect = yes
    xfm_coin_daily_effect = yes
    xfm_listing_daily_effect = yes
    xfm_debt_daily_effect = yes
    xfm_news_daily_effect = yes
    xfm_finance_run_ai_budget_effect = yes
    set_variable = { name = xfm_calendar_in_flight value = 0 }
    # No observer callback for deferred periods. The normal current tail retains
    # the actual gap in financial-day/current-date samples instead of backfill.
    if = {
        limit = { var:xfm_calendar_recovery = 0 exists = story_owner }
        story_owner = { trigger_event = { on_action = xfm_finance_daily_observation_on_action } }
    }
}

# Latest observed forward gaps are independent raw/helper diagnostics, never tick admission.
xfm_finance_daily_raw_gap_effect = {
    if = {
        limit = { NOT = { has_variable = xfm_daily_raw_gap_count } }
        set_variable = { name = xfm_daily_raw_gap_count value = 1 }
        set_variable = { name = xfm_daily_raw_gap_count_saturated value = 0 }
    }
    else_if = {
        limit = { var:xfm_daily_raw_gap_count < 1000000000 }
        change_variable = { name = xfm_daily_raw_gap_count add = 1 }
    }
    else = { set_variable = { name = xfm_daily_raw_gap_count_saturated value = 1 } }
    set_variable = { name = xfm_daily_raw_last_gap_prev_date value = var:xfm_daily_raw_script_date }
    set_variable = { name = xfm_daily_raw_last_gap_date value = current_date }
    set_variable = { name = xfm_daily_raw_last_gap_missing_dates value = { value = var:xfm_daily_raw_script_date_delta subtract = 1 } }
    set_variable = { name = xfm_daily_raw_last_gap_attempt_count value = var:xfm_daily_raw_attempt_count }
}
xfm_finance_daily_attempt_gap_effect = {
    if = {
        limit = { NOT = { has_variable = xfm_daily_attempt_gap_count } }
        set_variable = { name = xfm_daily_attempt_gap_count value = 1 }
        set_variable = { name = xfm_daily_attempt_gap_count_saturated value = 0 }
    }
    else_if = {
        limit = { var:xfm_daily_attempt_gap_count < 1000000000 }
        change_variable = { name = xfm_daily_attempt_gap_count add = 1 }
    }
    else = { set_variable = { name = xfm_daily_attempt_gap_count_saturated value = 1 } }
    set_variable = { name = xfm_daily_attempt_last_gap_prev_date value = var:xfm_daily_attempt_script_date }
    set_variable = { name = xfm_daily_attempt_last_gap_date value = current_date }
    set_variable = { name = xfm_daily_attempt_last_gap_missing_dates value = { value = var:xfm_daily_attempt_script_date_delta subtract = 1 } }
    set_variable = { name = xfm_daily_attempt_last_gap_attempt_count value = var:xfm_daily_attempt_count }
}
