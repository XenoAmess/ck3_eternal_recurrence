# Character entries. Registration and listing are distinct; no projected subscription is cash.
xfm_issuer_register_micro_effect = {
    set_variable = { name = xfm_requested_tier value = 1 }
    if = { limit = { NOT = { has_variable = xfm_requested_funding } } set_variable = { name = xfm_requested_funding value = 20 } }
    xfm_issuer_register_effect = yes
}
xfm_issuer_register_standard_effect = {
    set_variable = { name = xfm_requested_tier value = 2 }
    if = { limit = { NOT = { has_variable = xfm_requested_funding } } set_variable = { name = xfm_requested_funding value = 200 } }
    xfm_issuer_register_effect = yes
}
xfm_issuer_register_effect = {
    xfm_issuer_receipt_clear_effect = yes
    set_variable = { name = xfm_issuer_result value = 0 }
    if = {
        limit = { xfm_issuer_can_register_trigger = yes }
        save_scope_as = xfm_offer_applicant
        xfm_finance_ensure_account_effect = yes
        var:xfm_account = {
            set_variable = { name = xfm_request_cents value = 100 }
            if = { limit = { scope:xfm_offer_applicant.var:xfm_requested_tier = 2 } set_variable = { name = xfm_request_cents value = 500 } }
            xfm_issuer_receipt_begin_effect = yes
            xfm_finance_ensure_funds_effect = yes
        }
        if = {
            limit = { var:xfm_account = { xfm_finance_account_usable_trigger = yes var:xfm_free_cents >= var:xfm_request_cents } }
            xfm_finance_create_coin_effect = yes
            if = {
                limit = { var:xfm_result = 1 exists = scope:xfm_coin }
                set_variable = { name = xfm_pending_offer value = scope:xfm_coin }
                set_variable = { name = xfm_selected_coin value = scope:xfm_coin }
                scope:xfm_coin = { save_scope_as = xfm_offer_new_coin }
                var:xfm_account = {
                    xfm_issuer_receipt_before_fee_effect = yes
                    xfm_finance_fee_effect = yes
                    xfm_issuer_receipt_after_fee_effect = yes
                }
                global_var:xfm_exchange = {
                    xfm_finance_create_account_effect = yes
                    scope:xfm_offer_new_coin = { set_variable = { name = xfm_issuer_account value = scope:xfm_created_account } }
                    xfm_finance_create_account_effect = yes
                    scope:xfm_offer_new_coin = { set_variable = { name = xfm_founder_account value = scope:xfm_created_account } }
                }
                scope:xfm_offer_new_coin = { xfm_issuer_initialize_effect = yes }
                if = {
                    limit = { scope:xfm_offer_new_coin = { var:xfm_supply_units = 15000 } }
                    xfm_issuer_bind_effect = yes
                    scope:xfm_offer_new_coin = { xfm_issuer_public_ipo_enqueue_effect = yes }
                    set_variable = { name = xfm_issue_cooldown days = xfm_config_issuance_cooldown_days_value }
                    set_variable = { name = xfm_issuer_result value = 1 }
                    scope:xfm_offer_new_coin = { xfm_issuer_receipt_commit_effect = yes }
                }
                else = { scope:xfm_offer_new_coin = { set_variable = { name = xfm_construction_pending value = 1 } } }
            }
        }
    }
    xfm_issuer_receipt_clear_effect = yes
}

                set_variable = { name = xfm_price_directory_origin_v1 value = 1 }
            }
        }
    }

xfm_issuer_bind_effect = {
    if = { limit = { var:xfm_requested_entity_type = 1 } set_variable = { name = xfm_personal_coin value = scope:xfm_offer_new_coin } scope:xfm_offer_new_coin = { set_variable = { name = xfm_bound_entity value = scope:xfm_offer_applicant } } }
    if = { limit = { var:xfm_requested_entity_type = 2 } house = { set_variable = { name = xfm_entity_coin value = scope:xfm_offer_new_coin } set_variable = { name = xfm_issue_cooldown days = xfm_config_issuance_cooldown_days_value } save_scope_as = xfm_offer_bound } scope:xfm_offer_new_coin = { set_variable = { name = xfm_bound_entity value = scope:xfm_offer_bound } } }
    if = { limit = { var:xfm_requested_entity_type = 3 } dynasty = { set_variable = { name = xfm_entity_coin value = scope:xfm_offer_new_coin } set_variable = { name = xfm_issue_cooldown days = xfm_config_issuance_cooldown_days_value } save_scope_as = xfm_offer_bound } scope:xfm_offer_new_coin = { set_variable = { name = xfm_bound_entity value = scope:xfm_offer_bound } } }
    if = { limit = { var:xfm_requested_entity_type = 4 } primary_title = { set_variable = { name = xfm_entity_coin value = scope:xfm_offer_new_coin } set_variable = { name = xfm_issue_cooldown days = xfm_config_issuance_cooldown_days_value } save_scope_as = xfm_offer_bound } scope:xfm_offer_new_coin = { set_variable = { name = xfm_bound_entity value = scope:xfm_offer_bound } } }
    if = { limit = { var:xfm_requested_entity_type = 5 } faith = { set_variable = { name = xfm_entity_coin value = scope:xfm_offer_new_coin } set_variable = { name = xfm_issue_cooldown days = xfm_config_issuance_cooldown_days_value } save_scope_as = xfm_offer_bound } scope:xfm_offer_new_coin = { set_variable = { name = xfm_bound_entity value = scope:xfm_offer_bound } } }
    if = { limit = { var:xfm_requested_entity_type >= 6 } scope:xfm_offer_new_coin = { set_variable = { name = xfm_bound_entity value = scope:xfm_offer_new_coin } } }
}
