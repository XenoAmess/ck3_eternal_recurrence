# Explicit UI refresh only. This writes UI references/cursors, never financial state.
xfm_market_ui_portfolio_pages_effect = {
 save_scope_as = xfm_ui_page_actor
 remove_variable = xfm_ui_portfolio_account
 clear_variable_list = xfm_ui_positions_page
 clear_variable_list = xfm_ui_orders_page
 clear_variable_list = xfm_ui_history_page
 if = {
  limit = { xfm_market_row_owned_account_trigger = yes }
  set_variable = { name = xfm_ui_portfolio_account value = var:xfm_active_account }

  if = { limit = { NOT = { has_variable = xfm_ui_positions_offset } } set_variable = { name = xfm_ui_positions_offset value = 0 } }
  set_variable = { name = xfm_ui_positions_page_offset value = var:xfm_ui_positions_offset }
  set_variable = { name = xfm_ui_page_scan value = 0 }
  var:xfm_active_account = {
   every_in_list = {
    variable = xfm_positions
    save_scope_as = xfm_ui_page_item
    scope:xfm_ui_page_actor = {
     change_variable = { name = xfm_ui_page_scan add = 1 }
     if = {
      limit = { var:xfm_ui_page_scan > var:xfm_ui_positions_offset var:xfm_ui_page_scan <= xfm_market_ui_positions_page_end_value }
      add_to_variable_list = { name = xfm_ui_positions_page target = scope:xfm_ui_page_item }
     }
    }
   }
  }

  if = { limit = { NOT = { has_variable = xfm_ui_orders_offset } } set_variable = { name = xfm_ui_orders_offset value = 0 } }
  set_variable = { name = xfm_ui_orders_page_offset value = var:xfm_ui_orders_offset }
  set_variable = { name = xfm_ui_page_scan value = 0 }
  var:xfm_active_account = {
   every_in_list = {
    variable = xfm_orders
    save_scope_as = xfm_ui_page_item
    scope:xfm_ui_page_actor = {
     change_variable = { name = xfm_ui_page_scan add = 1 }
     if = {
      limit = { var:xfm_ui_page_scan > var:xfm_ui_orders_offset var:xfm_ui_page_scan <= xfm_market_ui_orders_page_end_value }
      add_to_variable_list = { name = xfm_ui_orders_page target = scope:xfm_ui_page_item }
     }
    }
   }
  }

  if = { limit = { NOT = { has_variable = xfm_ui_history_offset } } set_variable = { name = xfm_ui_history_offset value = 0 } }
  set_variable = { name = xfm_ui_history_page_offset value = var:xfm_ui_history_offset }
  set_variable = { name = xfm_ui_page_scan value = 0 }
  var:xfm_active_account = {
   every_in_list = {
    variable = xfm_order_history
    save_scope_as = xfm_ui_page_item
    scope:xfm_ui_page_actor = {
     change_variable = { name = xfm_ui_page_scan add = 1 }
     if = {
      limit = { var:xfm_ui_page_scan > var:xfm_ui_history_offset var:xfm_ui_page_scan <= xfm_market_ui_history_page_end_value }
      add_to_variable_list = { name = xfm_ui_history_page target = scope:xfm_ui_page_item }
     }
    }
   }
  }
 }
 remove_variable = xfm_ui_page_scan
 clear_saved_scope = xfm_ui_page_item
 clear_saved_scope = xfm_ui_page_actor
}
