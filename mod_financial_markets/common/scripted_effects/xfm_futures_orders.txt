xfm_futures_submit_effect = {

 save_scope_as = xfm_fut_actor

 if = {
  limit = {
   xfm_config_futures_open_trigger = yes
   exists = var:xfm_futures_contract exists = var:xfm_selected_coin
   var:xfm_futures_contract.var:xfm_coin = var:xfm_selected_coin

   }
  }
  if = { limit = { is_ai = yes } xfm_finance_ensure_account_effect = yes }
  else = { xfm_finance_select_account_effect = yes }
  scope:xfm_account = { xfm_finance_refresh_beneficiary_effect = yes save_scope_as = xfm_fut_account }
  var:xfm_futures_contract = { save_scope_as = xfm_fut_contract }

   set_variable = { name = xfm_band_high value = xfm_futures_band_high_value }
  }
  if = {
   limit = {
    scope:xfm_fut_account = { xfm_finance_account_usable_trigger = yes }
    xfm_futures_submit_capacity_trigger = yes
    scope:xfm_fut_account.var:xfm_open_positions_count < xfm_futures_account_position_capacity_value
    global_var:xfm_exchange.var:xfm_futures_open_positions_count < xfm_futures_global_position_capacity_value

    OR = { scope:xfm_fut_account = { NOT = { has_variable = xfm_futures_pending_count } } scope:xfm_fut_account.var:xfm_futures_pending_count = 0 }
    var:xfm_futures_limit_cents >= scope:xfm_fut_contract.var:xfm_band_low
    var:xfm_futures_limit_cents <= scope:xfm_fut_contract.var:xfm_band_high
   }
   # Reject related crossings before any reservation or first partial fill.
   set_variable = { name = xfm_futures_related_cross value = 0 }
   scope:xfm_fut_contract = {
    every_in_list = {
     variable = xfm_orders
     limit = {
      xfm_futures_order_open_trigger = yes
      NOT = { var:xfm_side = scope:xfm_fut_actor.var:xfm_futures_side }

       AND = { scope:xfm_fut_actor.var:xfm_futures_side = 1 var:xfm_limit_cents <= scope:xfm_fut_actor.var:xfm_futures_limit_cents }
       AND = { scope:xfm_fut_actor.var:xfm_futures_side = 2 var:xfm_limit_cents >= scope:xfm_fut_actor.var:xfm_futures_limit_cents }
      }
      OR = {
       var:xfm_account = scope:xfm_fut_account
       var:xfm_account.var:xfm_holder = scope:xfm_fut_account.var:xfm_holder
       AND = {
        exists = var:xfm_account.var:xfm_beneficial_owner
        exists = scope:xfm_fut_account.var:xfm_beneficial_owner
        var:xfm_account.var:xfm_beneficial_owner = scope:xfm_fut_account.var:xfm_beneficial_owner
       }
      }
     }
     scope:xfm_fut_actor = { set_variable = { name = xfm_futures_related_cross value = 1 } }
    }
   }
   if = {
    limit = { var:xfm_futures_related_cross = 0 }

     xfm_futures_prepare_opening_admission_effect = yes

# Actor scope, opening submit only. Reduce/forced exit/settlement/cancel never call this gate.
xfm_futures_prepare_opening_admission_effect = {
 set_variable = { name = xfm_futures_opening_admitted value = 0 }
 if = {
  limit = { exists = global_var:xfm_exchange }
  global_var:xfm_exchange = {
   # All three headers absent means the current intrusive queue has no reachable node.
   # This is not a zero count of legacy histories, retained claims or resident stories.
   if = {
    limit = { NOT = { has_variable = xfm_futures_archive_count } NOT = { has_variable = xfm_futures_archive_head } NOT = { has_variable = xfm_futures_archive_tail } }
    set_variable = { name = xfm_futures_archive_count value = 0 }
   }
   if = {
    limit = { has_variable = xfm_day has_variable = xfm_futures_archive_count }
    if = {
     limit = { var:xfm_day >= 0 var:xfm_day = xfm_futures_opening_day_floor_value var:xfm_futures_archive_count >= 0 var:xfm_futures_archive_count = xfm_futures_opening_archive_floor_value }
     # Old exchange: unknown earlier work is never backfilled as zero. Seal this day;
     # only the next actual financial day can create the first new allowance.
     if = {
      limit = { NOT = { has_variable = xfm_futures_opening_budget_v1 } NOT = { has_variable = xfm_futures_opening_budget_day } NOT = { has_variable = xfm_futures_opening_allocated_today } }
      set_variable = { name = xfm_futures_opening_budget_day value = var:xfm_day }
      set_variable = { name = xfm_futures_opening_allocated_today value = xfm_futures_opening_daily_capacity_value }
      set_variable = { name = xfm_futures_opening_budget_v1 value = 1 }
     }
     if = {
      limit = { has_variable = xfm_futures_opening_budget_v1 has_variable = xfm_futures_opening_budget_day has_variable = xfm_futures_opening_allocated_today }
      if = {
       limit = {
        var:xfm_futures_opening_budget_v1 = 1
        var:xfm_futures_opening_budget_day >= 0
        var:xfm_futures_opening_budget_day = xfm_futures_opening_budget_day_floor_value
        var:xfm_futures_opening_budget_day <= var:xfm_day
        var:xfm_futures_opening_allocated_today >= 0
        var:xfm_futures_opening_allocated_today = xfm_futures_opening_allocated_floor_value
        var:xfm_futures_opening_allocated_today <= xfm_futures_opening_daily_capacity_value
       }
       if = {
        limit = { var:xfm_futures_opening_budget_day < var:xfm_day }
        set_variable = { name = xfm_futures_opening_budget_day value = var:xfm_day }
        set_variable = { name = xfm_futures_opening_allocated_today value = 0 }
       }
       # This existing counter belongs to the intrusive archive queue, not all stories.
       # Missing/incoherent queue headers fail closed; no legacy resident count is invented.
       if = {
        limit = {
         OR = {
          AND = { var:xfm_futures_archive_count = 0 NOT = { exists = var:xfm_futures_archive_head } NOT = { exists = var:xfm_futures_archive_tail } }
          AND = { var:xfm_futures_archive_count > 0 exists = var:xfm_futures_archive_head exists = var:xfm_futures_archive_tail }
         }
        }
        if = {
         limit = { var:xfm_futures_archive_count = 0 }
         xfm_futures_finish_opening_admission_effect = yes
        }
        else = {
         if = {
          limit = {
           var:xfm_futures_archive_head = { OR = { story_type = xfm_futures_order story_type = xfm_futures_position } }
           var:xfm_futures_archive_tail = { OR = { story_type = xfm_futures_order story_type = xfm_futures_position } }
          }
          # Existing real cleanup visits at most64 linked nodes; today's/claim sources survive.
          if = { limit = { var:xfm_futures_archive_count > 2048 } xfm_futures_archive_prune_effect = yes }
          xfm_futures_finish_opening_admission_effect = yes
         }
        }
       }
      }
     }
    }
   }
  }
 }
}

   xfm_futures_cancel_order_effect = yes
