# 7 effects. Supply changes require issuer authorization at the caller.
xfm_finance_lookup_position_effect = {
 save_scope_as = xfm_fin_lookup_account
 clear_saved_scope = xfm_position
 remove_variable = xfm_result_position
 set_variable = { name = xfm_result value = 0 }
 if = {
  limit = { story_type = xfm_finance_account exists = var:xfm_request_coin }
  random_in_list = {
   variable = xfm_positions
   limit = { story_type = xfm_finance_position var:xfm_coin = scope:xfm_fin_lookup_account.var:xfm_request_coin }
   save_scope_as = xfm_position
  }
  if = {
   limit = { exists = scope:xfm_position }
   set_variable = { name = xfm_result_position value = scope:xfm_position }
   set_variable = { name = xfm_result value = 1 }
  }
 }
}
xfm_finance_ensure_position_effect = {
 save_scope_as = xfm_fin_position_account
 xfm_finance_lookup_position_effect = yes
 if = {
  limit = {
   NOT = { exists = scope:xfm_position }
   exists = var:xfm_request_coin
   var:xfm_request_coin = { story_type = xfm_finance_coin }
  }
  global_var:xfm_exchange.story_owner = { create_story = { type = xfm_finance_position save_scope_as = xfm_position } }
  if = {
   limit = { exists = scope:xfm_position scope:xfm_position = { story_type = xfm_finance_position } }
   scope:xfm_position = {
    set_variable = { name = xfm_account value = scope:xfm_fin_position_account }
    set_variable = { name = xfm_coin value = scope:xfm_fin_position_account.var:xfm_request_coin }
    set_variable = { name = xfm_quantity_units value = 0 }
    set_variable = { name = xfm_reserved_units value = 0 }
    set_variable = { name = xfm_counted value = 0 }
   }
   add_to_variable_list = { name = xfm_positions target = scope:xfm_position }
   var:xfm_request_coin = { add_to_variable_list = { name = xfm_holders target = scope:xfm_position } }
   set_variable = { name = xfm_result_position value = scope:xfm_position }
   set_variable = { name = xfm_result value = 1 }
  }
 }
}
xfm_finance_sync_position_count_effect = {
 if = {
  limit = { var:xfm_quantity_units > 0 var:xfm_counted = 0 }
  var:xfm_account = { change_variable = { name = xfm_holdings_count add = 1 } }
  set_variable = { name = xfm_counted value = 1 }
 }
 else_if = {
  limit = { var:xfm_quantity_units = 0 var:xfm_counted = 1 }
  var:xfm_account = { change_variable = { name = xfm_holdings_count subtract = 1 } }
  set_variable = { name = xfm_counted value = 0 }
 }
}
xfm_finance_mint_effect = {
 save_scope_as = xfm_fin_mint_coin
 set_variable = { name = xfm_result value = 0 }
 if = {
  limit = { story_type = xfm_finance_coin xfm_finance_valid_units_trigger = yes exists = var:xfm_request_target var:xfm_request_target = { story_type = xfm_finance_account } }
  var:xfm_request_target = {
   set_variable = { name = xfm_request_coin value = scope:xfm_fin_mint_coin }
   xfm_finance_ensure_position_effect = yes
  }
  if = {
   limit = { exists = scope:xfm_position scope:xfm_position = { var:xfm_coin = scope:xfm_fin_mint_coin var:xfm_account = scope:xfm_fin_mint_coin.var:xfm_request_target } }
   scope:xfm_position = {
    change_variable = { name = xfm_quantity_units add = scope:xfm_fin_mint_coin.var:xfm_request_units }
    xfm_finance_sync_position_count_effect = yes
   }
   change_variable = { name = xfm_supply_units add = var:xfm_request_units }
   set_variable = { name = xfm_result value = 1 }
  }
 }
}
xfm_finance_burn_effect = {
 save_scope_as = xfm_fin_burn_coin
 set_variable = { name = xfm_result value = 0 }
 if = {
  limit = {
   story_type = xfm_finance_coin xfm_finance_valid_units_trigger = yes
   var:xfm_supply_units >= var:xfm_request_units exists = var:xfm_request_target
   var:xfm_request_target = { xfm_finance_account_usable_trigger = yes }
  }
  var:xfm_request_target = {
   set_variable = { name = xfm_request_coin value = scope:xfm_fin_burn_coin }
   xfm_finance_lookup_position_effect = yes
  }
  if = {
   limit = { exists = scope:xfm_position scope:xfm_position = { xfm_finance_position_available_value >= scope:xfm_fin_burn_coin.var:xfm_request_units } }
   scope:xfm_position = {
    change_variable = { name = xfm_quantity_units subtract = scope:xfm_fin_burn_coin.var:xfm_request_units }
    xfm_finance_sync_position_count_effect = yes
   }
   change_variable = { name = xfm_supply_units subtract = var:xfm_request_units }
   set_variable = { name = xfm_result value = 1 }
  }
 }
}
xfm_finance_transfer_coin_effect = {
 save_scope_as = xfm_fin_coin_sender
 set_variable = { name = xfm_result value = 0 }
 clear_saved_scope = xfm_fin_source_position
 clear_saved_scope = xfm_fin_target_position
 if = {
  limit = {
   xfm_finance_account_usable_trigger = yes xfm_finance_valid_units_trigger = yes
   exists = var:xfm_request_target var:xfm_request_target != this
   var:xfm_request_target = { story_type = xfm_finance_account }
   exists = var:xfm_request_coin var:xfm_request_coin = { story_type = xfm_finance_coin }
  }
  xfm_finance_lookup_position_effect = yes
  if = {
   limit = { exists = scope:xfm_position scope:xfm_position = { xfm_finance_position_available_value >= scope:xfm_fin_coin_sender.var:xfm_request_units } }
   scope:xfm_position = { save_scope_as = xfm_fin_source_position }
   var:xfm_request_target = {
    set_variable = { name = xfm_request_coin value = scope:xfm_fin_coin_sender.var:xfm_request_coin }
    xfm_finance_ensure_position_effect = yes
   }
   if = {
    limit = { exists = scope:xfm_position scope:xfm_position = { var:xfm_account = scope:xfm_fin_coin_sender.var:xfm_request_target var:xfm_coin = scope:xfm_fin_coin_sender.var:xfm_request_coin } }
    scope:xfm_position = { save_scope_as = xfm_fin_target_position }
    scope:xfm_fin_source_position = {
     change_variable = { name = xfm_quantity_units subtract = scope:xfm_fin_coin_sender.var:xfm_request_units }
     xfm_finance_sync_position_count_effect = yes
    }
    scope:xfm_fin_target_position = {
     change_variable = { name = xfm_quantity_units add = scope:xfm_fin_coin_sender.var:xfm_request_units }
     xfm_finance_sync_position_count_effect = yes
    }
    set_variable = { name = xfm_result value = 1 }
   }
  }
 }
}
xfm_finance_rehome_story_effect = {
 clear_saved_scope = xfm_fin_new_custodian
 clear_saved_scope = xfm_fin_old_custodian
 story_owner = { save_scope_as = xfm_fin_old_custodian }
 if = {
  limit = {
   exists = global_var:xfm_exchange.story_owner
   global_var:xfm_exchange.story_owner = { is_alive = yes this != scope:xfm_fin_old_custodian NOT = { has_character_flag = about_to_die_but_not_dead_yet_flag } }
  }
  make_story_owner = global_var:xfm_exchange.story_owner
 }
 else = {
  random_ruler = {
   limit = { is_alive = yes this != scope:xfm_fin_old_custodian NOT = { has_character_flag = about_to_die_but_not_dead_yet_flag } }
   save_scope_as = xfm_fin_new_custodian
  }
  if = { limit = { exists = scope:xfm_fin_new_custodian } make_story_owner = scope:xfm_fin_new_custodian }
 }
}
