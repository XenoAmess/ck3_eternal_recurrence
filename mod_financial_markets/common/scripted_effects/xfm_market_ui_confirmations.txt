# 4 event-context helpers. Saved scopes/values belong to each confirmation event.
xfm_market_ui_capture_coin_effect = {
    clear_saved_scope = xfm_ui_confirmation_coin
    save_scope_value_as = { name = xfm_ui_confirmation_ready value = 0 }
    if = {
        limit = { exists = var:xfm_selected_coin var:xfm_selected_coin = { story_type = xfm_finance_coin } }
        var:xfm_selected_coin = {
            save_scope_as = xfm_ui_confirmation_coin
            save_scope_value_as = { name = xfm_ui_confirmation_maker_cents value = xfm_issuer_maker_value }
            save_scope_value_as = { name = xfm_ui_confirmation_supply value = var:xfm_supply_units }
            save_scope_value_as = { name = xfm_ui_confirmation_maximum_supply value = var:xfm_maximum_supply_units }
            save_scope_value_as = { name = xfm_ui_confirmation_concealed_units value = xfm_issuer_mint_units_value }
        }
        save_scope_value_as = { name = xfm_ui_confirmation_coin_id value = var:xfm_selected_coin.var:xfm_coin_id }
        save_scope_value_as = { name = xfm_ui_confirmation_lots value = var:xfm_order_lots }
        save_scope_value_as = { name = xfm_ui_confirmation_limit_cents value = var:xfm_order_limit_cents }
        save_scope_value_as = { name = xfm_ui_confirmation_ioc value = var:xfm_order_ioc }
        save_scope_value_as = { name = xfm_ui_confirmation_ready value = 1 }
    }
}
xfm_market_ui_capture_wallet_effect = {
    clear_saved_scope = xfm_ui_confirmation_account
    xfm_finance_select_account_effect = yes
    if = {
        limit = { exists = scope:xfm_account }
        scope:xfm_account = { save_scope_as = xfm_ui_confirmation_account }
        save_scope_value_as = { name = xfm_ui_confirmation_account_id value = scope:xfm_account.var:xfm_account_id }
    }
}
xfm_market_ui_restore_coin_effect = {
    set_variable = { name = xfm_selected_coin value = scope:xfm_ui_confirmation_coin }
    set_variable = { name = xfm_order_lots value = scope:xfm_ui_confirmation_lots }
    set_variable = { name = xfm_order_limit_cents value = scope:xfm_ui_confirmation_limit_cents }
    set_variable = { name = xfm_order_ioc value = scope:xfm_ui_confirmation_ioc }
}
xfm_market_ui_restore_wallet_effect = {
    set_variable = { name = xfm_active_account value = scope:xfm_ui_confirmation_account }
}
