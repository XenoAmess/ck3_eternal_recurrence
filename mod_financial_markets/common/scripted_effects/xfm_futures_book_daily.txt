xfm_futures_maintain_orders_effect = {
 save_scope_as = xfm_fut_maintain_contract
 if = {
  limit = { exists = var:xfm_coin var:xfm_coin.var:xfm_mark_cents > 0 }
  set_variable = { name = xfm_band_low value = xfm_futures_band_low_value }
  set_variable = { name = xfm_band_high value = xfm_futures_band_high_value }
 }
 every_in_list = {
  variable = xfm_orders
  limit = { xfm_futures_order_open_trigger = yes }
  if = {
   limit = { NOT = { has_variable = xfm_reduce_order } }
   if = {
    limit = {
     OR = {
      NOT = { xfm_config_futures_open_trigger = yes }
      var:xfm_account.var:xfm_account_frozen = 1
      scope:xfm_fut_maintain_contract = { NOT = { xfm_futures_live_oracle_trigger = yes } }
      scope:xfm_fut_maintain_contract.var:xfm_coin.var:xfm_derivatives_reduce_only = 1
      var:xfm_leverage > scope:xfm_fut_maintain_contract.var:xfm_coin.var:xfm_max_leverage
      var:xfm_limit_cents < scope:xfm_fut_maintain_contract.var:xfm_band_low
      var:xfm_limit_cents > scope:xfm_fut_maintain_contract.var:xfm_band_high
      AND = { scope:xfm_fut_maintain_contract.var:xfm_perpetual = 0 scope:xfm_fut_maintain_contract.var:xfm_expiry_day <= global_var:xfm_exchange.var:xfm_day }
     }
    }
    xfm_futures_cancel_risk_effect = yes
   }
  }
  else_if = {
   limit = { OR = { NOT = { exists = var:xfm_close_position } var:xfm_close_position = { NOT = { xfm_futures_position_open_trigger = yes } } } }
   xfm_futures_cancel_with_reason_effect = { REASON = 11 }
  }
  else_if = {
   limit = { var:xfm_remaining_lots > var:xfm_close_position.var:xfm_lots }
   set_variable = { name = xfm_remaining_lots value = var:xfm_close_position.var:xfm_lots }
   set_variable = { name = xfm_remaining_notional value = xfm_futures_remaining_notional_value }
   set_variable = { name = xfm_remaining_fee value = xfm_futures_remaining_fee_value }
   set_variable = { name = xfm_release_excess value = { value = var:xfm_reservation.var:xfm_remaining_cents subtract = var:xfm_remaining_fee floor = yes min = 0 } }
   save_scope_as = xfm_fut_resizing_order
   if = {
    limit = { var:xfm_release_excess > 0 }
    var:xfm_reservation = {
     set_variable = { name = xfm_request_cents value = scope:xfm_fut_resizing_order.var:xfm_release_excess }
     xfm_finance_release_effect = yes
    }
   }
  }
 }
}
