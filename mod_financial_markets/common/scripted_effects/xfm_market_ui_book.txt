# One read projection effect. These rows are individual orders, not aggregated levels.
xfm_market_ui_book_effect = {
    save_scope_as = xfm_ui_book_actor
    clear_variable_list = xfm_ui_bids
    clear_variable_list = xfm_ui_asks
    if = {
        limit = { exists = var:xfm_selected_coin var:xfm_selected_coin = { story_type = xfm_finance_coin } }
        var:xfm_selected_coin = {
            ordered_in_list = {
                variable = xfm_spot_orders
                limit = { var:xfm_status = 1 var:xfm_side = 1 var:xfm_remaining_lots > 0 }
                order_by = var:xfm_limit_cents
                max = 5
                check_range_bounds = no
                save_scope_as = xfm_ui_book_order
                scope:xfm_ui_book_actor = { add_to_variable_list = { name = xfm_ui_bids target = scope:xfm_ui_book_order } }
            }
            ordered_in_list = {
                variable = xfm_spot_orders
                limit = { var:xfm_status = 1 var:xfm_side = 2 var:xfm_remaining_lots > 0 }
                order_by = { value = 0 subtract = var:xfm_limit_cents }
                max = 5
                check_range_bounds = no
                save_scope_as = xfm_ui_book_order
                scope:xfm_ui_book_actor = { add_to_variable_list = { name = xfm_ui_asks target = scope:xfm_ui_book_order } }
            }
        }
    }
    xfm_market_ui_futures_book_effect = yes
}
