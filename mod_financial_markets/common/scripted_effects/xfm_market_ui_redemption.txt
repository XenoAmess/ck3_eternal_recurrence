# Character scope. Independent defaults and immutable event contexts for this service.
xfm_redemption_ui_init_effect = {
 if = { limit = { exists = var:xfm_selected_coin var:xfm_selected_coin = { story_type = xfm_finance_coin } } var:xfm_selected_coin = { xfm_redemption_refresh_effect = yes } }
 if = { limit = { NOT = { has_variable = xfm_redemption_price_cents } } set_variable = { name = xfm_redemption_price_cents value = 20 } }
 if = { limit = { NOT = { has_variable = xfm_redemption_budget_cents } } set_variable = { name = xfm_redemption_budget_cents value = 1000 } }
 if = { limit = { NOT = { has_variable = xfm_redemption_duration_days } } set_variable = { name = xfm_redemption_duration_days value = 30 } }
 if = { limit = { NOT = { has_variable = xfm_redemption_lots } } set_variable = { name = xfm_redemption_lots value = 1 } }
}
xfm_redemption_prepare_publish_effect = {
 save_scope_value_as = { name = xfm_red_publish_ready value = 0 }
 if = { limit = { exists = var:xfm_selected_coin } var:xfm_selected_coin = { xfm_redemption_refresh_effect = yes } }
 if = {
  limit = { xfm_redemption_controller_trigger = yes }
  save_scope_as = xfm_red_publish_signer
  var:xfm_selected_coin = {
   save_scope_as = xfm_red_publish_selected_coin
   var:xfm_issuer_account = { save_scope_as = xfm_red_publish_source }
  }
  save_scope_value_as = { name = xfm_red_publish_coin_id value = var:xfm_selected_coin.var:xfm_coin_id }
  save_scope_value_as = { name = xfm_red_publish_source_id value = var:xfm_selected_coin.var:xfm_issuer_account.var:xfm_account_id }
  save_scope_value_as = { name = xfm_red_publish_price value = var:xfm_redemption_price_cents }
  save_scope_value_as = { name = xfm_red_publish_budget value = var:xfm_redemption_budget_cents }
  save_scope_value_as = { name = xfm_red_publish_expiry value = xfm_redemption_input_expiry_value }
  save_scope_value_as = { name = xfm_red_publish_ready value = 1 }
 }
}
xfm_redemption_confirm_publish_effect = {
 set_variable = { name = xfm_redemption_result value = 0 }
 set_variable = { name = xfm_redemption_last_action_result value = 0 }
 if = {
  limit = {
   scope:xfm_red_publish_ready = 1 this = scope:xfm_red_publish_signer
   var:xfm_selected_coin = scope:xfm_red_publish_selected_coin
   scope:xfm_red_publish_selected_coin = {
    var:xfm_coin_id = scope:xfm_red_publish_coin_id
    var:xfm_issuer_account = scope:xfm_red_publish_source
   }
   scope:xfm_red_publish_source.var:xfm_account_id = scope:xfm_red_publish_source_id
  }
  set_variable = { name = xfm_redemption_price_cents value = scope:xfm_red_publish_price }
  set_variable = { name = xfm_redemption_budget_cents value = scope:xfm_red_publish_budget }
  set_variable = { name = xfm_redemption_expiry_day value = scope:xfm_red_publish_expiry }
  xfm_redemption_publish_effect = yes
 }
}
xfm_redemption_prepare_reclaim_effect = {
 save_scope_value_as = { name = xfm_red_reclaim_ready value = 0 }
 if = { limit = { exists = var:xfm_selected_coin } var:xfm_selected_coin = { xfm_redemption_refresh_effect = yes } }
 if = {
  limit = { xfm_redemption_controller_trigger = yes var:xfm_selected_coin = { exists = var:xfm_redemption_window xfm_redemption_account_trigger = yes } }
  save_scope_as = xfm_red_reclaim_signer
  var:xfm_selected_coin = {
   save_scope_as = xfm_red_reclaim_selected_coin
   var:xfm_redemption_window = { save_scope_as = xfm_red_reclaim_window }
   var:xfm_redemption_account = { save_scope_as = xfm_red_reclaim_source }
   var:xfm_issuer_account = { save_scope_as = xfm_red_reclaim_target }
  }
  save_scope_value_as = { name = xfm_red_reclaim_window_id value = scope:xfm_red_reclaim_window.var:xfm_window_id }
  save_scope_value_as = { name = xfm_red_reclaim_source_id value = scope:xfm_red_reclaim_source.var:xfm_account_id }
  save_scope_value_as = { name = xfm_red_reclaim_target_id value = scope:xfm_red_reclaim_target.var:xfm_account_id }
  save_scope_value_as = { name = xfm_red_reclaim_ready value = 1 }
 }
}
xfm_redemption_confirm_reclaim_effect = {
 set_variable = { name = xfm_redemption_result value = 0 }
 set_variable = { name = xfm_redemption_last_action_result value = 0 }
 if = {
  limit = {
   scope:xfm_red_reclaim_ready = 1 this = scope:xfm_red_reclaim_signer
   var:xfm_selected_coin = scope:xfm_red_reclaim_selected_coin
   scope:xfm_red_reclaim_selected_coin.var:xfm_redemption_window = scope:xfm_red_reclaim_window
   scope:xfm_red_reclaim_selected_coin.var:xfm_redemption_account = scope:xfm_red_reclaim_source
   scope:xfm_red_reclaim_selected_coin.var:xfm_issuer_account = scope:xfm_red_reclaim_target
   scope:xfm_red_reclaim_window.var:xfm_window_id = scope:xfm_red_reclaim_window_id
   scope:xfm_red_reclaim_source.var:xfm_account_id = scope:xfm_red_reclaim_source_id
   scope:xfm_red_reclaim_target.var:xfm_account_id = scope:xfm_red_reclaim_target_id
  }
  xfm_redemption_reclaim_effect = yes
 }
}
