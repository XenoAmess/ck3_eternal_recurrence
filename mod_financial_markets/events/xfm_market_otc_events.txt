namespace = xfm_market
xfm_market.120 = {
    type = character_event
    title = xfm_market_otc_received_title
    desc = xfm_market_otc_received_desc
    theme = stewardship
    left_portrait = scope:xfm_ui_otc_seller
    option = {
        name = xfm_market_otc_accept
        trigger = {
            scope:xfm_ui_otc_seller = { has_variable = xfm_otc_consent var:xfm_otc_offer_nonce = scope:xfm_ui_otc_nonce }
            global_var:xfm_exchange.var:xfm_day < scope:xfm_ui_otc_expiry_day
        }
        if = {
            limit = { is_ai = yes }
            # Revalidate the policy at actual acceptance, after the AI weight was evaluated.
            if = {
                limit = { xfm_otc_ai_eligible_trigger = yes }
                xfm_market_ui_otc_accept_effect = yes
                if = { limit = { var:xfm_issuer_result = 1 } set_variable = { name = xfm_otc_ai_last_buy_day value = global_var:xfm_exchange.var:xfm_day } }
            }
        }
        else = { xfm_market_ui_otc_accept_effect = yes }
        ai_chance = {
            base = 0
            modifier = { add = 10 xfm_otc_ai_fresh_trigger = yes }
            modifier = { add = 5 xfm_otc_ai_stale_trigger = yes }
        }
    }
    option = { name = xfm_market_otc_decline xfm_market_ui_otc_decline_effect = yes ai_chance = { base = 50 } }
}
