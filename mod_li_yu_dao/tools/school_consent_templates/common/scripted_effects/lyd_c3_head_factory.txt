# Authored template. Character ROOT; current faith/rite after an authorized detach or council recognition.
# Native basis: 00_religion_effects.txt:347-380. One owned HoF title; no secular land transfer.
lyd_c3_create_owned_temporal_head_effect = {
    if = {
        limit = {
            is_ai = no
            is_alive = yes
            is_adult = yes
            is_landed = yes
            has_variable = lyd_c3_head_creation_authorized
            religion = religion:confucianism_religion
            faith = { has_doctrine_parameter = temporal_head_of_faith NOT = { exists = religious_head_title } }
            rite = faith.main_rite
        }
        faith = { save_scope_as = my_faith save_scope_as = lyd_c3_factory_faith }
        create_dynamic_title = { tier = duchy name = REL_HEAD_TITLE_NAME adj = REL_HEAD_TITLE_ADJ }
        create_title_and_vassal_change = { type = created save_scope_as = lyd_c3_head_change add_claim_on_loss = no }
        scope:new_title = {
            set_variable = { name = lyd_c2_owned_head_title value = 1 }
            set_variable = { name = lyd_c2_owner_faith value = scope:lyd_c3_factory_faith }
            set_destroy_if_invalid_heir = yes
            set_no_automatic_claims = yes
            set_definitive_form = yes
            set_always_follows_primary_heir = yes
            change_title_holder = { holder = root change = scope:lyd_c3_head_change }
        }
        resolve_title_and_vassal_change = scope:lyd_c3_head_change
        faith = { set_religious_head_title = scope:new_title }
        scope:new_title = { generate_coa = religious_title add_title_law = temporal_head_of_faith_succession_law }
        set_variable = { name = lyd_c3_office_faith value = faith }
        remove_variable = lyd_c3_head_creation_authorized
        debug_log = "LYD_C3_OWNED_TEMPORAL_HEAD_CREATED"
    }
    else = { debug_log = "LYD_C3_HEAD_FACTORY_REJECTED" }
}
