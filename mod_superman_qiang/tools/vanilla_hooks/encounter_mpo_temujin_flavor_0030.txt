mpo_temujin_flavor.0030 = {
	type = character_event
	title = mpo_temujin_flavor.0030.t
	desc = mpo_temujin_flavor.0030.desc
	theme = pregnancy
	left_portrait = {
		character = root
		animation = anger
	}
	right_portrait = {
		character = scope:borte
		animation = worry
	}
	lower_right_portrait = scope:mergid
	
	trigger = {
		NOT = { has_variable = had_mpo_temujin_flavor_0030 } # To make sure it triggers even if you migrate straight away and lose the event time window
		character:172004 = {
			is_alive = yes
			is_spouse_of = root
		}
		OR = {
			dynasty:2001144 = {
				any_dynasty_member = {
					is_male = yes
					is_adult = yes
					is_ruler = yes
				}
			}
			any_neighboring_top_liege_realm_owner = {
				is_male = yes
				is_adult = yes
				is_ai = yes
			}
		}
	}

	immediate = {
		play_music_cue = "mx_cue_personal_negative"
		set_variable = had_mpo_temujin_flavor_0030
		character:172004 = {
			save_scope_as = borte
			if = {
				limit = {
					is_pregnant = yes
				}
				hidden_effect = {
					end_pregnancy = yes
				}
			}
		}
		if = {
			limit = {
				character:303116 = {
					is_alive = yes
				}
			}
			character:303116 = {
				save_scope_as = mergid
			}
		}
		else_if = {
			limit = {
				dynasty:2001144 = {
					any_dynasty_member = {
						is_male = yes
						is_adult = yes
						is_ruler = yes
					}
				}
			}
			dynasty:2001144 = {
				random_dynasty_member = {
					limit = {
						is_male = yes
						is_adult = yes
						is_ruler = yes
					}
					save_scope_as = mergid
				}
			}
		}
		else = {
			random_neighboring_top_liege_realm_owner = {
				limit = {
					is_ai = yes
					is_male = yes
					is_adult = yes
				}
				save_scope_as = mergid
			}
		}
		scope:borte = {
			custom_tooltip = mpo_temujin_flavor.0030.pregnancy_tt
			hidden_effect = {
				make_pregnant = {
					father = scope:mergid
					known_bastard = yes
				}
			}
			set_variable = suppress_bastardy
		}
	}

	option = { # I hate those guys
		name = mpo_temujin_flavor.0030.a
		if = {
			limit = {
				NOT = { has_relation_rival = scope:mergid }
			}
			set_relation_rival = {
				target = scope:mergid
				reason = rival_temujin_mergid
			}
		}
		else_if = {
			limit = {
				NOT = { has_relation_nemesis = scope:mergid }
			}
			set_relation_nemesis = {
				target = scope:mergid
				reason = rival_temujin_mergid
			}
		}
		custom_tooltip = {
			text = mpo_temujin_flavor.0030.a.tt
			add_to_variable_list = {
				name = retaliation_cb_var_list
				target = scope:mergid
			}
		}
		add_dread = medium_dread_gain
		stress_and_fulfillment_impact = {
			vengeful = major_stress_impact_loss
			compassionate = medium_stress_impact_gain
		}

		ai_chance = {
			base = 0 # To prevent AI Temüjin from being murdered immediately
		}
	}

	option = { # Let's forget about that, you are more important
		name = mpo_temujin_flavor.0030.b
		if = {
			limit = {
				NOT = { has_relation_lover = scope:borte }
			}
			set_relation_lover = {
				target = scope:borte
				reason = lover_temujin_borte
			}
		}
		else = {
			scope:borte = {
				add_opinion = {
					target = root
					modifier = grateful_opinion
					opinion = 20
				}
			}
		}
		set_relation_wedding_very_good_fertility = scope:borte
		stress_and_fulfillment_impact = {
			compassionate = major_stress_impact_loss
			vengeful = medium_stress_impact_gain
		}

		ai_chance = {
			base = 100
		}
	}

	option = { # I could not care less
		name = mpo_temujin_flavor.0030.c
		scope:borte = {
			add_opinion = {
				target = root
				modifier = upset_opinion
				opinion = -25
			}
		}
		stress_and_fulfillment_impact = {
			callous = major_stress_impact_loss
			compassionate = major_stress_impact_gain
			vengeful = major_stress_impact_gain
		}
		ai_chance = {
			base = 0
		}
	}
}
