				}
				set_variable = { name = des_preset_fixture_ref_observer value = scope:des_preset_fixture_observer }

					gender_female_chance = 50
					age = 30
					location = root.location
					culture = root.culture
					faith = root.var:des_preset_fixture_source_faith
					rite = root.var:des_preset_fixture_source_rite
					save_scope_as = des_preset_fixture_target
				}
				set_variable = { name = des_preset_fixture_ref_target value = scope:des_preset_fixture_target }
				set_variable = { name = des_preset_fixture_p1_input_target value = var:des_preset_fixture_ref_target }

				set_character_rite = rite:des_procreation_rite
				var:des_preset_fixture_ref_observer = {
					set_character_faith = faith:des_procreation_faith
					set_character_rite = rite:des_procreation_rite
				}
