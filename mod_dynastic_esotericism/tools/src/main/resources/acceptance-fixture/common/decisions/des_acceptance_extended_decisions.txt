				}

				debug_log = "DESAF: TEST PASS unchanged_unconverted_descendant"
			}
			else = { debug_log = "DESAF: TEST FAIL unchanged_unconverted_descendant" }
			scope:des_test_peer = {
				create_cadet_branch = { save_scope_as = des_test_cadet_house }
			}
			set_variable = { name = des_test_cadet_house value = scope:des_test_cadet_house }
			scope:des_test_child = { set_house = scope:des_test_outsider.house }
			if = {
				limit = {
					scope:des_test_peer = { dynasty = scope:des_test_dynasty house = scope:des_test_cadet_house }
					scope:des_test_child = { rite = scope:des_test_primary NOT = { dynasty = scope:des_test_dynasty } }

			trigger_event = { id = des_acceptance_extended.1 days = 1 }
