# Fixture aliases never write des_* product binding/trait/relationship state.
# Disclosed original-game input, not an interaction submission or payment receipt.
# Called only by fresh setup or the explicit first-head/successor edit preparation.
# These three fixture routes supply none of the six optional interaction scopes.
des_acceptance_prepare_original_rite_input_effect = {
    if = {
        limit = {
            can_have_realm_priest_puppet_trigger = yes
            is_incapable = no
            NOT = { exists = cp:councillor_court_chaplain }
        }
        assign_landed_realm_priest_effect = yes
    }
    if = {
        limit = {
            can_have_realm_priest_puppet_trigger = yes
            is_incapable = no
            exists = cp:councillor_court_chaplain
            NOT = { exists = puppet:theological_agent_puppet }
            cp:councillor_court_chaplain ?= {
                is_ai = yes
                is_adult = yes
                NOT = { has_trait = incapable }
                NOT = { exists = puppeteer }
            }
        }
        save_scope_as = actor
        cp:councillor_court_chaplain = { save_scope_as = recipient }
        set_puppet = { target = scope:recipient type = theological_agent_puppet }
        pam_generic_set_puppet_interaction_side_effects = yes
    }
}

# Shared relationship observation; this flag never certifies a native draft Can.
des_acceptance_read_original_rite_relation_effect = {
    save_scope_as = des_rite_input_actor
    remove_character_flag = des_test_original_rite_relation_ready
    if = {
        limit = { exists = cp:councillor_court_chaplain }
        debug_log = "DESAF: PRECONDITION PASS actual_court_chaplain"
    }
    else = { debug_log = "DESAF: NOT_READY actual_court_chaplain" }
    if = {
        limit = { exists = puppet:theological_agent_puppet }
        debug_log = "DESAF: PRECONDITION PASS actual_theological_puppet"
    }
    else = { debug_log = "DESAF: NOT_READY actual_theological_puppet" }
    if = {
        limit = {
            trigger_if = {
                limit = {
                    exists = cp:councillor_court_chaplain
                    cp:councillor_court_chaplain ?= { exists = puppeteer }
                }
                cp:councillor_court_chaplain = { puppeteer = scope:des_rite_input_actor }
            }
            trigger_else = { always = no }
        }
        debug_log = "DESAF: PRECONDITION PASS actual_reverse_puppeteer"
    }
    else = { debug_log = "DESAF: NOT_READY actual_reverse_puppeteer" }
    if = {
        limit = { is_puppet_action_valid = puppet_trigger_can_create_rite }
        debug_log = "DESAF: PRECONDITION PASS actual_puppet_action_valid"
    }
    else = { debug_log = "DESAF: NOT_READY actual_puppet_action_valid" }
    if = {
        limit = {
            can_have_realm_priest_puppet_trigger = yes
            is_incapable = no
            trigger_if = {
                limit = {
                    exists = cp:councillor_court_chaplain
                    exists = puppet:theological_agent_puppet
                }
                puppet:theological_agent_puppet = cp:councillor_court_chaplain
                cp:councillor_court_chaplain = {
                    exists = puppeteer
                    puppeteer ?= scope:des_rite_input_actor
                    is_ai = yes
                    is_adult = yes
                    NOT = { has_trait = incapable }
                }
            }
            trigger_else = { always = no }
            is_puppet_action_valid = puppet_trigger_can_create_rite
        }
        add_character_flag = des_test_original_rite_relation_ready
        debug_log = "DESAF: PRECONDITION PASS original_rite_relationship"
    }
    else = { debug_log = "DESAF: NOT_READY original_rite_relationship" }
}

# Creation and editing have distinct stock eligibility. Creation becomes used
# legitimately; no fixture resets either stock once-only variable.
des_acceptance_read_original_rite_input_effect = {
    remove_character_flag = des_test_original_rite_input_ready
    des_acceptance_read_original_rite_relation_effect = yes
    if = {
        limit = {
            has_character_flag = des_test_original_rite_relation_ready
            can_create_rite_trigger = yes
        }
        add_character_flag = des_test_original_rite_input_ready
        debug_log = "DESAF: PRECONDITION PASS original_rite_scripted_eligibility"
    }
    else = { debug_log = "DESAF: NOT_READY original_rite_scripted_eligibility" }
}

des_acceptance_read_original_edit_input_effect = {
    remove_character_flag = des_test_original_edit_input_ready
    des_acceptance_read_original_rite_relation_effect = yes
    if = {
        limit = {
            has_character_flag = des_test_original_rite_relation_ready
            exists = rite.head_of_rite
            rite.head_of_rite = this
            NOT = { has_variable = has_edited_a_rite }
            can_edit_rite_trigger = yes
        }
        add_character_flag = des_test_original_edit_input_ready
        debug_log = "DESAF: PRECONDITION PASS original_edit_scripted_eligibility"
    }
    else = { debug_log = "DESAF: NOT_READY original_edit_scripted_eligibility" }
}

des_acceptance_restore_refs_effect = {
    var:des_test_ref_target = { save_scope_as = des_test_target }
    var:des_test_ref_peer = { save_scope_as = des_test_peer }
    var:des_test_ref_child = { save_scope_as = des_test_child }
    var:des_test_ref_outsider = { save_scope_as = des_test_outsider }
    var:des_test_source_rite = { save_scope_as = des_test_source_rite }
    if = {
        limit = { has_variable = des_test_primary_rite }
        var:des_test_primary_rite = { save_scope_as = des_test_primary }
    }
    if = {
        limit = { has_variable = des_test_bound_dynasty }
        var:des_test_bound_dynasty = { save_scope_as = des_test_dynasty }
    }
}
