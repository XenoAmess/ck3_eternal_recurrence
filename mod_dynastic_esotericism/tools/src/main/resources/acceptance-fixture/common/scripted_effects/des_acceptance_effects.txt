    var:des_test_ref_outsider = { save_scope_as = des_test_outsider }
    var:des_test_source_rite = { save_scope_as = des_test_source_rite }

        limit = { has_variable = des_test_primary_rite }
        var:des_test_primary_rite = { save_scope_as = des_test_primary }
    }

        limit = { has_variable = des_test_bound_dynasty }
        var:des_test_bound_dynasty = { save_scope_as = des_test_dynasty }
    }
}
