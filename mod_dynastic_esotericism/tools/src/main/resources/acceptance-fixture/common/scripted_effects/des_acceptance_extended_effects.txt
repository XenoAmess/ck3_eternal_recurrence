        if = {
            limit = {
                has_variable = des_test_cell_root_actor
                has_variable = des_test_cell_root_title
                has_variable = des_test_cell_root_house
                has_variable = des_test_cell_root_dynasty
                this = var:des_test_cell_root_actor
                primary_title = var:des_test_cell_root_title
                primary_title = { exists = holder holder = scope:des_test_head_cells_root.var:des_test_cell_root_actor }
                house = var:des_test_cell_root_house
                dynasty = var:des_test_cell_root_dynasty
                trigger_if = {
                    limit = { has_variable = des_test_cell_root_house_head }
                    exists = house.house_head
                    house.house_head = var:des_test_cell_root_house_head
                }
                trigger_else = { NOT = { exists = house.house_head } }
                trigger_if = {
                    limit = { has_variable = des_test_cell_root_dynast }
                    exists = dynasty.dynast
                    dynasty.dynast = var:des_test_cell_root_dynast
                }
                trigger_else = { NOT = { exists = dynasty.dynast } }
            }
            debug_log = "DESAF: INPUT_READY root_identity_preserved"
        }
        else = { debug_log = "DESAF: INPUT_NOT_READY root_identity_preserved" }
    }
    else = { debug_log = "DESAF: INPUT_NOT_READY stock_head_cells_root_precondition" }
