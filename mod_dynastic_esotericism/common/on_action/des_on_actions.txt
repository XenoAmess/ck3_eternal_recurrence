# Neutral creation contract. Consumers may broadcast this before creating a Rite
# even when a paused old save predates the persisted defaults registry.
xar_rite_creation_collect_optional_default_doctrines = {
	on_actions = { des_register_optional_rite_creation_defaults }
}

des_register_optional_rite_creation_defaults = {
	effect = { des_register_optional_rite_creation_defaults_effect = yes }
}

des_new_game_initialize = {
	effect = { des_initialize_world_effect = yes }
}

des_new_game_reconcile = {
	effect = {
		des_initialize_world_effect = yes
		des_maintenance_effect = yes
	}
}

des_on_rite_created = {
	effect = {
		if = {
			limit = { has_global_variable = des_initialized is_alive = yes }
			# Resolve the native source before any product member synchronization.
			des_claim_native_created_rite_head_effect = yes
			des_sync_character_effect = yes
			rite = { des_sync_rite_effect = yes }
		}
	}
}

# Immediate native UI edit: record the actor and unbound starting state only.
# Gameplay synchronization and the one-shot transfer remain in on_rite_updated.
des_on_native_rite_edited = {
	effect = {
		if = {
			limit = { has_global_variable = des_initialized }
			des_record_native_rite_edit_head_claim_effect = yes
		}
	}
}

des_on_rite_updated = {
	effect = {
		if = {
			limit = { has_global_variable = des_initialized }
			# Only a native edit token may consume the pre-sync source snapshot.
			save_scope_as = des_updated_rite
			if = {
				limit = {
					exists = scope:consume_rite_edit_token
					has_variable = des_native_head_claim_pending
					rite_has_doctrine = des_doctrine_dynastic
				}
				des_finalize_native_rite_head_claim_effect = yes
			}
			head_of_rite ?= {
				if = {
					limit = { is_alive = yes rite = scope:des_updated_rite faith = scope:des_updated_rite.faith }
					des_ensure_own_rite_binding_effect = yes
				}
			}
			des_sync_rite_effect = yes
			if = {
				limit = { exists = scope:consume_rite_edit_token }
				if = {
					limit = {
						has_variable = des_native_head_claim_pending
						NOT = { rite_has_doctrine = des_doctrine_dynastic }
					}
					# A real edit that did not activate this product grants no claim.
					remove_variable = des_native_head_claim_pending
					remove_variable = des_native_head_claim_actor
					remove_variable = des_native_head_claim_kind
					remove_variable = des_native_head_claim_date
				}
			}
		}
	}
}

des_on_rite_change = {
	effect = {
		if = {
			limit = { has_global_variable = des_initialized is_alive = yes }
			des_sync_character_effect = yes
		}
	}
}

# Rite scope. Unknown scripted Rites receive ordinary adoption eligibility on
# day +1; an earlier native callback already resolved its own actor and policy.
des_deferred_rite_binding = {
	effect = {
		if = {
			limit = { has_global_variable = des_initialized has_variable = des_deferred_binding_queued }
			if = {
				limit = { has_variable = des_native_head_claim_pending }
				# A genuine immediate edit may precede its delayed native update.
				trigger_event = { on_action = des_deferred_rite_binding days = 1 }
			}
			else = {
				remove_variable = des_deferred_binding_queued
				if = {
					limit = { rite_has_doctrine = des_doctrine_dynastic }
					des_allow_ordinary_rite_binding_effect = yes
				}
				save_scope_as = des_deferred_rite
				if = {
					limit = {
						NOR = { has_variable = des_bound_dynasty has_variable = des_join_house }
						exists = var:des_deferred_binding_requester
					}
					var:des_deferred_binding_requester = {
						if = {
							limit = { is_alive = yes rite = scope:des_deferred_rite faith = scope:des_deferred_rite.faith }
							des_ensure_own_rite_binding_effect = yes
						}
					}
				}
				des_sync_rite_effect = yes
			}
		}
	}
}

des_on_character_finalization = {
	effect = {
		if = {
			limit = { has_global_variable = des_initialized is_alive = yes }
			des_teach_doctrines_effect = yes
			# Vanilla birth/creation can assign a House later in its own chain.
			trigger_event = { on_action = des_character_finalized days = 1 }
		}
	}
}

des_character_finalized = {
	effect = {
		if = {
			limit = { has_global_variable = des_initialized is_alive = yes }
			des_sync_character_effect = yes
		}
	}
}

# Root is the persistent Religion selected by des_ensure_maintenance_effect.
des_monthly_maintenance = {
	effect = {
		if = {
			limit = { has_global_variable = des_initialized }
			des_maintenance_effect = yes
			set_global_variable = { name = des_maintenance_queued value = yes days = 31 }
			trigger_event = { on_action = des_monthly_maintenance days = 30 }
		}
	}
}

		}
	}
}

des_character_watchdog = {
	effect = {
		if = {

		}
	}
}

des_on_death = {
	effect = {
		if = {
			limit = { has_global_variable = des_initialized }
			# Clear outgoing pairs before death without rewriting historical dynasty.
			save_scope_as = des_dying_observer
			des_snapshot_relationships_effect = { SOURCE = des_penalty_targets }
			every_in_list = {
				variable = des_relationship_snapshot
				save_scope_as = des_death_target
				scope:des_dying_observer = {
					des_remove_pair_effect = { TARGET = scope:des_death_target }
				}
			}
			clear_variable_list = des_relationship_snapshot
		}
	}
}
