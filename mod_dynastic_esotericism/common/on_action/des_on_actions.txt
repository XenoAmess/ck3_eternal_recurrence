# Neutral creation contract. Consumers may broadcast this before creating a Rite
# even when a paused old save predates the persisted defaults registry.
xar_rite_creation_collect_optional_default_doctrines = {
	on_actions = { des_register_optional_rite_creation_defaults }
}

    on_actions = { des_apply_registered_rite_creation_defaults_v2 }
}
des_apply_registered_rite_creation_defaults_v2 = {
    effect = { des_apply_registered_rite_creation_defaults_v2_effect = yes }
}

des_register_optional_rite_creation_defaults = {
	effect = { des_register_optional_rite_creation_defaults_effect = yes }
}

des_new_game_initialize = {
	effect = {
		# Only a new game grants the one-shot post-lobby preset correction.
		set_global_variable = { name = des_new_game_preset_clergy_pending value = yes }
		des_initialize_world_effect = yes
		# A fresh world is not a legacy leadership migration. Normal binding sync applies.
		set_global_variable = { name = des_leadership_024_checked value = yes }
	}
}

des_new_game_reconcile = {
	effect = {
		des_initialize_world_effect = yes
		des_reconcile_new_game_preset_clergy_effect = yes
		des_maintenance_effect = yes
	}
}

des_on_rite_created = {
	effect = {
		if = {
			limit = { has_global_variable = des_initialized is_alive = yes }
			# Resolve the native source before any product member synchronization.
			des_claim_native_created_rite_head_effect = yes
			des_sync_character_effect = yes
			rite = { des_sync_rite_effect = yes }
		}
	}
}

# Immediate native UI edit: record the actor and unbound starting state only.
# Gameplay synchronization and the one-shot transfer remain in on_rite_updated.
des_on_native_rite_edited = {
	effect = {
		if = {
			limit = { has_global_variable = des_initialized }
			des_record_native_rite_edit_head_claim_effect = yes
		}
	}
}

des_on_rite_updated = {
	effect = {
		if = {
			limit = { has_global_variable = des_initialized }
			# Only a native edit token may consume the pre-sync source snapshot.
			save_scope_as = des_updated_rite
			if = {
				limit = {
					exists = scope:consume_rite_edit_token
					has_variable = des_native_head_claim_pending
					rite_has_doctrine = des_doctrine_dynastic
				}
				des_finalize_native_rite_head_claim_effect = yes
			}
			head_of_rite ?= {
				if = {
					limit = { is_alive = yes rite = scope:des_updated_rite faith = scope:des_updated_rite.faith }
					des_ensure_own_rite_binding_effect = yes
				}
			}
			des_sync_rite_effect = yes
			if = {
				limit = { exists = scope:consume_rite_edit_token }
				if = {
					limit = {
						has_variable = des_native_head_claim_pending
						NOT = { rite_has_doctrine = des_doctrine_dynastic }
					}
					# A real edit that did not activate this product grants no claim.
					remove_variable = des_native_head_claim_pending
					remove_variable = des_native_head_claim_actor
					remove_variable = des_native_head_claim_kind
					remove_variable = des_native_head_claim_date
				}
			}
		}
	}
}

des_on_rite_change = {
	effect = {
		if = {
			limit = { has_global_variable = des_initialized is_alive = yes }
            # Stock old_rite is the real departed Rite. Capture the actor before
            # normal new-Rite synchronization can migrate its Dynasty/House.
            save_scope_as = des_leadership_changed_head
            scope:old_rite ?= { des_queue_matching_religious_head_recheck_effect = yes }
			des_sync_character_effect = yes
		}
	}
}

# Rite scope. Unknown scripted Rites receive ordinary adoption eligibility on
# day +1; an earlier native callback already resolved its own actor and policy.
des_deferred_rite_binding = {
	effect = {
		if = {
			limit = { has_global_variable = des_initialized has_variable = des_deferred_binding_queued }
			if = {
				limit = { has_variable = des_native_head_claim_pending }
				# A genuine immediate edit may precede its delayed native update.
				trigger_event = { on_action = des_deferred_rite_binding days = 1 }
			}
			else = {
				remove_variable = des_deferred_binding_queued
				if = {
					limit = { rite_has_doctrine = des_doctrine_dynastic }
					des_allow_ordinary_rite_binding_effect = yes
				}
				save_scope_as = des_deferred_rite
				if = {
					limit = {
						NOR = { has_variable = des_bound_dynasty has_variable = des_join_house }
						exists = var:des_deferred_binding_requester
					}
					var:des_deferred_binding_requester = {
						if = {
							limit = { is_alive = yes rite = scope:des_deferred_rite faith = scope:des_deferred_rite.faith }
							des_ensure_own_rite_binding_effect = yes
						}
					}
				}
				des_sync_rite_effect = yes
			}
		}
	}
}

des_on_character_finalization = {
	effect = {
		if = {
			limit = { has_global_variable = des_initialized is_alive = yes }
			des_teach_doctrines_effect = yes
			# Vanilla birth/creation can assign a House later in its own chain.
			trigger_event = { on_action = des_character_finalized days = 1 }
		}
	}
}

des_character_finalized = {
	effect = {
		if = {
			limit = { has_global_variable = des_initialized is_alive = yes }
			des_sync_character_effect = yes
		}
	}
}

des_on_death = {
	effect = {
		if = {
			limit = { has_global_variable = des_initialized }
			# Clear outgoing pairs before death without rewriting historical dynasty.
			save_scope_as = des_dying_observer
			des_snapshot_relationships_effect = { SOURCE = des_penalty_targets }
			every_in_list = {
				variable = des_relationship_snapshot
				save_scope_as = des_death_target
				scope:des_dying_observer = {
					des_remove_pair_effect = { TARGET = scope:des_death_target }
				}
			}
			clear_variable_list = des_relationship_snapshot
			# Do not retain a dead target and all incoming pairs until next year.
			des_clear_apostasy_effect = yes
			# on_death is before death: defer only a real current/recorded leader.
			des_queue_religious_head_death_recheck_effect = yes
		}
	}
}


# Character root is the actual new title holder (stock on_title_gain contract).
# A title is never proof of a new Rite head. Only local actual-Rite re-reads run.
des_on_religious_head_title_gain = {
    effect = {
        if = {
            limit = { has_global_variable = des_initialized is_alive = yes exists = rite }
            des_teach_doctrines_effect = yes
            rite = {
                des_reconcile_leadership_binding_metadata_effect = yes
                if = {
                    limit = {
                        des_created_dynasty_leadership_target_trigger = yes
                        NOT = { has_variable = des_leadership_sync_in_progress }
                    }
                    # Read this actor's actual bound Rite, never its Faith's
                    # main religious title. Source may settle after title_gain.
                    des_sync_created_dynasty_leadership_effect = yes
                    des_queue_house_leadership_recheck_effect = yes
                }
            }
        }
    }
}

# Annual fallback and local lifecycle repair share real postconditions.
# The declared yearly sibling list alone is not an engine completion-order proof.
des_annual_house_leadership_recovery = {
    effect = {
        if = {
            limit = { has_global_variable = des_initialized }
            every_in_global_list = {
                variable = des_bound_rites
                des_sync_created_dynasty_leadership_effect = yes
            }
        }
    }
}


# Stock root is the new head; saved scope:house is the affected House. Never
# route through the new head's Rite, since the new head can be cross-Rite.
des_on_bound_house_head_changed = {
    effect = {
        if = {
            limit = { has_global_variable = des_initialized }
            scope:house ?= {
                house_dynasty = {
                    save_scope_as = des_leadership_changed_dynasty
                    if = {
                        limit = { exists = var:des_bound_rite }
                        var:des_bound_rite = {
                            if = {
                                limit = {
                                    exists = var:des_bound_dynasty
                                    var:des_bound_dynasty = scope:des_leadership_changed_dynasty
                                }
                                des_sync_created_dynasty_leadership_effect = yes
                            }
                        }
                    }
                }
            }
        }
    }
}

# Stock root is the new Dynast; saved scope:dynasty is the affected Dynasty.
des_on_bound_dynast_changed = {
    effect = {
        if = {
            limit = { has_global_variable = des_initialized }
            scope:dynasty ?= {
                save_scope_as = des_leadership_changed_dynasty
                if = {
                    limit = { exists = var:des_bound_rite }
                    var:des_bound_rite = {
                        if = {
                            limit = {
                                exists = var:des_bound_dynasty
                                var:des_bound_dynasty = scope:des_leadership_changed_dynasty
                            }
                            des_sync_created_dynasty_leadership_effect = yes
                        }
                    }
                }
            }
        }
    }
}

# Keep the queued marker throughout the callback: an unresolved repair cannot
# renew itself. A later genuine lifecycle event or annual fallback may try again.
des_local_house_leadership_recheck = {
    effect = {
        if = {
            limit = { has_global_variable = des_initialized has_variable = des_leadership_recheck_queued }
            if = {
                limit = { has_variable = des_leadership_refresh_requested }
                # Consume before the official election can trigger callbacks.
                # Keep queued until the end so this callback cannot renew itself.
                remove_variable = des_leadership_refresh_requested
                if = {
                    limit = {
                        des_created_dynasty_leadership_target_trigger = yes
                        NOT = { des_has_own_living_religious_head_trigger = yes }
                    }
                    # Current stock: force-recalculate the best Head of this Rite.
                    # Existing valid successors are never reselected here.
                    force_refresh_head_of_rite = yes
                }
            }
            # Re-read the actual getter after refresh. A still-missing head or
            # either wrong House/Dynast field remains pending, never inferred P.
            des_sync_created_dynasty_leadership_effect = yes
            remove_variable = des_leadership_recheck_queued
        }
    }
}
