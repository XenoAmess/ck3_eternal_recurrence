# Keep stock qualified automatic/decision creation, but never replace these two empire titles with a duchy.
# All other Faiths keep the original dynamic duchy and CoA; NEW_HOLDER, transfer and laws remain stock.
set_up_dynamic_temporal_hof_title_effect = {
	faith = { save_scope_as = my_faith }
	if = {
		limit = { faith = faith:des_procreation_faith }
		title:e_des_procreation_papacy = { save_scope_as = new_title }
	}
	else_if = {
		limit = { faith = faith:des_vivhite_faith }
		title:e_des_vivhite_papacy = { save_scope_as = new_title }
	}
	else = {
		create_dynamic_title = {
			tier = duchy
			name = REL_HEAD_TITLE_NAME
			adj = REL_HEAD_TITLE_ADJ
		}
	}
	create_title_and_vassal_change = {
		type = created
		save_scope_as = change
		add_claim_on_loss = no
	}
	scope:new_title = {
		set_destroy_if_invalid_heir = yes
		set_no_automatic_claims = yes
		set_definitive_form = yes
		set_always_follows_primary_heir = yes
		change_title_holder = {
			holder = $NEW_HOLDER$
			change = scope:change
		}
	}
	resolve_title_and_vassal_change = scope:change
	faith = { set_religious_head_title = scope:new_title }
	if = {
		limit = {
			NOT = { has_realm_law = same_faith_succession_law }
		}
		add_realm_law_skip_effects = same_faith_succession_law
	}
	scope:new_title = {
		if = {
			limit = { scope:my_faith = faith:des_procreation_faith }
			set_coa = des_procreation_rite_coa
		}
		else_if = {
			limit = { scope:my_faith = faith:des_vivhite_faith }
			set_coa = des_vivhite_rite_coa
		}
		else = { generate_coa = religious_title }
		add_title_law = temporal_head_of_faith_succession_law
	}
}
