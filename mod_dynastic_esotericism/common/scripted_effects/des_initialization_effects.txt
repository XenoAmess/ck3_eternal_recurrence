# Any scope. New games run this automatically; an old save uses the native decision.
des_initialize_world_effect = {
	des_register_optional_rite_creation_defaults_effect = yes
	if = {
		limit = { NOT = { has_global_variable = des_initialized } }
		set_global_variable = { name = des_initialized value = yes }
		every_religion_global = {
			every_faith = {
				every_faith_rite = {
					des_initialize_rite_effect = yes
					# A None Rite may later be edited natively. Its first activation
					# must retain the barrier even if an on_change precedes edited.
					if = {
						limit = { rite_has_doctrine = des_doctrine_dynastic }
						des_allow_ordinary_rite_binding_effect = yes
					}
				}
				every_faith_character = {
					limit = { is_alive = yes }
					des_teach_doctrines_effect = yes
				}
			}
		}
		des_discover_rites_effect = yes
	}
	des_ensure_maintenance_effect = yes
}

# Rite scope. Exact object membership, not an inherited boolean, admits the
# ordinary creation path. This grants no native actor/reuse provenance.
des_allow_ordinary_rite_binding_effect = {
	set_variable = { name = des_ordinary_binding_ready value = yes }
	if = {
		limit = { NOT = { is_target_in_global_variable_list = { name = des_ordinary_binding_rites target = this } } }
		add_to_global_variable_list = { name = des_ordinary_binding_rites target = this }
	}
}

# Shared, optional creation contract. A consumer applies these Doctrine scopes
# only to its newly created Rite, before adoption or a native configuration step.
# Register our neutral choice without requiring any consumer mod to be installed.
# Repeated startup/load reconciliation must preserve other mods' entries.
des_register_optional_rite_creation_defaults_effect = {
	if = {
		limit = {
			NOT = {
				is_target_in_global_variable_list = {
					name = xar_rite_creation_optional_default_doctrines
					target = doctrine:des_doctrine_none
				}
			}
		}
		add_to_global_variable_list = {
			name = xar_rite_creation_optional_default_doctrines
			target = doctrine:des_doctrine_none
		}
	}
}

# Rite scope. Do not overwrite a submitted choice or enable existing rites.
des_initialize_rite_effect = {
	if = {
		limit = {
			NOR = {
				rite_has_doctrine = des_doctrine_none
				rite_has_doctrine = des_doctrine_dynastic
			}
		}
		add_rite_doctrine = des_doctrine_none
	}
}

# Character scope. The native knowledge effect requires a playable character;
# this eligibility check does not limit dynasty membership or apostasy rules.
des_teach_doctrines_effect = {
	if = {
		limit = { is_alive = yes is_playable_character = yes }
		if = {
			limit = { NOT = { knows_doctrine = doctrine:des_doctrine_none } }
			add_known_doctrine = des_doctrine_none
		}
		if = {
			limit = { NOT = { knows_doctrine = doctrine:des_doctrine_dynastic } }
			add_known_doctrine = des_doctrine_dynastic
		}
	}
}

# Keep the delayed queue on a persistent Religion, rather than a mortal founder.
# The lease also permits the annual watchdog to recover an interrupted queue.
des_ensure_maintenance_effect = {
	if = {
		limit = {
			has_global_variable = des_initialized
			NOT = { has_global_variable = des_maintenance_queued }
		}
		every_religion_global = {
			# Check at execution time, independently of iterator candidate snapshots.
			if = {
				limit = { NOT = { has_global_variable = des_maintenance_queued } }
				set_global_variable = { name = des_maintenance_queued value = yes days = 31 }
				trigger_event = { on_action = des_monthly_maintenance days = 30 }
			}
		}
	}
}
