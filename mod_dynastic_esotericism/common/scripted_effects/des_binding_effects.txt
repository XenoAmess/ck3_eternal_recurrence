# Character scope. A genuine adherent is the only founder of a new binding.
des_ensure_own_rite_binding_effect = {
	if = {
		limit = { exists = rite }
		rite = { des_initialize_rite_effect = yes }
		if = {
			limit = { rite = { rite_has_doctrine = des_doctrine_dynastic } }
			rite = { des_repair_rite_binding_effect = yes }
			if = {
				limit = {
					rite = {
						NOR = {
							has_variable = des_bound_dynasty
							has_variable = des_join_house
							has_variable = des_binding_in_progress
							has_variable = des_native_head_claim_pending
							has_variable = des_native_initial_binding_fault
						}
					}
				}
				if = {
					limit = { rite = { is_target_in_global_variable_list = { name = des_ordinary_binding_rites target = this } } }
					des_create_rite_dynasty_effect = yes
				}
				else = { des_queue_unknown_rite_binding_effect = yes }
			}
		}
	}
}

des_create_rite_dynasty_effect = {
	rite = {
		save_scope_as = des_binding_rite
		set_variable = { name = des_binding_in_progress value = yes }
	}
	save_scope_as = des_requested_dynasty_founder
	save_scope_as = des_selected_dynasty_founder
	# First let the requested founder hand the old House to a real living successor.
	scope:des_requested_dynasty_founder = { des_relinquish_old_house_head_effect = yes }
	# If the requested founder remains ineligible, prefer another real adherent.
	if = {
		limit = { OR = { is_house_head = yes is_dynast = yes } }
		scope:des_binding_rite = {
			faith = {
				random_faith_character = {
					limit = {
						is_alive = yes
						rite = scope:des_binding_rite
						is_house_head = no
						is_dynast = no
					}
					save_scope_as = des_selected_dynasty_founder
				}
			}
		}
	}
	# User-authorized final fallback is a real adult adherent, not a disposable identity.
	if = {
		limit = {
			scope:des_selected_dynasty_founder = { OR = { is_house_head = yes is_dynast = yes } }
			exists = scope:des_requested_dynasty_founder.location
			exists = scope:des_requested_dynasty_founder.culture
			exists = scope:des_binding_rite.faith
		}
		# Reset the result scope before production so a failed create cannot reuse stale identity.
		scope:des_selected_dynasty_founder = { save_scope_as = des_generated_dynasty_founder }
		create_character = {
			gender_female_chance = 50
			age = 30
			dynasty = none
			location = scope:des_requested_dynasty_founder.location
			culture = scope:des_requested_dynasty_founder.culture
			faith = scope:des_binding_rite.faith
			rite = scope:des_binding_rite
			save_scope_as = des_generated_dynasty_founder
		}
		scope:des_generated_dynasty_founder = {
			if = {
				limit = {
					is_alive = yes
					is_adult = yes
					is_lowborn = yes
					rite = scope:des_binding_rite
					faith = scope:des_binding_rite.faith
				}
				save_scope_as = des_selected_dynasty_founder
			}
		}
	}
	scope:des_selected_dynasty_founder = {
		if = {
			limit = {
				is_alive = yes
				rite = scope:des_binding_rite
				faith = scope:des_binding_rite.faith
				is_house_head = no
				is_dynast = no
				trigger_if = {
					limit = { exists = house.house_head }
					NOT = { this = house.house_head }
				}
				trigger_if = {
					limit = { exists = dynasty.dynast }
					NOT = { this = dynasty.dynast }
				}
			}
			create_dynasty = {
				spread_to_descendants = no
				save_scope_as = des_created_dynasty
			}
			if = {
				limit = {
					exists = scope:des_created_dynasty
					dynasty = scope:des_created_dynasty
					exists = house
				}
				house = { save_scope_as = des_created_house }
				scope:des_binding_rite = {
					set_variable = { name = des_bound_dynasty value = scope:des_created_dynasty }
					set_variable = { name = des_join_house value = scope:des_created_house }
					set_variable = { name = des_binding_mode value = flag:new_dynasty }
				}
				scope:des_created_dynasty = {
					set_variable = { name = des_bound_rite value = scope:des_binding_rite }
				}
				# Identity uses the exact new binding; it never assigns a character head.
				scope:des_binding_rite = { des_apply_rite_identity_effect = yes }
				scope:des_requested_dynasty_founder = { des_finalize_new_rite_head_effect = yes }
				add_to_global_variable_list = { name = des_bound_rites target = scope:des_binding_rite }
			}
			else = {
				scope:des_binding_rite = { set_variable = { name = des_binding_fault value = yes } }
				debug_log = "DES: failed to create a valid dynasty/house binding; no fallback identity assigned"
			}
		}
		else = {
			scope:des_binding_rite = { set_variable = { name = des_binding_fault value = yes } }
			debug_log = "DES: no valid native dynasty founder after eligible-adherent, House-head, and generated-adherent paths"
		}
	}
	scope:des_binding_rite = { remove_variable = des_binding_in_progress }
}

# Character scope. Successful native creation or an authorized first-binding
# native completion claim calls this effect. Never call it from repair, member
# synchronization, re-enabling, or maintenance.
des_finalize_new_rite_head_effect = {
	if = {
		limit = {
			is_alive = yes
			rite = scope:des_binding_rite
			faith = scope:des_binding_rite.faith
			scope:des_binding_rite = {
				des_has_binding_trigger = yes
				var:des_bound_dynasty = scope:des_created_dynasty
				var:des_join_house = scope:des_created_house
			}
			scope:des_created_house = { house_dynasty = scope:des_created_dynasty }
			scope:des_created_dynasty = {
				dynast = scope:des_selected_dynasty_founder
				# Fresh creation supplies one House; refuse any occupied other House.
				NOT = { any_dynasty_member = { NOT = { house = scope:des_created_house } } }
			}
		}
		# The real intermediate founder stays alive; only the requesting actor moves.
		if = {
			limit = { NOT = { house = scope:des_created_house } }
			set_house = scope:des_created_house
		}
		if = {
			limit = { house = scope:des_created_house dynasty = scope:des_created_dynasty }
			scope:des_created_house = {
				if = {
					limit = { NOT = { house_head = scope:des_requested_dynasty_founder } }
					set_house_head = scope:des_requested_dynasty_founder
				}
			}
		}
	}
	# Preserve an explicit diagnosis instead of treating membership as head success.
	if = {
		limit = {
			is_alive = yes
			rite = scope:des_binding_rite
			house = scope:des_created_house
			dynasty = scope:des_created_dynasty
			scope:des_created_house = { house_head = scope:des_requested_dynasty_founder }
			scope:des_created_dynasty = { dynast = scope:des_requested_dynasty_founder }
		}
		scope:des_binding_rite = { remove_variable = des_creator_head_fault }
	}
	else = {
		scope:des_binding_rite = { set_variable = { name = des_creator_head_fault value = yes } }
		debug_log = "DES: new binding exists but requested actor is not its actual House head and Dynast"
	}
}

# Character scope. Only a real other living member of this same old House may take over.
des_relinquish_old_house_head_effect = {
	if = {
		limit = { is_house_head = yes exists = house.house_head }
		save_scope_as = des_relinquishing_founder
		house = {
			random_house_member = {
				limit = {
					is_alive = yes
					house = scope:des_relinquishing_founder.house
					NOT = { this = scope:des_relinquishing_founder }
				}
				save_scope_as = des_old_house_successor
				house = { set_house_head = scope:des_old_house_successor }
			}
		}
		if = {
			limit = { is_house_head = yes }
			debug_log = "DES: no living other old House member available for head handover"
		}
	}
}

# Rite scope. Repair only missing components whose remaining object identity proves
# the relationship. A conflicting backlink is diagnosed, never overwritten.
des_repair_rite_binding_effect = {
	save_scope_as = des_repair_rite
	if = {
		limit = {
			NOT = { has_variable = des_bound_dynasty }
			exists = var:des_join_house.house_dynasty
			var:des_join_house.house_dynasty.var:des_bound_rite = this
		}
		set_variable = { name = des_bound_dynasty value = var:des_join_house.house_dynasty }
	}
	if = {
		limit = { exists = var:des_bound_dynasty }
		var:des_bound_dynasty = {
			if = {
				limit = { NOT = { has_variable = des_bound_rite } }
				set_variable = { name = des_bound_rite value = scope:des_repair_rite }
				debug_log = "DES: restored missing dynasty backlink from the persisted rite binding"
			}
		}
		if = {
			limit = {
				NOT = { exists = var:des_join_house }
				var:des_bound_dynasty.var:des_bound_rite = this
			}
			var:des_bound_dynasty = {
				every_dynasty_house = {
					limit = { NOT = { exists = scope:des_repair_rite.var:des_join_house } }
					save_scope_as = des_recovered_house
					scope:des_repair_rite = {
						set_variable = { name = des_join_house value = scope:des_recovered_house }
					}
				}
			}
		}
	}
	if = {
		limit = { des_has_binding_trigger = yes }
		remove_variable = des_binding_fault
		if = {
			limit = {
				NOT = { is_target_in_global_variable_list = { name = des_bound_rites target = this } }
			}
			add_to_global_variable_list = { name = des_bound_rites target = this }
		}
	}
	else_if = {
		limit = {
			OR = { has_variable = des_bound_dynasty has_variable = des_join_house }
			NOT = { has_variable = des_binding_fault }
		}
		set_variable = { name = des_binding_fault value = yes }
		debug_log = "DES: conflicting or unavailable persisted binding; migration paused for this rite"
	}
}

# Character scope. Unknown Rites cannot let a generic conversion wave migrate
# the real native actor before on_rite_created supplies its trustworthy Root.
des_queue_unknown_rite_binding_effect = {
	save_scope_as = des_first_ordinary_requester
	rite = {
		if = {
			limit = {
				NOT = { is_target_in_global_variable_list = { name = des_ordinary_binding_rites target = this } }
				NOR = {
					has_variable = des_deferred_binding_queued
					has_variable = des_bound_dynasty
					has_variable = des_join_house
				}
			}
			set_variable = { name = des_deferred_binding_requester value = scope:des_first_ordinary_requester }
			set_variable = { name = des_deferred_binding_date value = current_date }
			set_variable = { name = des_deferred_binding_queued value = yes }
			trigger_event = { on_action = des_deferred_rite_binding days = 1 }
		}
	}
}

# Character scope, called only by a genuine native create/edit callback.
# This runs before any product sync in that callback and while generic unknown
# Rite migration remains barred. Never infer a native actor from a follower.
des_capture_native_rite_source_effect = {
	save_scope_as = des_native_source_actor
	scope:des_native_source_rite = {
		set_variable = { name = des_native_head_claim_actor value = scope:des_native_source_actor }
		set_variable = { name = des_native_source_actor value = scope:des_native_source_actor }
		set_variable = { name = des_native_head_claim_date value = current_date }
		set_variable = { name = des_native_head_claim_pending value = yes }
		remove_variable = des_native_source_was_dynast
		remove_variable = des_native_source_dynasty
		remove_variable = des_native_source_house
	}
	if = {
		limit = { exists = dynasty exists = house house.house_dynasty = dynasty }
		save_scope_as = des_native_original_actor
		dynasty = { save_scope_as = des_native_original_dynasty }
		house = { save_scope_as = des_native_original_house }
		scope:des_native_source_rite = {
			set_variable = { name = des_native_source_dynasty value = scope:des_native_original_dynasty }
			set_variable = { name = des_native_source_house value = scope:des_native_original_house }
		}
		if = {
			limit = { is_dynast = yes dynasty.dynast = this }
			scope:des_native_source_rite = { set_variable = { name = des_native_source_was_dynast value = yes } }
		}
	}
}

# Character scope. The genuine on_rite_created Root is captured BEFORE sync.
des_claim_native_created_rite_head_effect = {
	if = {
		limit = {
			is_alive = yes
			exists = rite
			exists = scope:old_rite
			NOT = { rite = scope:old_rite }
		}
		save_scope_as = des_native_creation_actor
		rite = {
			if = {
				limit = {
					rite_has_doctrine = des_doctrine_dynastic
					NOR = {
						has_variable = des_bound_dynasty
						has_variable = des_join_house
						has_variable = des_binding_in_progress
						has_variable = des_native_head_claim_pending
						has_variable = des_native_head_claim_attempted
					}
				}
				save_scope_as = des_native_source_rite
				set_variable = { name = des_native_head_claim_kind value = flag:created }
				scope:des_native_creation_actor = { des_capture_native_rite_source_effect = yes }
				des_finalize_native_rite_head_claim_effect = yes
			}
		}
	}
}

# Rite scope. Record the native editor while this Rite is still unbound, without
# assuming whether the edited doctrines have already been applied. Only updated
# decides from the committed doctrine whether this is a first dynastic activation.
# An already bound historical Rite never acquires this pending first-binding claim.
des_record_native_rite_edit_head_claim_effect = {
	if = {
		limit = {
			exists = head_of_rite
			NOR = {
				has_variable = des_bound_dynasty
				has_variable = des_join_house
				has_variable = des_binding_in_progress
				has_variable = des_native_head_claim_pending
				has_variable = des_native_head_claim_attempted
			}
		}
		save_scope_as = des_native_edit_rite
		head_of_rite = {
			if = {
				limit = { is_alive = yes rite = scope:des_native_edit_rite faith = scope:des_native_edit_rite.faith }
				save_scope_as = des_native_edit_actor
				scope:des_native_edit_rite = {
					save_scope_as = des_native_source_rite
					set_variable = { name = des_native_head_claim_kind value = flag:edited }
					scope:des_native_edit_actor = { des_capture_native_rite_source_effect = yes }
				}
			}
		}
	}
}

# Rite scope. Only the native create/update completion routes consume this claim.
# Consume before native mutation; maintenance, ordinary conversion and succession
# never call this helper, and a failed attempt is diagnosed without replay.
des_finalize_native_rite_head_claim_effect = {
	if = {
		limit = {
			has_variable = des_native_head_claim_pending
			NOT = { has_variable = des_native_head_claim_attempted }
			rite_has_doctrine = des_doctrine_dynastic
		}
		set_variable = { name = des_native_head_claim_attempted value = yes }
		remove_variable = des_native_head_claim_pending
		save_scope_as = des_native_head_claim_rite
		if = {
			limit = {
				exists = var:des_native_head_claim_actor
				NOR = {
					has_variable = des_bound_dynasty
					has_variable = des_join_house
					has_variable = des_binding_in_progress
				}
				var:des_native_head_claim_actor = {
					is_alive = yes
					rite = scope:des_native_head_claim_rite
					faith = scope:des_native_head_claim_rite.faith
				}
			}
			if = {
				limit = { has_variable = des_native_source_was_dynast }
				# A failed original-identity check must never fall through to a new
				# Dynasty/NPC after losing the pre-operation source.
				des_reuse_native_dynast_binding_effect = yes
			}
			else = {
				var:des_native_head_claim_actor = { des_create_rite_dynasty_effect = yes }
			}
			des_allow_ordinary_rite_binding_effect = yes
			if = {
				limit = {
					des_has_active_binding_trigger = yes
					var:des_join_house.house_head = var:des_native_head_claim_actor
					var:des_bound_dynasty.dynast = var:des_native_head_claim_actor
				}
				set_variable = { name = des_native_head_claim_succeeded value = yes }
				remove_variable = des_native_initial_binding_fault
				set_variable = { name = des_native_head_claim_dynasty value = var:des_bound_dynasty }
				set_variable = { name = des_native_head_claim_house value = var:des_join_house }
			}
			else = {
				set_variable = { name = des_native_initial_binding_fault value = yes }
				set_variable = { name = des_creator_head_fault value = yes }
				debug_log = "DES: native creator/editor claim did not finish with the requested actor as House head and Dynast"
			}
		}
		else = {
			set_variable = { name = des_native_initial_binding_fault value = yes }
			set_variable = { name = des_creator_head_fault value = yes }
			debug_log = "DES: native creator/editor claim consumed without a valid binding and living same-Rite actor"
		}
	}
}

# Rite scope. Only a consumed, trustworthy native operation reaches this helper.
# No head setter, member migration, House creation or Dynasty creation is legal
# in this branch. The source must still be the actor's actual original identity.
des_reuse_native_dynast_binding_effect = {
	save_scope_as = des_native_reuse_rite
	if = {
		limit = {
			has_variable = des_native_source_was_dynast
			exists = var:des_native_source_actor
			exists = var:des_native_source_dynasty
			exists = var:des_native_source_house
			var:des_native_source_actor = {
				is_alive = yes
				is_dynast = yes
				dynasty = scope:des_native_reuse_rite.var:des_native_source_dynasty
				house = scope:des_native_reuse_rite.var:des_native_source_house
			}
			var:des_native_source_dynasty.dynast = var:des_native_source_actor
			var:des_native_source_house.house_head = var:des_native_source_actor
			var:des_native_source_house.house_dynasty = var:des_native_source_dynasty
			trigger_if = {
				limit = { exists = var:des_native_source_dynasty.var:des_bound_rite }
				var:des_native_source_dynasty.var:des_bound_rite = {
					des_has_binding_trigger = yes
					var:des_bound_dynasty = scope:des_native_reuse_rite.var:des_native_source_dynasty
				}
			}
		}
		# A Dynasty belongs to one Rite. Explicit transfer disables the old Rite
		# before clearing only its product sources; stock crimes are untouched.
		if = {
			limit = {
				exists = var:des_native_source_dynasty.var:des_bound_rite
				NOT = { var:des_native_source_dynasty.var:des_bound_rite = this }
			}
			var:des_native_source_dynasty.var:des_bound_rite = {
				save_scope_as = des_transferred_old_rite
				remove_rite_doctrine = des_doctrine_dynastic
				add_rite_doctrine = des_doctrine_none
				set_variable = { name = des_binding_transferred_to value = scope:des_native_reuse_rite }
				set_variable = { name = des_binding_transfer_date value = current_date }
				des_snapshot_relationships_effect = { SOURCE = des_apostates }
				every_in_list = {
					variable = des_relationship_snapshot
					if = {
						limit = { var:des_apostate_rite = scope:des_transferred_old_rite }
						des_clear_apostasy_effect = yes
					}
				}
				clear_variable_list = des_relationship_snapshot
				clear_variable_list = des_apostates
				scope:des_native_reuse_rite.var:des_native_source_dynasty = {
					if = {
						limit = { var:des_bound_rite = scope:des_transferred_old_rite }
						remove_variable = des_bound_rite
					}
				}
				remove_variable = des_bound_dynasty
				remove_variable = des_join_house
				remove_variable = des_binding_mode
				remove_variable = des_binding_fault
				remove_variable = des_creator_head_fault
				remove_variable = des_rite_coa_unsupported
				remove_variable = des_ordinary_binding_ready
				remove_list_global_variable = { name = des_ordinary_binding_rites target = this }
				remove_variable = des_native_head_claim_pending
				remove_variable = des_native_head_claim_attempted
				remove_variable = des_native_head_claim_succeeded
				remove_variable = des_native_initial_binding_fault
				remove_variable = des_native_head_claim_actor
				remove_variable = des_native_head_claim_kind
				remove_variable = des_native_head_claim_date
				remove_variable = des_native_head_claim_dynasty
				remove_variable = des_native_head_claim_house
				remove_variable = des_native_source_actor
				remove_variable = des_native_source_was_dynast
				remove_variable = des_native_source_dynasty
				remove_variable = des_native_source_house
				remove_variable = des_deferred_binding_queued
				remove_variable = des_deferred_binding_requester
				remove_variable = des_deferred_binding_date
				remove_list_global_variable = { name = des_bound_rites target = this }
			}
		}
		set_variable = { name = des_binding_mode value = flag:native_dynast_reuse }
		set_variable = { name = des_bound_dynasty value = var:des_native_source_dynasty }
		set_variable = { name = des_join_house value = var:des_native_source_house }
		var:des_native_source_dynasty = {
			set_variable = { name = des_bound_rite value = scope:des_native_reuse_rite }
		}
		des_apply_rite_identity_effect = yes
		add_to_global_variable_list = { name = des_bound_rites target = this }
	}
}
