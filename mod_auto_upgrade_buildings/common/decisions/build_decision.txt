enable_auto_build = {
    picture = {
        reference = "gfx/interface/illustrations/decisions/decision_misc.dds"
    }
    decision_group_type = realm
    ai_check_interval = 0
    desc = enable_auto_build_desc
    is_shown = {
        is_ai = no
        NOT = {
            has_character_flag = enable_auto_build
        }
    }

    is_valid = {
        is_ai = no
    }

    widget = {
        gui = "decision_view_widget_option_list_generic"
        controller = decision_option_list_controller
        decision_to_second_step_button = enable_auto_build_choose_policy_button
        show_from_start = yes

        item = {
            value = aub_policy_treasury_only_pause_choice
            localization = aub_policy_treasury_only_pause
            current_description = aub_policy_treasury_only_pause_desc
            is_valid = { always = yes }
            ai_chance = { value = 0 }
        }
        item = {
            value = aub_policy_treasury_only_continue_choice
            localization = aub_policy_treasury_only_continue
            current_description = aub_policy_treasury_only_continue_desc
            is_valid = { always = yes }
            ai_chance = { value = 0 }
        }
        item = {
            value = aub_policy_personal_only_pause_choice
            localization = aub_policy_personal_only_pause
            current_description = aub_policy_personal_only_pause_desc
            is_valid = { always = yes }
            ai_chance = { value = 0 }
        }
        item = {
            value = aub_policy_personal_only_continue_choice
            localization = aub_policy_personal_only_continue
            current_description = aub_policy_personal_only_continue_desc
            is_valid = { always = yes }
            ai_chance = { value = 0 }
        }
        item = {
            value = aub_policy_treasury_first_pause_choice
            localization = aub_policy_treasury_first_pause
            current_description = aub_policy_treasury_first_pause_desc
            is_valid = { always = yes }
            ai_chance = { value = 0 }
        }
        item = {
            value = aub_policy_treasury_first_continue_choice
            localization = aub_policy_treasury_first_continue
            current_description = aub_policy_treasury_first_continue_desc
            is_valid = { always = yes }
            is_default = yes
            ai_chance = { value = 100 }
        }
    }

    effect = {
        custom_tooltip = enable_auto_build_text
        remove_character_flag = aub_funding_treasury_only
        remove_character_flag = aub_funding_personal_only
        remove_character_flag = aub_pause_when_over_domain_limit
        if = {
            limit = {
                OR = {
                    scope:aub_policy_treasury_only_pause_choice = yes
                    scope:aub_policy_treasury_only_continue_choice = yes
                }
            }
            add_character_flag = aub_funding_treasury_only
            custom_tooltip = aub_funding_treasury_only_desc
        }
        else_if = {
            limit = {
                OR = {
                    scope:aub_policy_personal_only_pause_choice = yes
                    scope:aub_policy_personal_only_continue_choice = yes
                }
            }
            add_character_flag = aub_funding_personal_only
            custom_tooltip = aub_funding_personal_only_desc
        }
        else = {
            custom_tooltip = aub_funding_treasury_first_desc
        }
        if = {
            limit = {
                OR = {
                    scope:aub_policy_treasury_only_pause_choice = yes
                    scope:aub_policy_personal_only_pause_choice = yes
                    scope:aub_policy_treasury_first_pause_choice = yes
                }
            }
            add_character_flag = aub_pause_when_over_domain_limit
            custom_tooltip = aub_over_domain_pause_desc
        }
        else = {
            custom_tooltip = aub_over_domain_continue_desc
        }
        add_character_flag = enable_auto_build
        # 经旧存档兼容入口建立或复用唯一的全局循环。
        trigger_event = {
            id = auto_build.0003
            days = 1
        }
    }
    
    ai_potential = {
        always = no
    }

    ai_will_do = {
        base = 0
    }
}
