namespace = auto_build

# 启用入口兼容上游版本已经排队的 auto_build.0003，并汇入唯一全局循环。
auto_build.0003 = {
    type = character_event
    trigger = {
        is_ai = no
        has_character_flag = enable_auto_build
    }
    hidden = yes
    
    immediate = {
        hidden_effect = {
            aub_start_global_loop_effect = yes
        }
    }
}

# 自动建造执行事件
auto_build.0004 = {
    type = character_event
    trigger = {
        is_ai = no
        has_character_flag = enable_auto_build
    }
    hidden = yes
    
    immediate = {
        hidden_effect = {
            save_scope_as = aub_payer
            # Building on_complete receives the construction payer as
            # scope:character in vanilla. add_building preserves saved scopes,
            # so provide the same context for immediate automatic upgrades.
            save_scope_as = character
            # 仅遍历当前玩家直接持有、未出租且能承载普通省份建筑的地产。
            every_directly_owned_province = {
                limit = {
                    has_ongoing_construction = no
                    barony = { is_leased_out = no }
                    OR = {
                        has_holding_type = castle_holding
                        has_holding_type = city_holding
                        has_holding_type = church_holding
                        has_holding_type = tribal_holding
                        has_holding_type = temple_citadel_holding
                    }
                }
                province_owner = { save_scope_as = holder }
                province_owner = { save_scope_as = build_owner }
                county = { save_scope_as = county }
                aub_upgrade_all_supported_buildings_effect = yes
            }
        }
    }
}

# 单实例全局循环。没有启用该功能的真人玩家时，循环自行停止并释放 seed。
auto_build.0005 = {
    scope = none
    trigger = { has_global_variable = aub_auto_build_loop_started }
    hidden = yes

    immediate = {
        if = {
            limit = {
                any_player = {
                    is_ai = no
                    has_character_flag = enable_auto_build
                }
            }
            every_player = {
                limit = {
                    is_ai = no
                    has_character_flag = enable_auto_build
                }
                trigger_event = { id = auto_build.0004 }
            }
            trigger_event = { id = auto_build.0005 days = 15 }
        }
        else = {
            remove_global_variable = aub_auto_build_loop_started
        }
    }
}
