# Acceptance-only events for actual AI death and no-heir Game Over coverage.
namespace = xar

xar.0901 = {
	hidden = yes
	trigger = { is_ai = no }
	immediate = {
		if = {
			limit = {
				global_var:xa_test_ai_death_observed = 1
				global_var:xa_run_score = -777
			}
			debug_log = "XAR: TEST PASS ai_actual_death"
			debug_log = "XAR: TEST PASS ai_on_death_blocked"
			set_global_variable = { name = xa_test_no_heir_waits value = 0 }
			trigger_event = { id = xar.0902 days = 1 }
		}
		else = {
			debug_log = "XAR: TEST FAIL ai_on_death_blocked"
			debug_log = "XAR: TEST DONE death_edges"
		}
	}
}

xar.0902 = {
	hidden = yes
	trigger = { is_ai = no }
	immediate = {
		if = {
			limit = { exists = player_heir global_var:xa_test_no_heir_waits < 30 }
			player_heir = { add_trait = disinherited }
			change_global_variable = { name = xa_test_no_heir_waits add = 1 }
			trigger_event = { id = xar.0902 days = 1 }
		}
		else_if = {
			limit = { NOT = { exists = player_heir } }
			debug_log = "XAR: TEST PASS no_heir_precondition"
			add_character_flag = xa_test_no_heir_death_target
			death = { death_reason = death_old_age }
		}
		else = {
			debug_log = "XAR: TEST FAIL no_heir_precondition"
			debug_log = "XAR: TEST DONE death_edges"
		}
	}
}

# Recompute through the production scorer, then cross an event boundary before
# reading its committed globals through localization-free generated wire bits.
xar.0920 = {
	hidden = yes
	trigger = { is_ai = no }
	immediate = {
		xar_compute_score_effect = yes
		trigger_event = xar.0921
	}
}

xar.0921 = {
	hidden = yes
	trigger = {
		OR = {
			is_ai = no
			AND = {
				has_global_variable = xa_balance_active
				exists = global_var:xa_balance_fixture_character
				global_var:xa_balance_fixture_character = root
				has_global_variable = xa_player_pact_active
				exists = global_var:xa_player_pact_character
				global_var:xa_player_pact_character = root
			}
		}
	}
	immediate = {
		debug_log = "XAR: BALANCE SAMPLE BEGIN"
		xar_acceptance_balance_emit_wire_effect = yes
		debug_log = "XAR: BALANCE SAMPLE END"
		if = {
			limit = { global_var:xa_balance_sample_kind = 2 }
			debug_log = "XAR: BALANCE MIN 30"
		}
		else_if = {
			limit = { global_var:xa_balance_sample_kind = 3 }
			debug_log = "XAR: BALANCE DONE horizon_40"
		}
		else_if = {
			limit = {
				global_var:xa_balance_sample_kind = 4
				has_global_variable = xa_balance_min_reached
			}
			debug_log = "XAR: BALANCE DONE natural_death"
		}
		else_if = {
			limit = { global_var:xa_balance_sample_kind = 4 }
			debug_log = "XAR: BALANCE DONE early_death"
		}
	}
}

# Thirty years is the minimum useful observation; a survivor continues to the
# forty-year right-censoring horizon.
xar.0927 = {
	hidden = yes
	trigger = { is_ai = no }
	immediate = {
		if = {
			limit = { has_character_flag = xa_enabled }
			debug_log = "XAR: BALANCE MIN enabled"
		}
		else = { debug_log = "XAR: BALANCE MIN missing enabled" }
		if = {
			limit = { has_character_flag = xa_balance_fixture_player }
			debug_log = "XAR: BALANCE MIN fixture flag"
		}
		else = { debug_log = "XAR: BALANCE MIN missing fixture flag" }
		if = {
			limit = {
				has_global_variable = xa_player_pact_active
				exists = global_var:xa_player_pact_character
				global_var:xa_player_pact_character = root
			}
			debug_log = "XAR: BALANCE MIN player scope"
		}
		else = { debug_log = "XAR: BALANCE MIN missing player scope" }
		set_global_variable = xa_balance_min_reached
		set_global_variable = { name = xa_balance_sample_kind value = 2 }
		trigger_event = xar.0920
	}
}
