# Opening shop: spend the imported points copy on attributes (25 points each).
xar.0001 = {
	type = character_event
	theme = stewardship
	title = xar.0001.title
	desc = xar.0001.desc

	immediate = {
		debug_log = "XAR: shop event fired"
		if = {
			limit = { NOT = { has_global_variable = xa_local_points } }
			set_global_variable = {
				name = xa_local_points
				value = 0
			}
		}
		save_scope_value_as = {
			name = xar_points
			value = global_var:xa_local_points
		}
	}

	option = {
		name = xar.0001.diplomacy
		trigger = { global_var:xa_local_points >= 25 }
		change_global_variable = {
			name = xa_local_points
			add = -25
		}
		add_diplomacy_skill = 1
		if = {
			limit = { global_var:xa_local_points >= 25 }
			trigger_event = { id = xar.0001 }
		}
	}

	option = {
		name = xar.0001.martial
		trigger = { global_var:xa_local_points >= 25 }
		change_global_variable = {
			name = xa_local_points
			add = -25
		}
		add_martial_skill = 1
		if = {
			limit = { global_var:xa_local_points >= 25 }
			trigger_event = { id = xar.0001 }
		}
	}

	option = {
		name = xar.0001.stewardship
		trigger = { global_var:xa_local_points >= 25 }
		change_global_variable = {
			name = xa_local_points
			add = -25
		}
		add_stewardship_skill = 1
		if = {
			limit = { global_var:xa_local_points >= 25 }
			trigger_event = { id = xar.0001 }
		}
	}

	option = {
		name = xar.0001.intrigue
		trigger = { global_var:xa_local_points >= 25 }
		change_global_variable = {
			name = xa_local_points
			add = -25
		}
		add_intrigue_skill = 1
		if = {
			limit = { global_var:xa_local_points >= 25 }
			trigger_event = { id = xar.0001 }
		}

	option = {

# Death: show the run score, the previous record, and the delta.
# Score was computed and the record bit already written by xar_on_death.
xar.1001 = {
	type = character_event
	theme = skull
	title = xar.1001.title
	desc = xar.1001.desc

	immediate = {
		debug_log = "XAR: score event fired"
		save_scope_value_as = {
			name = xar_score
			value = global_var:xa_run_score
		}
		save_scope_value_as = {
			name = xar_old
			value = global_var:xa_old_record
		}
		save_scope_value_as = {
			name = xar_delta
			value = global_var:xa_score_delta
		}
	}

	option = {
		name = xar.1001.new_record
		trigger = { global_var:xa_run_score > global_var:xa_old_record }
	}

	option = {
		name = xar.1001.no_record
		trigger = { NOT = { global_var:xa_run_score > global_var:xa_old_record } }
	}
}
