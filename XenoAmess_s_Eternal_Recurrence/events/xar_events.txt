namespace = xar

# Opening shop: spend the imported points copy on attributes (25 points each).
xar.0001 = {
	type = character_event
	theme = stewardship
	title = xar.0001.title
	desc = xar.0001.desc

	immediate = {
		debug_log = "XAR: shop event fired"
		if = {
			limit = { NOT = { has_global_variable = xa_local_points } }
			set_global_variable = {
				name = xa_local_points
				value = 0
			}
		}
		save_scope_value_as = {
			name = xar_points
			value = global_var:xa_local_points
		}
	}

	option = {
		name = xar.0001.diplomacy
		trigger = { global_var:xa_local_points >= 25 }
		change_global_variable = {
			name = xa_local_points
			add = -25
		}
		add_diplomacy_skill = 1
		if = {
			limit = { global_var:xa_local_points >= 25 }
			trigger_event = { id = xar.0001 }
		}
	}

	option = {
		name = xar.0001.martial
		trigger = { global_var:xa_local_points >= 25 }
		change_global_variable = {
			name = xa_local_points
			add = -25
		}
		add_martial_skill = 1
		if = {
			limit = { global_var:xa_local_points >= 25 }
			trigger_event = { id = xar.0001 }
		}
	}

	option = {
		name = xar.0001.stewardship
		trigger = { global_var:xa_local_points >= 25 }
		change_global_variable = {
			name = xa_local_points
			add = -25
		}
		add_stewardship_skill = 1
		if = {
			limit = { global_var:xa_local_points >= 25 }
			trigger_event = { id = xar.0001 }
		}
	}

	option = {
		name = xar.0001.intrigue
		trigger = { global_var:xa_local_points >= 25 }
		change_global_variable = {
			name = xa_local_points
			add = -25
		}
		add_intrigue_skill = 1
		if = {
			limit = { global_var:xa_local_points >= 25 }
			trigger_event = { id = xar.0001 }
		}
	}

	option = {
		name = xar.0001.learning
		trigger = { global_var:xa_local_points >= 25 }
		change_global_variable = {
			name = xa_local_points
			add = -25
		}
		add_learning_skill = 1
		if = {
			limit = { global_var:xa_local_points >= 25 }
			trigger_event = { id = xar.0001 }
		}
	}

	option = {
		name = xar.0001.prowess
		trigger = { global_var:xa_local_points >= 25 }
		change_global_variable = {
			name = xa_local_points
			add = -25
		}
		add_prowess_skill = 1
		if = {
			limit = { global_var:xa_local_points >= 25 }
			trigger_event = { id = xar.0001 }
		}
	}

	option = {
		name = xar.0001.finish
	}
