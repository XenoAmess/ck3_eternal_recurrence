namespace = xar

# The pact: offered at game start. Declining disables the mod for this run.
xar.0002 = {
	type = character_event
	trigger = { is_ai = no }
	theme = mental_health
	override_background = { reference = xar_glassfire }
	title = xar.0002.title
	desc = xar.0002.desc
	immediate = {
		if = {
			limit = { has_global_variable = xa_full_ui_test_active }
			debug_log = "XAR: UI pact window opened"
		}
	}

	option = {
		name = xar.0002.accept
		xar_enable_player_pact_effect = yes
		# Init shop state HERE, once. Do NOT gate on has_global_variable: the
		# loader statically registers every referenced global (check = true,
		# value still none), so gated init never runs and reads fail. And
		# xar.0001 re-fires on every page flip, so its immediate is no place
		# for one-time init either.
		xar_initialize_run_state_effect = yes
		if = {
			limit = { has_global_variable = xa_full_ui_test_active }
			set_global_variable = { name = xa_local_points value = 200 }
			debug_log = "XAR: UI pact accepted"
			trigger_event = { id = xar.0001 }
		}
		else_if = {
			limit = { global_var:xa_local_points = 0 }
			debug_log = "XAR: first life flow opened"
			trigger_event = xar.0010
		}
		else = { trigger_event = xar.0001 }
	}

	option = {
		name = xar.0002.decline
		trigger_event = xar.0003
	}
}

# ---------- blessing/curse sessions ----------
# After the shop closes, the lord offers blessings; each blessing taken forces
# one of two constrained curses. Each pair grants 1 Glassfire Gaze XP; declining
# a session costs 1% final score. The lord returns every 3 years (xar.0006).

xar.0004 = {
	type = character_event
	trigger = { is_ai = no }
	theme = mental_health
	override_background = { reference = xar_glassfire }
	title = xar.0004.title
	desc = xar.0004.desc

	immediate = {
		debug_log = "XAR: blessing event fired"
		if = {
			limit = { has_global_variable = xa_ui_test_active }
			set_global_variable = { name = xa_bless_a value = 0 }
			set_global_variable = { name = xa_bless_b value = 50 }
			set_global_variable = { name = xa_bless_c value = 99 }
		}
		else = { xar_draw_blessings_effect = yes }
		save_scope_value_as = { name = xar_bs_n value = global_var:xa_bless_session }
		save_scope_value_as = { name = xar_bc_n value = global_var:xa_bless_count }
		save_scope_value_as = { name = xar_br_n value = global_var:xa_bless_reject_count }
		if = {
			limit = { has_global_variable = xa_ui_test_active }
			debug_log = "XAR: UI bless window opened"
		}
	}

	option = {
		name = xar_bless_option_a
		xar_apply_blessing_effect = { SLOT = a }
		if = {
			limit = { has_global_variable = xa_ui_test_active }
			debug_log = "XAR: UI bless accepted"
		}
		trigger_event = { id = xar.0005 }

# Self-test import waiter (rule xar_selftest): polls daily until the GUI-bridge
# import protocol is consumed, then arms the checker and kills
# root. (The bridge tick is starved while modal windows are open, so dying at
# game start races it. 30-day cap keeps the run deterministic if the bridge
# broke.) The selftest waits on consumed rather than merely seeing the imported
# value, proving that the ready consumer also copied spendable points.
xar.0008 = {
	hidden = yes
	trigger = { is_ai = no }
	immediate = {
		if = {
			limit = { global_var:xa_import_consumed = 1 }
			debug_log = "XAR: import delivered, proceeding to death"
			# runtime coverage: apply every pool entry before dying
			xar_test_sweep_effect = yes
			set_global_variable = {
				name = xa_test_score_preview
				value = xar_current_score_value
			}
			set_global_variable = { name = xa_selftest_armed value = 1 }
			death = { death_reason = death_old_age }
		}
		else_if = {
			limit = { global_var:xa_selftest_waits >= 30 }
			debug_log = "XAR: TEST FAIL import_timeout"
			set_global_variable = { name = xa_selftest_armed value = 1 }
			death = { death_reason = death_old_age }
		}
		else = {
			change_global_variable = { name = xa_selftest_waits add = 1 }
			trigger_event = { id = xar.0008 days = 1 }
		}
	}
}
# Self-test checker (rule xar_selftest): runs NESTED inside xar.1001 (the
# settlement window hard-pauses the game, so a delayed event would never fire).
# Validates import bridge, scoring, blessing counters, record write.
# Ends the run with XAR: TEST DONE.
xar.0007 = {
	hidden = yes
	trigger = { is_ai = no }
	immediate = {
		if = {
			limit = { global_var:xa_run_score > 0 }
			debug_log = "XAR: TEST PASS score_positive"
		}
		else = { debug_log = "XAR: TEST FAIL score_positive" }
		if = {
			limit = {
				global_var:xa_run_score >= { value = global_var:xa_score_before_reject multiply = 0.989 }
				global_var:xa_run_score <= { value = global_var:xa_score_before_reject multiply = 0.991 }
			}
			debug_log = "XAR: TEST PASS reject_penalty"
		}
		else = { debug_log = "XAR: TEST FAIL reject_penalty" }
		if = {
			limit = {
				global_var:xa_run_score >= { value = global_var:xa_test_score_preview subtract = 0.01 }
				global_var:xa_run_score <= { value = global_var:xa_test_score_preview add = 0.01 }
			}
			debug_log = "XAR: TEST PASS score_preview"
		}
		else = { debug_log = "XAR: TEST FAIL score_preview" }
		if = {
			limit = { global_var:xa_bless_count = 1 }
			debug_log = "XAR: TEST PASS bless_count"
		}
		else = { debug_log = "XAR: TEST FAIL bless_count" }
		# record beaten + write path taken (the tutorial-bit persistence itself
		# is asserted runner-side by reading tutorial.txt before restore)
		if = {
			limit = { global_var:xa_record_candidate > global_var:xa_old_record }
			debug_log = "XAR: TEST PASS record_write"
		}
		else = { debug_log = "XAR: TEST FAIL record_write" }
		debug_log = "XAR: TEST DONE"
	}
}
