# ---------- apply (SLOT = a/b/c) ----------

xar_apply_blessing_effect = {
	if = { limit = { global_var:xa_bless_$SLOT$ = 0 } add_gold = 150 }
	if = { limit = { global_var:xa_bless_$SLOT$ = 1 } add_prestige = 225 }
	if = { limit = { global_var:xa_bless_$SLOT$ = 2 } add_piety = 225 }
	if = { limit = { global_var:xa_bless_$SLOT$ = 3 } change_influence = 75 }
	if = { limit = { global_var:xa_bless_$SLOT$ = 4 } add_character_modifier = { modifier = xar_bless_health days = 3650 } }
	if = { limit = { global_var:xa_bless_$SLOT$ = 5 } add_diplomacy_skill = 3 }
	if = { limit = { global_var:xa_bless_$SLOT$ = 6 } add_martial_skill = 3 }
	if = { limit = { global_var:xa_bless_$SLOT$ = 7 } add_learning_lifestyle_xp = 750 }
	if = { limit = { global_var:xa_bless_$SLOT$ = 8 } add_trait = physique_good_1 }
	if = { limit = { global_var:xa_bless_$SLOT$ = 9 } add_renown = 150 }
	change_global_variable = { name = xa_bless_session add = 1 }
	change_global_variable = { name = xa_bless_count add = 1 }
}
