# Acceptance-only proof of the production one-pair-per-meeting bargain cycle.
xar_acceptance_bargain_reopen_start_effect = {
	if = {
		limit = { is_ai = no global_var:xa_global_record_imported = 2 }
		xar_enable_player_pact_effect = yes
		xar_initialize_run_state_effect = yes
		set_global_variable = xa_bargain_reopen_active
		set_global_variable = { name = xa_bargain_reopen_count value = 0 }
		debug_log = "XAR: TEST bargain reopen begin"
		if = {
			limit = {
				has_character_flag = xa_enabled
				has_trait = xar_glassfire_gaze
				global_var:xa_bless_count = 0
				global_var:xa_bless_session = 0
				global_var:xa_bless_reject_count = 0
				NOT = { has_trait_xp = { trait = xar_glassfire_gaze value >= 1 } }
			}
			debug_log = "XAR: TEST PASS bargain_initial_0"
		}
		else = { debug_log = "XAR: TEST FAIL bargain_initial_0" }
		trigger_event = xar.0004
	}
}

xar_acceptance_bargain_open_effect = {
	if = {
		limit = {
			is_ai = no
			has_character_flag = xa_enabled
			has_global_variable = xa_bargain_reopen_active
		}
		if = {
			limit = { global_var:xa_bless_count = 0 }
			if = {
				limit = {
					global_var:xa_bless_session = 0
					global_var:xa_bless_reject_count = 0
					global_var:xa_bargain_reopen_count = 0
					NOT = { has_trait_xp = { trait = xar_glassfire_gaze value >= 1 } }
				}
				debug_log = "XAR: TEST PASS bargain_pair_1_open"
			}
			else = { debug_log = "XAR: TEST FAIL bargain_pair_1_open" }
			set_global_variable = { name = xa_bless_a value = 0 }
			set_global_variable = { name = xa_bless_b value = 1 }
			set_global_variable = { name = xa_bless_c value = 2 }
		}
		else_if = {
			limit = { global_var:xa_bless_count = 1 }
			if = {
				limit = {
					global_var:xa_bless_session = 0
					global_var:xa_bless_reject_count = 0
					global_var:xa_bargain_reopen_count = 1
					has_trait_xp = { trait = xar_glassfire_gaze value >= 1 }
					has_trait_xp = { trait = xar_glassfire_gaze value < 2 }
				}
				debug_log = "XAR: TEST PASS bargain_pair_2_open"
			}
			else = { debug_log = "XAR: TEST FAIL bargain_pair_2_open" }
			set_global_variable = { name = xa_bless_a value = 1 }
			set_global_variable = { name = xa_bless_b value = 2 }
			set_global_variable = { name = xa_bless_c value = 3 }
		}
		else_if = {
			limit = { global_var:xa_bless_count = 2 }
			if = {
				limit = {
					global_var:xa_bless_session = 0
					global_var:xa_bless_reject_count = 0
					global_var:xa_bargain_reopen_count = 2
					has_trait_xp = { trait = xar_glassfire_gaze value >= 2 }
					has_trait_xp = { trait = xar_glassfire_gaze value < 3 }
				}
				debug_log = "XAR: TEST PASS bargain_pair_3_open"
			}
			else = { debug_log = "XAR: TEST FAIL bargain_pair_3_open" }
			set_global_variable = { name = xa_bless_a value = 2 }
			set_global_variable = { name = xa_bless_b value = 3 }
			set_global_variable = { name = xa_bless_c value = 4 }
		}
		else_if = {
			limit = { global_var:xa_bless_count = 3 }
			if = {
				limit = {
					global_var:xa_bless_session = 0
					global_var:xa_bless_reject_count = 0
					global_var:xa_bargain_reopen_count = 3
					has_trait_xp = { trait = xar_glassfire_gaze value >= 3 }
					has_trait_xp = { trait = xar_glassfire_gaze value < 4 }
				}
				debug_log = "XAR: TEST PASS bargain_pair_3_full_reopen"
			}
			else = { debug_log = "XAR: TEST FAIL bargain_pair_3_full_reopen" }
			debug_log = "XAR: TEST DONE bargain-reopen"
		}
		else = {
			debug_log = "XAR: TEST FAIL bargain_unexpected_open_count"
			debug_log = "XAR: TEST DONE bargain-reopen"
		}
	}
}

				global_var:xa_bless_reject_count = 0
				global_var:xa_curse_a = 2
