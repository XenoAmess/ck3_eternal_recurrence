# Paid custom courtier creation. No character or title exists before confirm.
xar_cc_initialize_effect = {
	if = {
		limit = { NOT = { has_character_flag = xar_cc_initialized } }
		set_variable = { name = xar_cc_female value = 0 }
		set_variable = { name = xar_cc_age value = 30 }
		set_variable = { name = xar_cc_education value = 2 }

		set_variable = { name = xar_cc_cmd_organizer value = 0 }
		set_variable = { name = xar_cc_cmd_military_engineer value = 0 }
		set_variable = { name = xar_cc_cmd_aggressive_attacker value = 0 }
		set_variable = { name = xar_cc_cmd_unyielding_defender value = 0 }

		set_variable = { name = xar_cc_phys_beauty value = 0 }
		set_variable = { name = xar_cc_phys_physique value = 0 }
		set_variable = { name = xar_cc_phys_intellect value = 0 }

		set_variable = { name = xar_cc_pers_brave value = 0 }
		set_variable = { name = xar_cc_pers_craven value = 0 }
		set_variable = { name = xar_cc_pers_calm value = 0 }
		set_variable = { name = xar_cc_pers_wrathful value = 0 }
		set_variable = { name = xar_cc_pers_diligent value = 0 }
		set_variable = { name = xar_cc_pers_lazy value = 0 }
		add_character_flag = xar_cc_initialized
	}
}

xar_cc_complete_purchase_effect = {
	if = {
		limit = {
			xar_cc_ui_access_trigger = yes
			xar_cc_valid_configuration_trigger = yes
			gold >= xar_courtier_creator_cost
			exists = culture
			exists = faith
			exists = location
		}
		# Closing first makes queued or double-clicked executions harmless.
		remove_character_flag = xar_cc_open
		create_character = {
			location = root.location
			employer = root
			culture = root.culture
			faith = root.faith
			dynasty = none
			age = root.var:xar_cc_age
			gender_female_chance = {
				if = {
					limit = { root.var:xar_cc_female = 1 }
					add = 100
				}
				else = { add = 0 }
			}
			random_traits = no
			diplomacy = 6
			martial = 6
			stewardship = 6
			intrigue = 6
			learning = 6
			prowess = 6
			save_scope_as = xar_cc_created_courtier
		}
		if = {
			limit = { exists = scope:xar_cc_created_courtier }
			remove_short_term_gold = xar_courtier_creator_cost
			scope:xar_cc_created_courtier = {
				if = {
					limit = { root.var:xar_cc_education = 1 }
					add_trait = education_diplomacy_3
				}
				else_if = {
					limit = { root.var:xar_cc_education = 2 }
					add_trait = education_martial_3
				}
				else_if = {
					limit = { root.var:xar_cc_education = 3 }
					add_trait = education_stewardship_3
				}
				else_if = {
					limit = { root.var:xar_cc_education = 4 }
					add_trait = education_intrigue_3
				}
				else_if = {
					limit = { root.var:xar_cc_education = 5 }
					add_trait = education_learning_3
				}

				if = { limit = { root.var:xar_cc_cmd_organizer = 1 } add_trait = organizer }
				if = { limit = { root.var:xar_cc_cmd_military_engineer = 1 } add_trait = military_engineer }
				if = { limit = { root.var:xar_cc_cmd_aggressive_attacker = 1 } add_trait = aggressive_attacker }
				if = { limit = { root.var:xar_cc_cmd_unyielding_defender = 1 } add_trait = unyielding_defender }

				if = { limit = { root.var:xar_cc_phys_beauty = 1 } add_trait = beauty_good_1 }
				if = { limit = { root.var:xar_cc_phys_physique = 1 } add_trait = physique_good_1 }
				if = { limit = { root.var:xar_cc_phys_intellect = 1 } add_trait = intellect_good_1 }

				if = { limit = { root.var:xar_cc_pers_brave = 1 } add_trait = brave }
				if = { limit = { root.var:xar_cc_pers_craven = 1 } add_trait = craven }
				if = { limit = { root.var:xar_cc_pers_calm = 1 } add_trait = calm }
				if = { limit = { root.var:xar_cc_pers_wrathful = 1 } add_trait = wrathful }
				if = { limit = { root.var:xar_cc_pers_diligent = 1 } add_trait = diligent }
				if = { limit = { root.var:xar_cc_pers_lazy = 1 } add_trait = lazy }

				add_character_flag = {
					flag = blocked_from_leaving
					years = 25
				}
				add_character_flag = xar_cc_created_courtier
				force_character_skill_recalculation = yes
			}
			if = {
				limit = {
					scope:xar_cc_created_courtier = {
						NOT = { is_courtier_of = root }
					}
				}
				add_courtier = scope:xar_cc_created_courtier
			}
			send_interface_toast = {
				title = xar.cc.toast.title
				left_icon = scope:xar_cc_created_courtier
				custom_tooltip = xar.cc.toast.desc
			}
		}
	}
}
