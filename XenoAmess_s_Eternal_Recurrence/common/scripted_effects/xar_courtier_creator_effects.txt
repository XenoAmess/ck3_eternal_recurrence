# Paid custom courtier creation. No character or title exists before confirm.
xar_cc_initialize_effect = {
	if = {
		limit = { NOT = { has_character_flag = xar_cc_v2_initialized } }
		set_variable = { name = xar_cc_female value = 0 }
		set_variable = { name = xar_cc_age value = 30 }
		set_variable = { name = xar_cc_diplomacy value = 6 }
		set_variable = { name = xar_cc_martial value = 6 }
		set_variable = { name = xar_cc_stewardship value = 6 }
		set_variable = { name = xar_cc_intrigue value = 6 }
		set_variable = { name = xar_cc_learning value = 6 }
		set_variable = { name = xar_cc_prowess value = 6 }
		set_variable = { name = xar_cc_same_house value = 0 }
		set_variable = { name = xar_cc_selected_culture value = root.culture }
		set_variable = { name = xar_cc_selected_faith value = root.faith }

		clear_variable_list = xar_cc_selected_education
		clear_variable_list = xar_cc_selected_commander
		clear_variable_list = xar_cc_selected_physical
		clear_variable_list = xar_cc_selected_personality
		clear_variable_list = xar_cc_selected_other
		trait:education_martial_3 = { save_scope_as = xar_cc_default_education }
		add_to_variable_list = {
			name = xar_cc_selected_education
			target = scope:xar_cc_default_education
		}
		set_variable = { name = xar_cc_commander_count value = 0 }
		set_variable = { name = xar_cc_personality_count value = 0 }
		add_character_flag = xar_cc_v2_initialized
	}
}

xar_cc_rebuild_culture_faith_catalogs_effect = {
	root = {
		clear_variable_list = xar_cc_catalog_cultures
		clear_variable_list = xar_cc_catalog_faiths
	}
	every_culture_global = {
		save_scope_as = xar_cc_catalog_culture
		root = {
			add_to_variable_list = {
				name = xar_cc_catalog_cultures
				target = scope:xar_cc_catalog_culture
			}
		}
	}
	every_religion_global = {
		every_faith = {
			save_scope_as = xar_cc_catalog_faith
			root = {
				add_to_variable_list = {
					name = xar_cc_catalog_faiths
					target = scope:xar_cc_catalog_faith
				}
			}
		}
	}
	root = {
		if = {
			limit = { NOT = { exists = var:xar_cc_selected_culture } }
			set_variable = { name = xar_cc_selected_culture value = root.culture }
		}
		else_if = {
			limit = {
				NOT = {
					is_target_in_variable_list = {
						name = xar_cc_catalog_cultures
						target = var:xar_cc_selected_culture
					}
				}
			}
			set_variable = { name = xar_cc_selected_culture value = root.culture }
		}
		if = {
			limit = { NOT = { exists = var:xar_cc_selected_faith } }
			set_variable = { name = xar_cc_selected_faith value = root.faith }
		}
		else_if = {
			limit = {
				NOT = {
					is_target_in_variable_list = {
						name = xar_cc_catalog_faiths
						target = var:xar_cc_selected_faith
					}
				}
			}
			set_variable = { name = xar_cc_selected_faith value = root.faith }
		}
	}
}

xar_cc_remove_selected_trait_effect = {
	if = {
		limit = {
			is_target_in_variable_list = {
				name = xar_cc_selected_education
				target = $TRAIT$
			}
		}
		remove_list_variable = { name = xar_cc_selected_education target = $TRAIT$ }
	}
	if = {
		limit = {
			is_target_in_variable_list = {
				name = xar_cc_selected_commander
				target = $TRAIT$
			}
		}
		remove_list_variable = { name = xar_cc_selected_commander target = $TRAIT$ }
		change_variable = { name = xar_cc_commander_count add = -1 }
	}
	if = {
		limit = {
			is_target_in_variable_list = {
				name = xar_cc_selected_physical
				target = $TRAIT$
			}
		}
		remove_list_variable = { name = xar_cc_selected_physical target = $TRAIT$ }
	}
	if = {
		limit = {
			is_target_in_variable_list = {
				name = xar_cc_selected_personality
				target = $TRAIT$
			}
		}
		remove_list_variable = { name = xar_cc_selected_personality target = $TRAIT$ }
		change_variable = { name = xar_cc_personality_count add = -1 }
	}
	if = {
		limit = {
			is_target_in_variable_list = {
				name = xar_cc_selected_other
				target = $TRAIT$
			}
		}
		remove_list_variable = { name = xar_cc_selected_other target = $TRAIT$ }
	}
}

xar_cc_collect_invalid_selected_traits_effect = {
	every_in_list = {
		variable = $LIST$
		save_scope_as = xar_cc_candidate_trait
		root = {
			if = {
				limit = {
					NOT = { xar_cc_candidate_trait_metadata_valid_trigger = yes }
				}
				add_to_variable_list = {
					name = xar_cc_invalid_selected_traits
					target = scope:xar_cc_candidate_trait
				}
			}
		}
	}
}

xar_cc_prune_invalid_selected_traits_effect = {
	clear_variable_list = xar_cc_invalid_selected_traits
	xar_cc_collect_invalid_selected_traits_effect = { LIST = xar_cc_selected_education }
	xar_cc_collect_invalid_selected_traits_effect = { LIST = xar_cc_selected_commander }
	xar_cc_collect_invalid_selected_traits_effect = { LIST = xar_cc_selected_physical }
	xar_cc_collect_invalid_selected_traits_effect = { LIST = xar_cc_selected_personality }
	xar_cc_collect_invalid_selected_traits_effect = { LIST = xar_cc_selected_other }
	every_in_list = {
		variable = xar_cc_invalid_selected_traits
		save_scope_as = xar_cc_candidate_trait
		root = {
			xar_cc_remove_selected_trait_effect = {
				TRAIT = scope:xar_cc_candidate_trait
			}
		}
	}
	clear_variable_list = xar_cc_invalid_selected_traits
}

xar_cc_adjust_numeric_variable_effect = {
	set_variable = {
		name = $VARIABLE$
		value = {
			value = var:$VARIABLE$
			add = $DELTA$
			min = $MIN$
			max = $MAX$
		}
	}
}

xar_cc_apply_selected_trait_list_effect = {
	every_in_list = {
		variable = $LIST$
		save_scope_as = xar_cc_selected_trait
		scope:xar_cc_created_courtier = {
			add_trait = scope:xar_cc_selected_trait
		}
	}
}

xar_cc_complete_purchase_effect = {
	if = {
		limit = {
			xar_cc_ui_access_trigger = yes
			xar_cc_valid_configuration_trigger = yes
			gold >= xar_courtier_creator_cost
			exists = var:xar_cc_selected_culture
			exists = var:xar_cc_selected_faith
			is_target_in_variable_list = {
				name = xar_cc_catalog_cultures
				target = var:xar_cc_selected_culture
			}
			is_target_in_variable_list = {
				name = xar_cc_catalog_faiths
				target = var:xar_cc_selected_faith
			}
		}
		# Closing first makes queued or double-clicked executions harmless.
		remove_character_flag = xar_cc_open
		save_scope_value_as = { name = xar_cc_purchase_diplomacy value = var:xar_cc_diplomacy }
		save_scope_value_as = { name = xar_cc_purchase_martial value = var:xar_cc_martial }
		save_scope_value_as = { name = xar_cc_purchase_stewardship value = var:xar_cc_stewardship }
		save_scope_value_as = { name = xar_cc_purchase_intrigue value = var:xar_cc_intrigue }
		save_scope_value_as = { name = xar_cc_purchase_learning value = var:xar_cc_learning }
		save_scope_value_as = { name = xar_cc_purchase_prowess value = var:xar_cc_prowess }
		create_character = {
			employer = root
			culture = root.var:xar_cc_selected_culture
			faith = root.var:xar_cc_selected_faith
			dynasty = none
			age = root.var:xar_cc_age
			gender_female_chance = {
				if = {
					limit = { root.var:xar_cc_female = 1 }
					add = 100
				}
				else = { add = 0 }
			}
			random_traits = no
			diplomacy = 0
			martial = 0
			stewardship = 0
			intrigue = 0
			learning = 0
			prowess = 0
			save_scope_as = xar_cc_created_courtier
		}
		if = {
			limit = { exists = scope:xar_cc_created_courtier }
			scope:xar_cc_created_courtier = {
				add_diplomacy_skill = scope:xar_cc_purchase_diplomacy
				add_martial_skill = scope:xar_cc_purchase_martial
				add_stewardship_skill = scope:xar_cc_purchase_stewardship
				add_intrigue_skill = scope:xar_cc_purchase_intrigue
				add_learning_skill = scope:xar_cc_purchase_learning
				add_prowess_skill = scope:xar_cc_purchase_prowess
			}
			xar_cc_apply_selected_trait_list_effect = { LIST = xar_cc_selected_education }
			xar_cc_apply_selected_trait_list_effect = { LIST = xar_cc_selected_commander }
			xar_cc_apply_selected_trait_list_effect = { LIST = xar_cc_selected_physical }
			xar_cc_apply_selected_trait_list_effect = { LIST = xar_cc_selected_personality }
			xar_cc_apply_selected_trait_list_effect = { LIST = xar_cc_selected_other }
			scope:xar_cc_created_courtier = {
				if = {
					limit = {
						root.var:xar_cc_same_house = 1
						exists = root.house
					}
					set_house = root.house
				}
				add_character_flag = {
					flag = blocked_from_leaving
					years = 25
				}
				force_character_skill_recalculation = yes
			}
			remove_short_term_gold = xar_courtier_creator_cost
			if = {
				limit = {
					scope:xar_cc_created_courtier = {
						NOT = { is_courtier_of = root }
					}
				}
				add_courtier = scope:xar_cc_created_courtier
			}
			# XAR_ACCEPTANCE_ONLY_BEGIN
			if = {
				limit = { has_global_variable = xa_test_cc_active }
				set_global_variable = {
					name = xa_test_cc_created_courtier
					value = scope:xar_cc_created_courtier
				}
				xar_acceptance_courtier_purchase_observer_effect = yes
			}
			# XAR_ACCEPTANCE_ONLY_END
			send_interface_toast = {
				title = xar.cc.toast.title
				left_icon = scope:xar_cc_created_courtier
				custom_tooltip = xar.cc.toast.desc
			}
		}
	}
}
