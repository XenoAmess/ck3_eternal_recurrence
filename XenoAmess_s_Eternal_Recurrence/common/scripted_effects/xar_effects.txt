# Import consumer. The GUI bridge is the only producer of ready=1; points and
# the opening flow are therefore never started from an uninitialized record.
xar_consume_import_effect = {
	if = {
		limit = {
			global_var:xa_import_ready = 1
			global_var:xa_import_consumed = 0
			is_ai = no
		}
		xar_calculate_inherited_budget_effect = yes
		set_global_variable = { name = xa_import_ready value = 0 }
		set_global_variable = { name = xa_import_consumed value = 1 }
		debug_log = "XAR: import consumed, opening flow"
		# XAR_ACCEPTANCE_ONLY_BEGIN
		if = {
			limit = { has_game_rule = xar_selftest }
			if = {
				limit = { global_var:xa_global_record_imported = 1 }
				xar_acceptance_death_edges_start_effect = yes
			}
			else_if = {
				limit = { global_var:xa_global_record_imported = 2 }
				xar_acceptance_bargain_reopen_start_effect = yes
			}
			else_if = {
				limit = { global_var:xa_global_record_imported = 3 }
				xar_acceptance_progression_start_effect = yes
			}
			else_if = {
				limit = { global_var:xa_global_record_imported = 4 }
				xar_acceptance_scoring_matrix_start_effect = yes
			}
			else_if = {
				limit = { global_var:xa_global_record_imported = 5 }
				xar_acceptance_death_with_heir_start_effect = yes
			}
			else_if = {
				limit = { global_var:xa_global_record_imported = 6 }
				xar_acceptance_courtier_start_effect = yes
			}
			else = {
				debug_log = "XAR: TEST selftest begin"
				xar_selftest_effect = yes
			}
		}
		# XAR_ACCEPTANCE_ONLY_END
		if = {
			limit = { has_game_rule = xar_on }
			debug_log = "XAR: rule is on, offering pact"
			trigger_event = xar.0002
		}
	}
}

xar_calculate_inherited_budget_effect = {
	set_global_variable = { name = xa_local_points value = 0 }
	if = {
		limit = { global_var:xa_inheritance_percent = 25 }
		set_global_variable = { name = xa_local_points value = { value = global_var:xa_global_record_imported multiply = 0.25 floor = yes } }
	}
	else_if = {
		limit = { global_var:xa_inheritance_percent = 50 }
		set_global_variable = { name = xa_local_points value = { value = global_var:xa_global_record_imported multiply = 0.5 floor = yes } }
	}
	else_if = {
		limit = { global_var:xa_inheritance_percent = 100 }
		set_global_variable = { name = xa_local_points value = { value = global_var:xa_global_record_imported } }
	}
}

# Freeze the selected challenge rules into one explicit run identity.
# Track code = score basis * 4 + inheritance setting index.
xar_initialize_challenge_mode_effect = {
	set_global_variable = { name = xa_inheritance_percent value = 100 }
	set_global_variable = { name = xa_track_code value = 3 }
	set_global_variable = { name = xa_score_basis value = 0 }
	if = {
		limit = { has_game_rule = xar_inherit_0 }
		set_global_variable = { name = xa_inheritance_percent value = 0 }
		set_global_variable = { name = xa_track_code value = 0 }
	}
	else_if = {
		limit = { has_game_rule = xar_inherit_25 }
		set_global_variable = { name = xa_inheritance_percent value = 25 }
		set_global_variable = { name = xa_track_code value = 1 }
	}
	else_if = {
		limit = { has_game_rule = xar_inherit_50 }
		set_global_variable = { name = xa_inheritance_percent value = 50 }
		set_global_variable = { name = xa_track_code value = 2 }
	}
	if = {
		limit = { has_game_rule = xar_score_growth }
		set_global_variable = { name = xa_score_basis value = 1 }
		change_global_variable = { name = xa_track_code add = 4 }
	}
}

# Descendant engine adapters used by the generated production effect. Scoring
# coefficients, depth expansion, and cleanup expansion come from scoring_data.py.

# Counts the scoped living descendant into the per-tier bucket of their highest
# held title. Called once per living descendant by xar_desc_count_self.
xar_descendant_tier_count_effect = {
	if = {
		limit = { highest_held_title_tier >= tier_hegemony }
		change_global_variable = {
			name = xa_d_t5
			add = 1
		}
	}
	else_if = {
		limit = { highest_held_title_tier >= tier_empire }
		change_global_variable = {
			name = xa_d_t4
			add = 1
		}
	}
	else_if = {
		limit = { highest_held_title_tier >= tier_kingdom }
		change_global_variable = {
			name = xa_d_t3
			add = 1
		}
	}
	else_if = {
		limit = { highest_held_title_tier >= tier_duchy }
		change_global_variable = {
			name = xa_d_t2
			add = 1
		}
	}
	else_if = {
		limit = { highest_held_title_tier >= tier_county }
		change_global_variable = {
			name = xa_d_t1
			add = 1
		}
	}
}

# Root of the caller is the dead character; scope:xar_score_dead/_dyn/_house
# are saved by the generated caller. every_child never errors on childless
# characters, unlike parent-chain trigger links. Dedup uses xar_desc_counted;
# generated cleanup nodes remove it so later runs recount cleanly.
xar_desc_count_self = {
	if = {
		limit = {
			is_alive = yes
			NOT = { has_character_flag = xar_desc_counted }
		}
		add_character_flag = xar_desc_counted
		xar_descendant_tier_count_effect = yes
		if = {
			limit = {
				exists = scope:xar_score_dyn
				exists = dynasty
				dynasty = { this = scope:xar_score_dyn }
			}
			change_global_variable = { name = xa_a_dyn add = 1 }
		}
		if = {
			limit = {
				exists = scope:xar_score_house
				exists = house
				house = { this = scope:xar_score_house }
			}
			change_global_variable = { name = xa_a_hou add = 1 }
		}
	}
}
