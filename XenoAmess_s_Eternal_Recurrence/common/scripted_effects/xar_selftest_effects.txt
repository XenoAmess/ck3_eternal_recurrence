# Self-test harness (game rule xar_enabled = xar_selftest). Drives the whole
# logic chain with scripted assertions at game start; each check logs
# "XAR: TEST PASS/FAIL <name>" to debug.log. Runner: tools/run_acceptance.py.
# NOT for players.

xar_selftest_effect = {
	# ---------- T2: pact accept simulation (same effects as xar.0002 accept) ----------
	add_character_flag = xa_enabled
	add_trait = xar_glassfire_gaze
	set_global_variable = { name = xa_shop_page value = 1 }
	set_global_variable = { name = xa_price_dip value = 25 }
	set_global_variable = { name = xa_price_mar value = 25 }
	set_global_variable = { name = xa_price_ste value = 25 }
	set_global_variable = { name = xa_price_int value = 25 }
	set_global_variable = { name = xa_price_lea value = 25 }
	set_global_variable = { name = xa_price_pro value = 25 }
	set_global_variable = { name = xa_price_gold value = 25 }
	set_global_variable = { name = xa_price_pres value = 15 }
	set_global_variable = { name = xa_price_pie value = 15 }
	set_global_variable = { name = xa_price_inf value = 15 }
	set_global_variable = { name = xa_price_dyn value = 100 }
	set_global_variable = { name = xa_price_life value = 100 }
	set_global_variable = { name = xa_price_reform value = 1133 }
	set_global_variable = { name = xa_lifespan_bought value = 0 }
	set_global_variable = { name = xa_bless_count value = 0 }
	set_global_variable = { name = xa_bless_session value = 0 }
	set_global_variable = { name = xa_bless_reject_count value = 0 }
	set_global_variable = { name = xa_selected_bless_rarity value = 0 }

	if = {
		limit = { has_trait = xar_glassfire_gaze }
		debug_log = "XAR: TEST PASS pact_trait"
	}
	else = { debug_log = "XAR: TEST FAIL pact_trait" }
	if = {
		limit = { has_character_flag = xa_enabled }
		debug_log = "XAR: TEST PASS pact_flag"
	}
	else = { debug_log = "XAR: TEST FAIL pact_flag" }
	if = {
		limit = { global_var:xa_price_dip = 25 }
		debug_log = "XAR: TEST PASS shop_init"
	}
	else = { debug_log = "XAR: TEST FAIL shop_init" }

	# ---------- T3: shop buy simulation (charge + 1.2x inflation) ----------
	set_global_variable = { name = xa_test_points value = 250 }
	change_global_variable = { name = xa_test_points add = { value = global_var:xa_price_dip multiply = -1 } }
	add_diplomacy_skill = 1
	change_global_variable = { name = xa_price_dip multiply = 1.2 }
	if = {
		limit = {
			global_var:xa_price_dip >= 29.9
			global_var:xa_price_dip <= 30.1
		}
		debug_log = "XAR: TEST PASS shop_inflate"
	}
	else = { debug_log = "XAR: TEST FAIL shop_inflate" }
	if = {
		limit = {
			global_var:xa_test_points >= 224.9
			global_var:xa_test_points <= 225.1
		}
		debug_log = "XAR: TEST PASS shop_charge"
	}
	else = { debug_log = "XAR: TEST FAIL shop_charge" }

	# ---------- T4: blessing draw, three distinct ----------
	xar_draw_blessings_effect = yes
	if = {
		limit = {
			NOT = { global_var:xa_bless_a = global_var:xa_bless_b }
			NOT = { global_var:xa_bless_b = global_var:xa_bless_c }
			NOT = { global_var:xa_bless_a = global_var:xa_bless_c }
		}
		debug_log = "XAR: TEST PASS draw_bless_distinct"
	}
	else = { debug_log = "XAR: TEST FAIL draw_bless_distinct" }

	# ---------- T5: apply a legendary blessing through the real dispatcher ----------
	set_global_variable = { name = xa_bless_a value = 99 }
	xar_apply_blessing_effect = { SLOT = a }
	if = {
		limit = {
			global_var:xa_bless_count = 1
			global_var:xa_selected_bless_rarity = 2
		}
		debug_log = "XAR: TEST PASS bless_apply"
	}
	else = { debug_log = "XAR: TEST FAIL bless_apply" }

	# ---------- T6: curse draw, two distinct and both >= legendary-1 ----------
	xar_draw_curses_effect = yes
	if = {
		limit = {
			NOT = { global_var:xa_curse_a = global_var:xa_curse_b }
			global_var:xa_curse_a_rarity >= 1
			global_var:xa_curse_b_rarity >= 1
		}
		debug_log = "XAR: TEST PASS draw_curse_constrained"
	}
	else = { debug_log = "XAR: TEST FAIL draw_curse_constrained" }

	# ---------- T7: applying the mandatory curse completes the pair: +1 XP ----------
	xar_apply_curse_effect = { SLOT = a }
	if = {
		limit = {
			has_trait_xp = { trait = xar_glassfire_gaze value >= 1 }
			has_trait_xp = { trait = xar_glassfire_gaze value < 2 }
		}
		debug_log = "XAR: TEST PASS curse_pair_xp"
	}
	else = { debug_log = "XAR: TEST FAIL curse_pair_xp" }
	add_trait_xp = { trait = xar_glassfire_gaze value = 9 }
	if = {
		limit = { has_trait_xp = { trait = xar_glassfire_gaze value >= 10 } }
		debug_log = "XAR: TEST PASS trait_level_1"
	}
	else = { debug_log = "XAR: TEST FAIL trait_level_1" }
	add_trait_xp = { trait = xar_glassfire_gaze value = 90 }
	if = {
		limit = { has_trait_xp = { trait = xar_glassfire_gaze value >= 100 } }
		debug_log = "XAR: TEST PASS trait_xp_cap"
	}
	else = { debug_log = "XAR: TEST FAIL trait_xp_cap" }

	# ---------- T8: real pact -> shop -> blessing/curse UI chain ----------
	# Production options emit assertions only while this flag is active. The
	# runner clicks every transition; the final curse schedules the death waiter.
	set_global_variable = { name = xa_test_dip_before value = diplomacy }
	set_global_variable = { name = xa_test_gold_before value = gold }
	set_global_variable = xa_full_ui_test_active
	debug_log = "XAR: UI full flow armed"
	trigger_event = xar.0002
}
