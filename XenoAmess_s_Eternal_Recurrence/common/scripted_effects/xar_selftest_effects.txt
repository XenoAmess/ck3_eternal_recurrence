# NOT for players.

			global_var:xa_import_ready = 0
			global_var:xa_import_consumed = 1

	else = { debug_log = "XAR: TEST FAIL inherit_100_uncapped" }
	set_global_variable = { name = xa_global_record_imported value = 5000 }

	xar_calculate_inherited_budget_effect = yes
	if = { limit = { global_var:xa_local_points = 166600 } debug_log = "XAR: TEST PASS inherit_166600" }
	else = { debug_log = "XAR: TEST FAIL inherit_166600" }

	# Production quantizer boundaries: same tier, next tier, and storage cap.

	set_global_variable = { name = xa_run_score value = 104 }
	xar_quantize_record_candidate_effect = yes
	if = {

			global_var:xa_record_candidate = 100
			NOT = { global_var:xa_record_candidate > global_var:xa_old_record }
		}
		debug_log = "XAR: TEST PASS record_same_threshold"
	}
	else = { debug_log = "XAR: TEST FAIL record_same_threshold" }
	set_global_variable = { name = xa_run_score value = 105 }
	xar_quantize_record_candidate_effect = yes
	if = {
		limit = {
			global_var:xa_record_candidate = 105

		}

	}

	}
	else = { debug_log = "XAR: TEST FAIL pact_trait" }
	if = {
		limit = { has_character_flag = xa_enabled }
		debug_log = "XAR: TEST PASS pact_flag"
	}
	else = { debug_log = "XAR: TEST FAIL pact_flag" }
	if = {
		limit = { global_var:xa_price_dip = 25 }
		debug_log = "XAR: TEST PASS shop_init"
	}
	else = { debug_log = "XAR: TEST FAIL shop_init" }

			global_var:xa_contract_progress = 0
			global_var:xa_import_ready = 1

			has_character_flag = xa_bought_sixfold_apotheosis
		}
