xar_compute_and_commit_death_settlement_effect = {
	if = {
		limit = { has_character_flag = xa_settlement_committed }
		debug_log = "XAR: duplicate death settlement ignored"
	}
	else = {
		xar_compute_score_effect = yes
		xar_commit_death_settlement_effect = yes
	}
}

xar_commit_death_settlement_effect = {
	set_global_variable = { name = xa_settlement_ready value = 0 }
	set_global_variable = {
		name = xa_settlement_source_character
		value = scope:xar_dead
	}
	set_global_variable = {
		name = xa_settlement_final_score
		value = { value = global_var:xa_run_score }
	}
	set_global_variable = {
		name = xa_settlement_score_before_reject
		value = { value = global_var:xa_score_before_reject }
	}
	set_global_variable = {
		name = xa_settlement_record_candidate
		value = { value = global_var:xa_record_candidate }
	}
	set_global_variable = {
		name = xa_settlement_old_record
		value = { value = global_var:xa_old_record }
	}
	set_global_variable = {
		name = xa_settlement_record_delta
		value = { value = global_var:xa_score_delta }
	}
	set_global_variable = {
		name = xa_settlement_blessing_count
		value = { value = global_var:xa_bless_count }
	}
	set_global_variable = {
		name = xa_settlement_refusal_count
		value = { value = global_var:xa_bless_reject_count }
	}
	set_global_variable = {
		name = xa_settlement_contract_progress
		value = { value = global_var:xa_contract_progress }
	}
	set_global_variable = { name = xa_settlement_record_written value = 0 }
	if = {
		limit = { global_var:xa_record_candidate > global_var:xa_old_record }
		set_global_variable = { name = xa_settlement_record_written value = 1 }
	}
	# Rogue mode has exactly one playable life per save: a fresh commit is
	# always serial 1. Direct assignment also migrates already-signed saves
	# whose newly introduced global did not exist when they were created.
	set_global_variable = { name = xa_settlement_commit_serial value = 1 }
	scope:xar_dead = { add_character_flag = xa_settlement_committed }
	set_global_variable = { name = xa_settlement_ready value = 1 }
	# Publishing before the tutorial writer is intentional. In a real CK3 death
	# the writer can block for tens of seconds; native consumers use
	# record_written=1 to wait for tutorial.txt persistence without losing the
	# already complete terminal payload.
	if = {
		limit = { global_var:xa_record_candidate > global_var:xa_old_record }
		debug_log = "XAR: new record, writing bit"
		xar_write_record_effect = yes
	}
}
