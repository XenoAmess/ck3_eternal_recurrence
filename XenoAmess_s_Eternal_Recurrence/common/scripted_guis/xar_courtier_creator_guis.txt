xar_cc_toggle_trait_gui = {
	scope = character
	saved_scopes = { trait }
	is_shown = { xar_cc_ui_access_trigger = yes }
	is_valid = {
		xar_cc_ui_access_trigger = yes
		exists = scope:trait
		OR = {
			xar_cc_trait_is_selected_trigger = { TRAIT = scope:trait }
			AND = {
				scope:trait = { save_temporary_scope_as = xar_cc_candidate_trait }
				xar_cc_candidate_trait_metadata_valid_trigger = yes
				OR = {
					is_target_in_variable_list = {
						name = xar_cc_catalog_education
						target = scope:xar_cc_candidate_trait
					}
					AND = {
						is_target_in_variable_list = {
							name = xar_cc_catalog_commander
							target = scope:xar_cc_candidate_trait
						}
						OR = {
							AND = {
								var:xar_cc_commander_count < 2
								variable_list_size = {
									name = xar_cc_selected_commander
									value < 2
								}
							}
							xar_cc_candidate_replaces_selected_trait_trigger = yes
						}
					}
					is_target_in_variable_list = {
						name = xar_cc_catalog_physical
						target = scope:xar_cc_candidate_trait
					}
					AND = {
						is_target_in_variable_list = {
							name = xar_cc_catalog_personality
							target = scope:xar_cc_candidate_trait
						}
						OR = {
							AND = {
								var:xar_cc_personality_count < 3
								variable_list_size = {
									name = xar_cc_selected_personality
									value < 3
								}
							}
							xar_cc_candidate_replaces_selected_trait_trigger = yes
						}
					}
					is_target_in_variable_list = {
						name = xar_cc_catalog_other
						target = scope:xar_cc_candidate_trait
					}
				}
			}
		}
	}
	effect = {
		if = {
			limit = {
				xar_cc_ui_access_trigger = yes
				exists = scope:trait
			}
			scope:trait = { save_scope_as = xar_cc_candidate_trait }
			if = {
				limit = {
					xar_cc_trait_is_selected_trigger = {
						TRAIT = scope:xar_cc_candidate_trait
					}
				}
				xar_cc_remove_selected_trait_effect = {
					TRAIT = scope:xar_cc_candidate_trait
				}
			}
			else_if = {
				limit = { xar_cc_candidate_trait_metadata_valid_trigger = yes }
				xar_cc_remove_selected_trait_conflicts_effect = yes
				if = {
					limit = {
						is_target_in_variable_list = {
							name = xar_cc_catalog_education
							target = scope:xar_cc_candidate_trait
						}
					}
					clear_variable_list = xar_cc_selected_education
					add_to_variable_list = {
						name = xar_cc_selected_education
						target = scope:xar_cc_candidate_trait
					}
				}
				else_if = {
					limit = {
						is_target_in_variable_list = {
							name = xar_cc_catalog_commander
							target = scope:xar_cc_candidate_trait
						}
						var:xar_cc_commander_count < 2
						variable_list_size = {
							name = xar_cc_selected_commander
							value < 2
						}
					}
					add_to_variable_list = {
						name = xar_cc_selected_commander
						target = scope:xar_cc_candidate_trait
					}
					change_variable = { name = xar_cc_commander_count add = 1 }
				}
				else_if = {
					limit = {
						is_target_in_variable_list = {
							name = xar_cc_catalog_physical
							target = scope:xar_cc_candidate_trait
						}
					}
					add_to_variable_list = {
						name = xar_cc_selected_physical
						target = scope:xar_cc_candidate_trait
					}
				}
				else_if = {
					limit = {
						is_target_in_variable_list = {
							name = xar_cc_catalog_personality
							target = scope:xar_cc_candidate_trait
						}
						var:xar_cc_personality_count < 3
						variable_list_size = {
							name = xar_cc_selected_personality
							value < 3
						}
					}
					add_to_variable_list = {
						name = xar_cc_selected_personality
						target = scope:xar_cc_candidate_trait
					}
					change_variable = { name = xar_cc_personality_count add = 1 }
				}
				else_if = {
					limit = {
						is_target_in_variable_list = {
							name = xar_cc_catalog_other
							target = scope:xar_cc_candidate_trait
						}
					}
					add_to_variable_list = {
						name = xar_cc_selected_other
						target = scope:xar_cc_candidate_trait
					}
				}
			}
		}
	}
}
