# Player-rooted interfaces for the paid courtier creator.
xar_cc_window_gate_gui = {
	scope = character
	is_shown = { xar_cc_ui_access_trigger = yes }
}

xar_cc_cancel_gui = {
	scope = character
	is_shown = { xar_cc_ui_access_trigger = yes }
	is_valid = { xar_cc_ui_access_trigger = yes }
	effect = {
		if = {
			limit = { xar_cc_ui_access_trigger = yes }
			remove_character_flag = xar_cc_open
		}
	}
}

xar_cc_confirm_gui = {
	scope = character
	is_shown = { xar_cc_ui_access_trigger = yes }
	is_valid = {
		xar_cc_ui_access_trigger = yes
		custom_description = {
			text = xar.cc.invalid_configuration
			xar_cc_valid_configuration_trigger = yes
		}
		custom_description = {
			text = xar.cc.insufficient_gold
			gold >= xar_courtier_creator_cost
		}
	}
	effect = { xar_cc_complete_purchase_effect = yes }
}

xar_cc_toggle_cmd_organizer_gui = {
	scope = character
	is_shown = { xar_cc_ui_access_trigger = yes }
	is_valid = {
		xar_cc_ui_access_trigger = yes
		OR = { var:xar_cc_cmd_organizer = 1 xar_cc_commander_count < 2 }
	}
	effect = {
		if = {
			limit = { var:xar_cc_cmd_organizer = 1 }
			set_variable = { name = xar_cc_cmd_organizer value = 0 }
		}
		else = { set_variable = { name = xar_cc_cmd_organizer value = 1 } }
	}
}
