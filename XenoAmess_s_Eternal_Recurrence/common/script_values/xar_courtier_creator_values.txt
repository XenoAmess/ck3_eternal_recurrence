# Native ruler-designer absolute skill costs, adapted to character variables.
xar_cc_diplomacy_skill_cost = {
	value = 0
	if = {
		limit = {
			var:xar_cc_diplomacy > 0
			var:xar_cc_diplomacy < 5
		}
		add = 2
		multiply = var:xar_cc_diplomacy
	}
	else_if = {
		limit = {
			var:xar_cc_diplomacy > 4
			var:xar_cc_diplomacy < 9
		}
		add = 4
		multiply = {
			value = var:xar_cc_diplomacy
			subtract = 4
		}
		add = 8
	}
	else_if = {
		limit = {
			var:xar_cc_diplomacy > 8
			var:xar_cc_diplomacy < 13
		}
		add = 7
		multiply = {
			value = var:xar_cc_diplomacy
			subtract = 8
		}
		add = 24
	}
	else_if = {
		limit = {
			var:xar_cc_diplomacy > 12
			var:xar_cc_diplomacy < 17
		}
		add = 11
		multiply = {
			value = var:xar_cc_diplomacy
			subtract = 12
		}
		add = 52
	}
	else_if = {
		limit = { var:xar_cc_diplomacy > 16 }
		add = 17
		multiply = {
			value = var:xar_cc_diplomacy
			subtract = 16
		}
		add = 96
	}
}

xar_cc_arbitrary_age_cost = {
	value = 0
	if = {
		limit = { var:xar_cc_age <= 18 }
		add = 60
	}
	else_if = {
		limit = { var:xar_cc_age <= 30 }
		add = 60
		subtract = {
			value = var:xar_cc_age
			subtract = 18
			multiply = 2.5
		}
	}
	else_if = {
		limit = { var:xar_cc_age < 45 }
		add = 30
		subtract = {
			value = var:xar_cc_age
			subtract = 30
			multiply = 2
		}
	}
}
