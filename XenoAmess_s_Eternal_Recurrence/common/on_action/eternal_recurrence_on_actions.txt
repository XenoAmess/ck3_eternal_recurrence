# Keep this adapter filename neutral: vanilla errors attribute merged on_action
# callers to this file, while actual mod failures still expose xar_* symbols.
# Game start hook. We only add an on_actions entry so vanilla's events/effect stay intact.
# NOTE: on_game_start_after_lobby is used (not on_game_start) because that is the hook
# where players and game rules are known, per the vanilla comment in game_start.txt.
on_game_start_after_lobby = {
	on_actions = {
		# XAR_ACCEPTANCE_ONLY_BEGIN
		xar_acceptance_balance_fixture_on_game_start
		# XAR_ACCEPTANCE_ONLY_END
		xar_on_game_start
	}
}

# Request the global record import. The GUI bridge moves request -> ready, then
# the consumer copies points and starts the pact/selftest from ready state.
xar_on_game_start = {
	effect = {
		every_player = {
			# Existing pact saves predate the stable player-scope witness.
			if = {
				limit = { has_character_flag = xa_enabled }
				set_global_variable = xa_player_pact_active
				save_scope_as = xar_existing_pact_player
				set_global_variable = {
					name = xa_player_pact_character
					value = scope:xar_existing_pact_player
				}
			}
			if = {
				limit = {
					OR = {
						# XAR_ACCEPTANCE_ONLY_BEGIN
						has_game_rule = xar_selftest
						# XAR_ACCEPTANCE_ONLY_END
						has_game_rule = xar_on
					}
				}
				xar_initialize_challenge_mode_effect = yes
				set_global_variable = { name = xa_import_requested value = 1 }
				set_global_variable = { name = xa_import_ready value = 0 }
				set_global_variable = { name = xa_import_consumed value = 0 }
				debug_log = "XAR: import requested"
			}
		}
	}
}

# Death hook. Same non-destructive pattern.
on_death = {
	on_actions = {
		xar_on_death
		# XAR_ACCEPTANCE_ONLY_BEGIN
		xar_acceptance_observe_death
		# XAR_ACCEPTANCE_ONLY_END
	}
}

# Contract behavior hooks. Every adapter scopes to the acting character, then
# the shared effect enforces xa_enabled + is_ai=no before changing progress.
on_war_won_attacker = { on_actions = { xar_contract_war_attacker } }
on_war_won_defender = { on_actions = { xar_contract_war_defender } }
on_hook_used = { on_actions = { xar_contract_hook_used } }
on_county_faith_change = { on_actions = { xar_contract_faith_change } }
on_birth_mother = { on_actions = { xar_contract_birth_mother } }
on_birth_father = { on_actions = { xar_contract_birth_father } }
on_building_completed = { on_actions = { xar_contract_building_completed } }
on_birthday = { on_actions = { xar_contract_birthday } }

xar_contract_war_attacker = { effect = { scope:attacker = { xar_add_contract_progress_effect = { ID = 1 } } } }
xar_contract_war_defender = { effect = { scope:defender = { xar_add_contract_progress_effect = { ID = 1 } } } }
xar_contract_hook_used = { effect = { root = { xar_add_contract_progress_effect = { ID = 2 } } } }
xar_contract_faith_change = { effect = { root.holder = { xar_add_contract_progress_effect = { ID = 3 } } } }
xar_contract_birth_mother = { effect = { scope:mother = { xar_add_contract_progress_effect = { ID = 4 } } } }
xar_contract_birth_father = { effect = { scope:father = { xar_add_contract_progress_effect = { ID = 4 } } } }
xar_contract_building_completed = {
	effect = {
		if = {
			limit = { exists = scope:character }
			scope:character = { xar_add_contract_progress_effect = { ID = 5 } }
		}
		else = { root.county.holder = { xar_add_contract_progress_effect = { ID = 5 } } }
	}
}
xar_contract_birthday = {
	effect = {
		if = {
			limit = { root = { stress < 100 } }
			root = { xar_add_contract_progress_effect = { ID = 6 } }
		}
	}
}

# Queue a normal succession's delayed carrier while the dying root and its saved
# scope are still valid, then score before the carrier runs. No-heir deaths keep
# the synchronous event chain because there is no valid living recipient.
xar_on_death = {
	effect = {
		if = {
			limit = {
				is_ai = no
				OR = {
					has_character_flag = xa_enabled
					AND = {
						has_global_variable = xa_player_pact_active
						exists = global_var:xa_player_pact_character
						global_var:xa_player_pact_character = root
					}
				}
			}
			# XAR_ACCEPTANCE_ONLY_BEGIN
			if = {
				limit = {
					has_global_variable = xa_balance_active
					exists = global_var:xa_balance_fixture_character
					global_var:xa_balance_fixture_character = root
				}
				debug_log = "XAR: BALANCE production death gate entered"
			}
			# XAR_ACCEPTANCE_ONLY_END
			debug_log = "XAR: computing score on death"
			if = {
				limit = { exists = player_heir }
				save_scope_as = xar_dead
				player_heir = {
					save_scope_as = xar_death_carrier
					trigger_event = { id = xar.1003 delayed = yes }
				}
				# XAR_ACCEPTANCE_ONLY_BEGIN
				if = {
					limit = { has_global_variable = xa_death_with_heir_active }
					debug_log = "XAR: TEST death-with-heir carrier queued"
					debug_log = "XAR: TEST death-with-heir compute entered"
				}
				if = {
					limit = { has_global_variable = xa_balance_active }
					debug_log = "XAR: BALANCE production carrier queued"
				}
				# XAR_ACCEPTANCE_ONLY_END
				xar_compute_score_effect = yes
				# XAR_ACCEPTANCE_ONLY_BEGIN
				if = {
					limit = { has_global_variable = xa_balance_active }
					debug_log = "XAR: BALANCE production score computed inline"
				}
				# XAR_ACCEPTANCE_ONLY_END
			}
			else = { trigger_event = xar.1000 }
		}
	}
}
