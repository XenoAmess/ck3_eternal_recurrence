# Character scope. True when the character holds at least one surviving
# dynamic Later-Dynasty hegemony created by this mod.
rmtm_holds_restoration_hegemony_trigger = {
	any_held_title = {
		has_variable = rmtm_restoration_hegemony
	}
}

# Character scope. A Later title keeps its hegemony tier for identity and UI,
# but must not receive the vanilla Son-of-Heaven vassalization shortcuts.
rmtm_primary_title_is_restoration_hegemony_trigger = {
	primary_title = {
		has_variable = rmtm_restoration_hegemony
	}
}
