# Last-loaded extension of the vanilla ministry entitlement. The current Son of
# Heaven always wins; only while h_china is vacant may the one globally selected
# Later title continue the Three Departments and Six Ministries.
tgp_has_access_to_ministry_trigger = {
	government_has_flag = government_is_celestial
	OR = {
		has_title = title:h_china
		AND = {
			title:h_china = { NOT = { exists = holder } }
			exists = global_var:rmtm_ministry_entitlement_title
			primary_title = global_var:rmtm_ministry_entitlement_title
			global_var:rmtm_ministry_entitlement_title = {
				has_variable = rmtm_restoration_hegemony
			}
		}
	}
}
