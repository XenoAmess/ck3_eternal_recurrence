#########################################################
# Reclaim the Motherland - Phase 2 loyalty resolution
#########################################################

# Character scope: a direct, landed Pro-Hegemon vassal.
# Required saved scopes: old_emperor_temp and rmtm_loyalty_candidate.
rmtm_resolve_pro_hegemon_loyalty_effect = {
	if = {
		limit = { rmtm_pro_hegemon_hard_defect_trigger = yes }
		set_variable = { name = rmtm_loyalty_outcome value = flag:defect }
		set_variable = { name = rmtm_loyalty_reason value = flag:open_breach }
		add_to_list = rmtm_defecting_direct_vassals
	}
	else_if = {
		limit = { rmtm_pro_hegemon_hard_stay_trigger = yes }
		set_variable = { name = rmtm_loyalty_outcome value = flag:stay }
		set_variable = { name = rmtm_loyalty_reason value = flag:bound_by_grace }
		add_to_list = rmtm_final_loyal_direct_vassals
	}
	else = {
		set_variable = {
			name = rmtm_loyalty_score
			value = rmtm_pro_hegemon_loyalty_score_value
		}
		# The shattering effect cannot synchronously wait for several human clients.
		# A player-controlled vassal therefore receives the same frozen inputs but a
		# deterministic score boundary, never a hidden random choice on their behalf.
		if = {
			limit = { is_ai = no }
			if = {
				limit = { var:rmtm_loyalty_score >= 50 }
				set_variable = { name = rmtm_loyalty_outcome value = flag:stay }
				set_variable = { name = rmtm_loyalty_reason value = flag:weighed_allegiance }
				add_to_list = rmtm_final_loyal_direct_vassals
			}
			else = {
				set_variable = { name = rmtm_loyalty_outcome value = flag:defect }
				set_variable = { name = rmtm_loyalty_reason value = flag:weighed_ambition }
				add_to_list = rmtm_defecting_direct_vassals
			}
		}
		else = {
			random = {
				chance = var:rmtm_loyalty_score
				set_variable = { name = rmtm_loyalty_outcome value = flag:stay }
			}
			if = {
				limit = { var:rmtm_loyalty_outcome = flag:stay }
				set_variable = { name = rmtm_loyalty_reason value = flag:weighed_allegiance }
				add_to_list = rmtm_final_loyal_direct_vassals
			}
			else = {
				set_variable = { name = rmtm_loyalty_outcome value = flag:defect }
				set_variable = { name = rmtm_loyalty_reason value = flag:weighed_ambition }
				add_to_list = rmtm_defecting_direct_vassals
			}
		}
	}
}
