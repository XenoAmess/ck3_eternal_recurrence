cmake_minimum_required(VERSION 3.24)
project(maintenance_writer_cold_a01 LANGUAGES CXX ASM_MASM)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(maintenance_writer_cold_target cold_target.cpp synthetic_write.asm)
add_executable(maintenance_writer_cold_observer cold_observer.cpp)
foreach(target maintenance_writer_cold_target maintenance_writer_cold_observer)
 target_compile_options(${target} PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/EHsc> $<$<COMPILE_LANGUAGE:CXX>:/W4>)
 target_link_options(${target} PRIVATE /INCREMENTAL:NO)
endforeach()
