```python
"""Documentation recipe: fresh two-mod native fixture profile, without launch."""
from __future__ import annotations
import argparse
import ast
from datetime import datetime, timezone
import hashlib
import json
from pathlib import Path
import re
import shutil

EXACT_EXE_SHA = 'ae1ba6ff060ba603842f6f4a2ded0af4b7d3666b3dd271f75fb01b0da8e81b2d'
PRODUCT_IDS = {'remove-mandala': 'mod_remove_mandala',
               'tributary-expansion-directives': 'mod_tributary_expansion_directives',
               'auto-upgrade-buildings': 'mod_auto_upgrade_buildings',
               'celestial-commerce-corruption': 'mod_celestial_commerce_corruption',
               'zhongguo-style': 'mod_zhongguo_style', 'ox-here': 'ox_here'}

def digest(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()

def pure_function(path, name, context):
    parsed = ast.parse(path.read_text(encoding='utf-8-sig'))
    functions = [node for node in parsed.body if isinstance(node, ast.FunctionDef) and node.name == name]
    if len(functions) != 1:
        raise ValueError('public source function changed: ' + name)
    exec(compile(ast.fix_missing_locations(ast.Module(body=functions, type_ignores=[])), str(path), 'exec'), context)
    return context[name]

def inventory(root):
    return {path.relative_to(root).as_posix(): digest(path) for path in sorted(root.rglob('*')) if path.is_file()}

def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--repo', type=Path, required=True)
    parser.add_argument('--game-dir', type=Path, required=True)
    parser.add_argument('--product', choices=sorted(PRODUCT_IDS), required=True)
    parser.add_argument('--production', type=Path, required=True)
    parser.add_argument('--manifest', type=Path, required=True)
    parser.add_argument('--fixture', type=Path, required=True)
    parser.add_argument('--state-dir', type=Path, required=True)
    args = parser.parse_args()
    repo, game, production, manifest, fixture, state = [path.resolve() for path in
        (args.repo, args.game_dir, args.production, args.manifest, args.fixture, args.state_dir)]
    if state.exists() or state == repo or repo in state.parents:
        raise ValueError('new external state directory required')
    if state in production.parents or state in fixture.parents:
        raise ValueError('frozen inputs cannot reside inside the new state directory')
    executable = game / 'binaries/ck3.exe'
    if digest(executable) != EXACT_EXE_SHA:
        raise ValueError('exact CK3 1.20.0.2 executable required before any profile write')
    receipt_path = fixture.with_suffix('.prepare.json')
    receipt = json.loads(receipt_path.read_text(encoding='utf-8-sig'))
    if receipt.get('exe_sha256') != EXACT_EXE_SHA or receipt.get('game_version') != '1.20.0.2':
        raise ValueError('use the current public exact-EXE fixture preparer')
    if receipt.get('product') != PRODUCT_IDS[args.product]:
        raise ValueError('fixture product identity mismatch')
    fixture_files = inventory(fixture)
    if fixture_files != receipt['prepared_file_sha256']:
        raise ValueError('prepared fixture bytes changed')
    if not manifest.is_file() or not (production / 'descriptor.mod').is_file():
        raise ValueError('verified official staging and matching manifest required')
    projection = json.loads(manifest.read_text(encoding='utf-8-sig'))
    if projection.get('format_version') != 1 or projection.get('product_id') != PRODUCT_IDS[args.product]:
        raise ValueError('standalone-product manifest identity/format mismatch')
    manifest_files = {row['path']: row['sha256'] for row in projection['files']}
    if len(manifest_files) != len(projection['files']) or inventory(production) != manifest_files:
        raise ValueError('production bytes do not match the frozen manifest')
    descriptors = {}
    for key, root in [('product', production), ('fixture', fixture)]:
        descriptors[key] = (root / 'descriptor.mod').read_text(encoding='utf-8-sig')
        if re.search(r'(?m)^\s*(remote_file_id|path|archive)\s*=', descriptors[key]):
            raise ValueError('inner descriptor contains publication/path identity')
    rules = game / 'game/common/game_rules/00_game_rules.txt'
    defaults = pure_function(repo / 'tools/run_acceptance.py', 'declared_vanilla_rule_defaults',
                             {'re': re, 'VANILLA_GAME_RULES': rules, 'RunnerError': ValueError})(rules)
    if len(defaults) != 85:
        raise ValueError('review the exact vanilla defaults instead of copying old preset tokens')
    settings = [setting for _, setting in defaults]
    if args.product == 'remove-mandala':
        settings.append('mrm_enabled')
    if args.product == 'zhongguo-style':
        settings.extend(['zg361_on', 'zg361_freq_yearly', 'zg361_ratio_strict'])
    if len(set(settings)) != len(settings):
        raise ValueError('duplicate game-rule settings')
    settings_text = pure_function(repo / 'tools/run_terminal_acceptance.py', 'render_settings', {})()
    profile = state / 'profile'
    for relative in ['mod', 'mod-content', 'logs', 'save games', 'player/game_rules', 'run']:
        (profile / relative).mkdir(parents=True, exist_ok=True)
    for key, source in [('product', production), ('fixture', fixture)]:
        target = profile / 'mod-content' / key
        shutil.copytree(source, target)
        if inventory(source) != inventory(target):
            raise ValueError('mounted source copy differs')
        (profile / 'mod' / (key + '.mod')).write_text(
            descriptors[key].rstrip() + '\npath="' + target.as_posix() + '"\n', encoding='utf-8-sig', newline='\n')
    shutil.copyfile(manifest, profile / 'mod-content/product.manifest.json')
    (profile / 'dlc_load.json').write_text(json.dumps({'enabled_mods': ['mod/product.mod', 'mod/fixture.mod'],
                                                    'disabled_dlcs': []}) + '\n', encoding='utf-8')
    (profile / 'player/game_rules/presets.txt').write_text(
        'game_rules_preset={\n\tname="LastAppliedRules"\n\tsetting={ ' + ' '.join(settings) +
        ' }\n\tironman=no\n}\n', encoding='utf-8', newline='\n')
    (profile / 'tutorial.txt').write_text('last_lesson_chain="reactive_advice"\ncompleted_lessons={\n}\n', encoding='utf-8')
    (profile / 'pdx_settings.txt').write_text(settings_text, encoding='utf-8', newline='\n')
    report = {'schema': 'ck3.public-documentation.fresh-two-mod-profile.v1',
              'prepared_at_utc': datetime.now(timezone.utc).isoformat(), 'product': args.product,
              'game_version': '1.20.0.2', 'game_executable': str(executable), 'exe_sha256': digest(executable),
              'production': str(production), 'manifest': str(manifest), 'manifest_sha256': digest(manifest),
              'fixture_receipt': str(receipt_path), 'fixture_receipt_sha256': digest(receipt_path),
              'state_dir': str(state), 'profile_dir': str(profile), 'files_sha256': inventory(profile),
              'vanilla_rule_defaults': defaults, 'selected_settings': settings,
              'runtime_status': 'NOT_RUN', 'game_launched': False, 'native_injected': False,
              'seed_or_checkpoint_imported': False, 'required_harness_option': '--fixture-profile'}
    (state / 'preparation.json').write_text(json.dumps(report, indent=2) + '\n', encoding='utf-8')
    print(json.dumps({'state_dir': str(state), 'runtime_status': 'NOT_RUN'}))

if __name__ == '__main__':
    main()
```
