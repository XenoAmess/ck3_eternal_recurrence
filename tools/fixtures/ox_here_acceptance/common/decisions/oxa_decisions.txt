oxa_initialize_decision = {
	picture = { reference = "gfx/interface/illustrations/decisions/decision_invite_knights.dds" }
	decision_group_type = oxa_acceptance
	sort_order = 1
	desc = oxa_initialize_decision_desc
	selection_tooltip = oxa_initialize_decision_tooltip
	is_shown = {
		is_ai = no
		NOT = { has_character_flag = oxa_initialized }
	}
	is_valid = { is_ai = no }
	effect = {
		add_character_flag = oxa_initialize_pending
	}
	ai_check_interval = 0
	ai_potential = { always = no }
}
