oxa_initialize_gui = {
	scope = character
	is_shown = {
		is_ai = no
		has_character_flag = oxa_initialize_pending
		NOT = { has_character_flag = oxa_initialized }
	}
	effect = { oxa_initialize_effect = yes }
}
