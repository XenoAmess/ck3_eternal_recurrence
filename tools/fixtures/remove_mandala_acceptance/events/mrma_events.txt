namespace = mrma

mrma.1 = {
	scope = none
	hidden = yes
	immediate = {
		if = {
			limit = {
				global_var:mrma_player_ruler = {
					has_government = wanua_government
				}
			}
			debug_log = "MRMA: TEST PASS day_one_ruler_wanua"
		}
		else = {
			debug_log = "MRMA: TEST FAIL day_one_ruler_wanua"
		}
	}
}

mrma.2 = {
	scope = none
	hidden = yes
	immediate = {
		if = {
			limit = {
				global_var:mrma_holding_province = {
					has_holding_type = castle_holding
				}
			}
			debug_log = "MRMA: TEST PASS day_two_holding_castle"
		}
		else = {
			debug_log = "MRMA: TEST FAIL day_two_holding_castle"
		}
		random_ruler = {
			limit = {
				is_ai = yes
				is_independent_ruler = yes
				government_has_flag = government_is_feudal
			}
			set_global_variable = {
				name = mrma_ai_ruler
				value = this
			}
			change_government = mandala_government
		}
		trigger_event = { id = mrma.5 days = 1 }
	}
}

mrma.5 = {
	scope = none
	hidden = yes
	immediate = {
		if = {
			limit = {
				global_var:mrma_ai_ruler = {
					has_government = wanua_government
				}
			}
			debug_log = "MRMA: TEST PASS ai_conversion_redirected"
		}
		else = {
			debug_log = "MRMA: TEST FAIL ai_conversion_redirected"
		}
		global_var:mrma_player_ruler = {
			change_government = mandala_government
		}
		global_var:mrma_holding_province = {
			set_holding_type = temple_citadel_holding
		}
		# Exercise the exact recurring dispatchers without spending an entire
		# wall-clock year in the live harness. Static validation separately binds
		# them to months=1 and yearly_global_pulse in production.
		trigger_event = { on_action = mrm_monthly_dispatch days = 1 }
		trigger_event = { id = mrma.3 days = 2 }
		trigger_event = { on_action = mrm_yearly_dispatch days = 3 }
		trigger_event = { id = mrma.4 days = 4 }
	}
}
