			set_variable = { name = zga_direct_governor_n value = 0 }

				remove_variable = zg361_prev_merit_level

			limit = { scope:zga_song_emperor = { var:zga_direct_governor_n >= 3 } }
			debug_log = "ZGA: TEST PASS song_direct_governors_at_least_three"
		}
		else = { debug_log = "ZGA: TEST FAIL song_direct_governors_at_least_three" }

		# Prefer a non-independent celestial liege with its own reviewable cohort.
		# Falling back to any eligible non-independent liege still proves that
		# independence is not part of the production entry predicate.
		scope:zga_song_emperor = {
			random_vassal = {
				limit = {
					is_independent_ruler = no
					zg361_is_celestial_liege_trigger = yes
					any_vassal = {
						zg361_is_reviewable_vassal_trigger = yes
						count >= 3
					}
				}
				save_scope_as = zga_non_independent_liege
				add_character_flag = zga_ai_review_candidate
			}
		}
		if = {
			limit = { NOT = { exists = scope:zga_non_independent_liege } }
			scope:zga_song_emperor = {
				random_vassal = {
					limit = {
						is_independent_ruler = no
						zg361_is_celestial_liege_trigger = yes
					}
					save_scope_as = zga_non_independent_liege
				}
			}
		}
		if = {
			limit = {
				exists = scope:zga_non_independent_liege
				scope:zga_non_independent_liege = {
					is_independent_ruler = no
					highest_held_title_tier >= tier_duchy
					zg361_is_celestial_liege_trigger = yes
				}
			}
			debug_log = "ZGA: TEST PASS non_independent_celestial_liege_entry"
		}
		else = { debug_log = "ZGA: TEST FAIL non_independent_celestial_liege_entry" }

		if = {
			limit = {
				exists = scope:zga_non_independent_liege
				scope:zga_non_independent_liege = {
					has_character_flag = zga_ai_review_candidate
				}
			}
			scope:zga_non_independent_liege = {
				debug_log = "ZGA: TEST INFO ai_non_independent_review_scheduled"
				trigger_event = { id = zga_acceptance.1 days = 0 }
			}
		}
		else = {
			debug_log = "ZGA: TEST INFO ai_non_independent_review_candidate_unavailable"
		}

		scope:zga_song_emperor = {
			add_character_flag = zga_initialized
			add_character_flag = zga_prepare_player_review
			set_variable = { name = zg361_ratio_override value = 10 }
		}
		set_player_character = scope:zga_song_emperor
		debug_log = "ZGA: TEST PASS switched_to_song_emperor"
	}
}

# Integrated receipt probe: exercise the real 3.25 and appeal effects against
# one disposable AI official before the main first-cycle review resets results.
zga_verify_penalty_and_appeal_effect = {
	random_vassal = {
		limit = {
			is_ai = yes
			zg361_is_reviewable_vassal_trigger = yes
			government_has_flag = government_has_treasury
			government_has_flag = government_has_merit
		}
		save_scope_as = zga_receipt_probe
	}
	if = {
		limit = { exists = scope:zga_receipt_probe }
		set_variable = { name = zg361_review_serial value = 7001 }
		set_variable = { name = zg361_last_325_n value = 0 }
		set_variable = { name = zg361_last_35_n value = 0 }
		scope:zga_receipt_probe = {
			set_variable = { name = zg361_streak_bottom value = 0 }
			add_gold = 500
			add_treasury = 500
			change_merit = { value = 500 }
			set_variable = { name = zga_gold_before_penalty value = gold }
			set_variable = { name = zga_treasury_before_penalty value = treasury }
			set_variable = { name = zga_merit_before_penalty value = merit }
			set_variable = {
				name = zga_gold_after_penalty
				value = { value = var:zga_gold_before_penalty subtract = zg361_perf_gold_penalty_value }
			}
			set_variable = {
				name = zga_treasury_after_penalty
				value = { value = var:zga_treasury_before_penalty subtract = zg361_perf_treasury_penalty_value }
			}
			set_variable = {
				name = zga_merit_after_penalty
				value = { value = var:zga_merit_before_penalty subtract = medium_merit_value }
			}
			zg361_grade_325_apply_effect = yes
			if = {
				limit = {
					gold = var:zga_gold_after_penalty
					treasury = var:zga_treasury_after_penalty
					merit = var:zga_merit_after_penalty
					has_character_modifier = zg361_grade_325
					has_character_modifier = zg361_pip
					has_variable = zg361_last_gold_penalty_paid
					var:zg361_last_gold_penalty_paid = zg361_perf_gold_penalty_value
					has_variable = zg361_last_treasury_penalty_paid
					var:zg361_last_treasury_penalty_paid = zg361_perf_treasury_penalty_value
					has_variable = zg361_last_merit_penalty_paid
					var:zg361_last_merit_penalty_paid = medium_merit_value
				}
				debug_log = "ZGA: TEST PASS grade_325_three_account_penalty"
			}
			else = { debug_log = "ZGA: TEST FAIL grade_325_three_account_penalty" }

			zg361_appeal_regrade_to_35_effect = yes
			set_variable = {
				name = zga_merit_after_appeal
				value = { value = var:zga_merit_before_penalty add = minor_merit_value }
			}
			if = {
				limit = {
					gold = var:zga_gold_before_penalty
					treasury = var:zga_treasury_before_penalty
					merit = var:zga_merit_after_appeal
					has_character_modifier = zg361_grade_35
					NOT = { has_character_modifier = zg361_grade_325 }
					NOT = { has_character_modifier = zg361_pip }
					NOT = { has_variable = zg361_last_gold_penalty_paid }
					NOT = { has_variable = zg361_last_treasury_penalty_paid }
					NOT = { has_variable = zg361_last_merit_penalty_paid }
					NOT = { has_variable = zg361_last_penalty_reviewer }
					NOT = { has_variable = zg361_last_penalty_serial }
				}
				debug_log = "ZGA: TEST PASS appeal_exact_fixed_refund_and_salary_stop"
			}
			else = { debug_log = "ZGA: TEST FAIL appeal_exact_fixed_refund_and_salary_stop" }

			# A duplicate call must not refund or grant the 3.5 merit award twice.
			zg361_appeal_regrade_to_35_effect = yes

				limit = {
					gold = var:zga_gold_before_penalty
					treasury = var:zga_treasury_before_penalty

}

zga_prepare_player_review_effect = {
	# A scripted-GUI state can queue the same on_start more than once in one frame.
	# Re-check and advance the phase inside the observer effect so duplicates are no-ops.
	if = {
		limit = {
			has_character_flag = zga_prepare_player_review
			NOT = { has_character_flag = zga_waiting_for_scoreboard }
		}
		remove_character_flag = zga_prepare_player_review
		add_character_flag = zga_waiting_for_scoreboard
	if = {
		limit = {
			has_title = title:h_china
			highest_held_title_tier >= tier_duchy
			government_has_flag = government_is_celestial
			zg361_is_celestial_liege_trigger = yes
		}
		debug_log = "ZGA: TEST PASS player_song_review_entry"
	}
	else = { debug_log = "ZGA: TEST FAIL player_song_review_entry" }
	zga_verify_penalty_and_appeal_effect = yes

	# One product-owned turnkey effect resolves all 361 cards. The generated
	# fixture only verifies each resulting variable and the aggregate ledgers.
	zga_verify_361_mechanism_batch_effect = yes

	# First real product invocation: every direct official is a newcomer. The
	# product must still rank the entire cohort and queue calibration, while
	# protecting every newcomer from 3.25 and writing next-cycle snapshots.
	zg361_run_review_effect = yes
	set_variable = { name = zga_missing_snapshot_n value = 0 }
	set_variable = { name = zga_reviewable_n value = 0 }
	set_variable = { name = zga_newcomer_bottom_n value = 0 }
	set_variable = { name = zga_pending_grade_n value = 0 }
	every_vassal = {
		limit = { zg361_is_reviewable_vassal_trigger = yes }
		root = { change_variable = { name = zga_reviewable_n add = 1 } }
		if = {
			limit = { NOT = { has_variable = zg361_prev_merit_level } }
			root = { change_variable = { name = zga_missing_snapshot_n add = 1 } }
		}
		if = {
			limit = { has_variable = zg361_pending_grade }
			root = { change_variable = { name = zga_pending_grade_n add = 1 } }
		}
		if = {
			limit = {
				has_character_flag = zga_was_newcomer
				has_variable = zg361_pending_grade
				var:zg361_pending_grade = 1
			}
			root = { change_variable = { name = zga_newcomer_bottom_n add = 1 } }
		}
	}
	set_variable = { name = zga_pending_grade_sum value = var:zg361_pending_375_n }
	change_variable = { name = zga_pending_grade_sum add = var:zg361_pending_35_n }
	change_variable = { name = zga_pending_grade_sum add = var:zg361_pending_325_n }
	if = {
		limit = {
			var:zga_reviewable_n >= 3
			var:zga_missing_snapshot_n = 0
			var:zg361_cohort_n = var:zga_reviewable_n
			var:zga_pending_grade_n = var:zga_reviewable_n
			var:zga_pending_grade_sum = var:zga_reviewable_n
			var:zga_newcomer_bottom_n = 0
			var:zg361_pending_325_n = 0
			NOT = { has_variable = zg361_sb_m_01_char }
		}
		debug_log = "ZGA: TEST PASS newcomer_snapshot_prepared_by_product"
		debug_log = "ZGA: TEST PASS newcomer_first_review_protected"
		debug_log = "ZGA: TEST PASS player_calibration_pending"
	}
	else = {
		debug_log = "ZGA: TEST FAIL newcomer_snapshot_prepared_by_product"
		debug_log = "ZGA: TEST FAIL newcomer_first_review_protected"
		debug_log = "ZGA: TEST FAIL player_calibration_pending"
	}
	remove_variable = zga_missing_snapshot_n
	remove_variable = zga_newcomer_bottom_n
	remove_variable = zga_pending_grade_n
	remove_variable = zga_pending_grade_sum

	# A duplicate entry while calibration is pending must be a complete no-op.
	set_variable = { name = zga_review_serial_before_duplicate value = var:zg361_review_serial }
	zg361_run_review_effect = yes
	if = {
		limit = {
			var:zg361_review_serial = var:zga_review_serial_before_duplicate
			NOT = { has_variable = zg361_sb_m_01_char }
		}
		debug_log = "ZGA: TEST PASS pending_review_idempotent"
	}
	else = { debug_log = "ZGA: TEST FAIL pending_review_idempotent" }
	remove_variable = zga_review_serial_before_duplicate
	}
}

# After the manager-board screenshot, switch to one actually reviewed direct
# official and make the former player (now AI) run the real review effect again.
# The production grade effect must recognize the new human player and schedule
# the appropriate zg361.2/.3/.4 personal-result event itself.
zga_personal_result_effect = {
	if = {
		limit = {
			has_character_flag = zga_personal_result_pending
			has_character_flag = zga_player_review_verified
			has_variable = zg361_sb_m_01_char
		}
		remove_character_flag = zga_personal_result_pending
		save_scope_as = zga_reviewing_superior
		random_vassal = {
			limit = {
				is_alive = yes
				is_ai = yes
				zg361_is_current_liege_review_record_trigger = yes
			}
			save_scope_as = zga_personal_result_target
		}
		if = {
			limit = { exists = scope:zga_personal_result_target }
			scope:zga_personal_result_target = {
				add_character_flag = zga_personal_result_target
				# Make this real official refuse a mandate issued by this exact
				# superior. Extra negative evidence ensures the later real review
				# reaches the 3.25 branch without assigning a grade in the fixture.
				set_variable = { name = zg361_slacking value = 1 }
				set_variable = { name = zg361_slandered value = 1 }
				set_variable = { name = zg361_slander_backfire value = 1 }
				set_variable = { name = zg361_jingcha_pending value = 1 }
				set_variable = { name = zg361_jingcha_mandate_year value = current_year }
				set_variable = { name = zg361_jingcha_mandate_superior value = liege }
				set_variable = { name = zg361_jingcha_mandate_reviewer value = liege }
				zg361_refuse_jingcha_effect = yes
				set_variable = { name = zga_kpi_with_refusal value = zg361_kpi_value }
				remove_variable = zg361_skipped_jingcha_superior
				remove_variable = zg361_skipped_jingcha_year
				set_variable = { name = zga_kpi_without_refusal value = zg361_kpi_value }
				set_variable = {
					name = zga_expected_refusal_kpi
					value = {
						value = var:zga_kpi_without_refusal
						subtract = zg361_skipped_jingcha_kpi_malus_value
					}
				}
				set_variable = { name = zg361_skipped_jingcha_superior value = liege }
				set_variable = { name = zg361_skipped_jingcha_year value = current_year }
				if = {
					limit = {
						var:zga_kpi_with_refusal = var:zga_expected_refusal_kpi
						NOT = { has_variable = zg361_jingcha_pending }
						NOT = { has_variable = zg361_jingcha_mandate_superior }
						NOT = { has_variable = zg361_jingcha_mandate_year }
						liege = {
							has_opinion_modifier = {
								target = prev
								modifier = zg361_refused_jingcha
							}
						}
					}
					debug_log = "ZGA: TEST PASS jingcha_refusal_superior_opinion_and_kpi_minus_50"
				}
				else = { debug_log = "ZGA: TEST FAIL jingcha_refusal_superior_opinion_and_kpi_minus_50" }
			}
			set_player_character = scope:zga_personal_result_target
			scope:zga_reviewing_superior = {
				trigger_event = { id = zga_acceptance.2 days = 0 }
			}
		}
		else = { debug_log = "ZGA: TEST FAIL personal_result_target_unavailable" }
	}
}
