# Only fixture-owned variables change. No production flag/resource/queue writes.
aubgu_fail_effect = {
    set_variable = { name = aubgu_phase value = -1 }
    trigger_event = { id = aubgu.900 }
}

# Official current_date arithmetic is in DAYS (stock00_custom_loc_effects).
aubgu_read_elapsed_effect = {
    set_variable = {
        name = aubgu_elapsed_days
        value = { value = current_date subtract = var:aubgu_initial_date }
    }
}

aubgu_observe_first_enabled_effect = {
    aubgu_read_elapsed_effect = yes
    if = {
        limit = {
            var:aubgu_phase = 1
            aubgu_same_day_resources_trigger = yes
            aubgu_default_enabled_trigger = yes
        }
        debug_log = "AUBGUI: OBSERVE first_enabled_default_policy_same_day_resources"
        set_variable = { name = aubgu_phase value = 2 }
        trigger_event = { id = aubgu.810 }
    }
    else = {
        debug_log = "AUBGUI: TEST FAIL first_enabled_actor_date_resources_or_flags"
        aubgu_fail_effect = yes
    }
}
