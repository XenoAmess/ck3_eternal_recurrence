# Acceptance-only player-manager seed generator. Never mounted by release.
# The empty-scope load hook runs before queued visible character events. The
# after-lobby hook remains as the new-game equivalent; both call one
# idempotent effect, and the load-safe scripted GUI is the final fallback.
on_game_start = {
	on_actions = {
		zga_phase2_manager_seed_on_game_start
	}
}

on_game_start_after_lobby = {
	on_actions = {
		zga_phase2_manager_seed_on_game_start
	}
}

zga_phase2_manager_seed_on_game_start = {
	effect = {
		every_player = {
			limit = {
				is_ai = no
				is_alive = yes
				is_landed = yes
				has_game_rule = zg361_on
				OR = {
					AND = {
						zg361_is_celestial_liege_trigger = yes
						zg361_review_now_business_valid_trigger = yes
						prestige >= 150
						NOT = { has_character_flag = zga_phase2_manager_seed_bootstrap_started }
						NOT = { has_character_flag = zga_phase2_manager_seed_handoff_pending }
					}
					AND = {
						zg361_is_reviewable_vassal_trigger = yes
						NOT = { has_character_flag = zga_phase2_manager_seed_handoff_started }
						liege = {
							is_ai = yes
							zg361_is_celestial_liege_trigger = yes
							zg361_review_now_business_valid_trigger = yes
							prestige >= 150
							NOT = { has_character_flag = zg361_b1_cycle_active }
							NOT = { has_character_flag = zg361_review_in_progress }
							trigger_if = {
								limit = { has_variable = zg361_p2c_active }
								var:zg361_p2c_active != 1
							}
							trigger_else = { always = yes }
							trigger_if = {
								limit = { has_variable = zg361_pp_portfolio_queue_active }
								var:zg361_pp_portfolio_queue_active != 1
							}
							trigger_else = { always = yes }
						}
					}
				}
			}
			zga_phase2_manager_seed_maybe_begin_effect = yes
		}
	}
}
