# Root is the live human player. Eligibility is checked again immediately
# before selecting one already-existing direct reviewable vassal.
zga_phase2_manager_seed.100 = {
	type = character_event
	hidden = yes

	immediate = {
		if = {
			limit = {
				is_ai = no
				is_alive = yes
				is_landed = yes
				zg361_is_celestial_liege_trigger = yes
				has_game_rule = zg361_on
				zg361_review_now_business_valid_trigger = yes
				prestige >= 150
				NOT = { has_character_flag = zg361_b1_cycle_active }
				NOT = { has_character_flag = zg361_review_in_progress }
				trigger_if = {
					limit = { has_variable = zg361_p2c_active }
					var:zg361_p2c_active != 1
				}
				trigger_else = { always = yes }
				trigger_if = {
					limit = { has_variable = zg361_pp_portfolio_queue_active }
					var:zg361_pp_portfolio_queue_active != 1
				}
				trigger_else = { always = yes }
				any_vassal = {
					count >= 1
					zg361_is_reviewable_vassal_trigger = yes
				}
			}
			add_character_flag = zga_phase2_manager_seed_bootstrap_started
			save_scope_as = zga_phase2_manager_owner
			if = {
				limit = {
					exists = scope:zga_phase2_manager_subject
					scope:zga_phase2_manager_subject = {
						is_ai = yes
						is_alive = yes
						is_landed = yes
						zg361_is_reviewable_vassal_trigger = yes
						liege = root
					}
				}
				scope:zga_phase2_manager_subject = {
					save_scope_as = zga_phase2_manager_subject
				}
			}
			else = {
				ordered_vassal = {
					limit = {
						zg361_is_reviewable_vassal_trigger = yes
						liege = root
					}
					order_by = age
					position = 0
					save_scope_as = zga_phase2_manager_subject
				}
			}
			if = {
				limit = { exists = scope:zga_phase2_manager_subject }
				trigger_event = zga_phase2_manager_seed.1
			}
			else = {
				debug_log = "ZGAP2MANAGERSEED: RED existing direct reviewable vassal unavailable"
			}
		}
		else = {
			debug_log = "ZGAP2MANAGERSEED: RED player-manager seed preconditions unavailable"
		}
	}
}

# Hidden post-switch carrier. The load-safe scripted GUI switches control
# immediately after the exact pending B2 PIP makes the subject reviewable and
# before any later destructive event can invalidate it. A zero-day scheduled
# event avoids relying on same-effect
# player-state recomputation: root must now be the human manager, while the
# former player must be the same existing direct AI vassal.
zga_phase2_manager_seed.11 = {
	type = character_event
	hidden = yes
	immediate = {
		if = {
			limit = {
				is_ai = no
				is_alive = yes
				is_landed = yes
				this = scope:zga_phase2_manager_owner
				has_character_flag = zga_phase2_manager_seed_handoff_pending
				zg361_is_celestial_liege_trigger = yes
				has_game_rule = zg361_on
				zg361_review_now_business_valid_trigger = yes
				prestige >= 150
				NOT = { has_character_flag = zg361_b1_cycle_active }
				NOT = { has_character_flag = zg361_review_in_progress }
				trigger_if = {
					limit = { has_variable = zg361_p2c_active }
					var:zg361_p2c_active != 1
				}
				trigger_else = { always = yes }
				trigger_if = {
					limit = { has_variable = zg361_pp_portfolio_queue_active }
					var:zg361_pp_portfolio_queue_active != 1
				}
				trigger_else = { always = yes }
				exists = scope:zga_phase2_manager_subject
				scope:zga_phase2_manager_subject = {
					is_ai = yes
					is_alive = yes
					is_landed = yes
					zg361_is_reviewable_vassal_trigger = yes
					liege = root
				}
			}
			remove_character_flag = zga_phase2_manager_seed_handoff_pending
			debug_log = "ZGAP2MANAGERSEED: post-switch manager binding verified"
			trigger_event = zga_phase2_manager_seed.100
		}
		else = {
			zga_phase2_manager_seed_log_post_switch_gates_effect = yes
			debug_log = "ZGAP2MANAGERSEED: RED post-switch manager binding unavailable"
		}
	}
}
