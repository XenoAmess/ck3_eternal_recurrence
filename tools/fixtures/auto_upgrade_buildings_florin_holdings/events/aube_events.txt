    immediate = {

        if = {

                        limit = { holding_district = { has_building = tribe_02 } scope:aub_payer = { gold = var:aube_expected_gold prestige = var:aube_expected_prestige piety = var:aube_piety_before treasury = var:aube_treasury_before var:aub_upgrades_this_scan = 1 } }

                }
                else = { debug_log = "AUBE: TEST FAIL tribal_source_owner_government_or_native_target" }

            set_variable = { name = aube_expected_gold value = { value = gold subtract = regular_building_cost_tier_2 } }

            if = {
                limit = {
                    var:aube_direct_count > 0 is_ai = no is_alive = yes has_government = nomad_government
                    NOT = { any_directly_owned_province = { OR = { NOT = { has_holding_type = herder_holding } NOT = { holding_district = { has_building = herder_camp_01 } } NOT = { province_owner = { this = scope:aub_payer } } barony = { is_leased_out = yes } has_ongoing_holding_construction = yes } } }

                    limit = { has_variable = aube_nomad_zero_write var:aube_herder_count = var:aube_direct_count gold = var:aube_gold_before prestige = var:aube_prestige_before piety = var:aube_piety_before treasury = var:aube_treasury_before var:aub_upgrades_this_scan = 0 NOT = { any_directly_owned_province = { OR = { NOT = { has_holding_type = herder_holding } NOT = { holding_district = { has_building = herder_camp_01 } } } } } }

                }

        }
