zqn_setup_courtiers_effect = {

  name = "ZQN Natural Accept Courtier"
  gender = male age = 30 random_traits = no
  location = scope:actor.capital_province culture = scope:actor.culture
  faith = faith:catholic trait = cynical trait = fickle
  save_scope_as = zqn_high
 }
 create_character = {
  name = "ZQN Natural Refuse Courtier"
  gender = male age = 30 random_traits = no
  location = scope:actor.capital_province culture = scope:actor.culture
  faith = faith:catholic trait = zealous trait = stubborn
  save_scope_as = zqn_low
 }
 add_courtier = scope:zqn_high
 add_courtier = scope:zqn_low
 scope:zqn_high = {
  set_character_rite = rite:roman_rite
  add_opinion = { target = scope:actor modifier = love_opinion opinion = 200 }
 }
 scope:zqn_low = {
  set_character_rite = rite:roman_rite
  add_opinion = { target = scope:actor modifier = hate_opinion opinion = -200 }
 }
 set_variable = { name = zqn_high value = scope:zqn_high }
 set_variable = { name = zqn_low value = scope:zqn_low }

 set_variable = { name = xqol_mass_conversion_threshold_draft value = 50 }
 if = {
  limit = {
   NOT = { has_variable = xqol_mass_conversion_pending }

   NOT = { any_vassal_or_below = { has_character_flag = xqol_conversion_ruler_candidate } }

   is_character_interaction_valid = { recipient = var:zqn_low interaction = xqol_mass_conversion_courtier_interaction }
   var:zqn_high = { is_ai = yes is_alive = yes is_landed = no faith = faith:catholic rite = rite:roman_rite }
   var:zqn_low = { is_ai = yes is_alive = yes is_landed = no faith = faith:catholic rite = rite:roman_rite }
  }
  debug_log = "ZQN120: TEST READY ordinary_courtiers_valid"
  zqn_seed_async_queue_effect = yes

}
