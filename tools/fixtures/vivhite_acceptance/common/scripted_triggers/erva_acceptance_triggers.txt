	var:ervc_cc_female = 0
	var:ervc_cc_age = 30
	var:ervc_cc_diplomacy = 6
	var:ervc_cc_martial = 6
	var:ervc_cc_stewardship = 6
	var:ervc_cc_intrigue = 6
	var:ervc_cc_learning = 6
	var:ervc_cc_prowess = 6
	var:ervc_cc_same_house = 0
	variable_list_size = { name = ervc_cc_selected_education value = 1 }
	is_target_in_variable_list = {
		name = ervc_cc_selected_education
		target = trait:education_martial_3
	}
	ervc_courtier_creator_cost >= 120
	ervc_courtier_creator_cost < 121
}

	is_ai = yes
	is_alive = yes
	is_female = no
	age >= 30
	age < 31
	is_courtier_of = root
	culture = root.culture
	faith = root.faith
	NOT = { exists = dynasty }
	has_trait = education_martial_3
	diplomacy >= 6
	martial >= 6
	stewardship >= 6
	intrigue >= 6
	learning >= 6
	prowess >= 6
	has_character_flag = blocked_from_leaving
}
