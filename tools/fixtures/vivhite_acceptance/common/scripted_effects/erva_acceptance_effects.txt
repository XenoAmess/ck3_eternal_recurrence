# Runtime acceptance observes production state from a separate fixture mod.
erva_acceptance_zero_gold_effect = {
	set_variable = { name = erva_acceptance_gold_reset value = gold }
	if = {
		limit = { var:erva_acceptance_gold_reset > 0 }
		remove_short_term_gold = var:erva_acceptance_gold_reset
	}
	else_if = {
		limit = { var:erva_acceptance_gold_reset < 0 }
		add_gold = { value = var:erva_acceptance_gold_reset multiply = -1 }
	}
	remove_variable = erva_acceptance_gold_reset
}

erva_acceptance_count_untracked_courtiers_effect = {
	set_variable = { name = erva_acceptance_untracked_count value = 0 }
	every_courtier = {
		limit = { NOT = { has_character_flag = erva_acceptance_baseline_courtier } }
		root = {
			change_variable = { name = erva_acceptance_untracked_count add = 1 }
		}
	}
}

erva_acceptance_initialize_effect = {
	if = {
		limit = {
			is_ai = no
			is_alive = yes
			has_character_flag = erva_acceptance_start_pending
			NOT = { has_character_flag = erva_acceptance_initialized }
		}
		remove_character_flag = erva_acceptance_start_pending
		add_character_flag = erva_acceptance_initialized
		every_courtier = { add_character_flag = erva_acceptance_baseline_courtier }
		erva_acceptance_zero_gold_effect = yes
		add_gold = 1000
		if = {
			limit = { has_game_rule = erva_acceptance_standalone }
			add_character_flag = erva_phase_standalone_cancel
			debug_log = "ERVA: TEST BEGIN standalone"
		}
		# ERVA_DUAL_ONLY_BEGIN
		else_if = {
			limit = { has_game_rule = erva_acceptance_dual }
			set_global_variable = { name = xa_contract_id value = 0 }
			add_character_flag = xa_enabled
			add_character_flag = erva_phase_dual_ervc_stage
			debug_log = "ERVA: TEST BEGIN dual"
		}
		# ERVA_DUAL_ONLY_END
		else = { debug_log = "ERVA: TEST FAIL unsupported_mode" }
		if = {
			limit = { exists = character:1132 }
			character:1132 = { trigger_event = erva.0001 }
		}
		else = { debug_log = "ERVA: TEST FAIL ai_fixture_missing" }
	}
}
