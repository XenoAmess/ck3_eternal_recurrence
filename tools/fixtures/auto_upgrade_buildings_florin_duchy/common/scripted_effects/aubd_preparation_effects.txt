# Character scope; call again at every event boundary.
aubd_bind_preparation_scopes_effect = {
    save_scope_as = aub_payer
    save_scope_as = character
    province:2619 = {
        save_scope_as = aubd_target_province
        province_owner = { save_scope_as = holder }
        province_owner = { save_scope_as = build_owner }
        county = { save_scope_as = county }
    }
}

# Province scope. Find the actual built primary's own district. Never assume
# a holding/province scope is a duchy district, or invent a district getter.
aubd_locate_built_military_district_effect = {
    if = {
        limit = { exists = scope:aubd_military_district }
        clear_saved_scope = aubd_military_district
    }
    scope:aub_payer = { set_variable = { name = aubd_primary_matches value = 0 } }
    every_built_buildings = {
        limit = {
            is_primary_building = yes
            building_district = { has_building = military_duchy_building_01 }
        }
        building_district = { save_scope_as = aubd_military_district }
        scope:aub_payer = { change_variable = { name = aubd_primary_matches add = 1 } }
    }
}

# Finite read-only day1 observations. Stock GUI, not a script construction
# effect, supplies the first district. These do not attest the GUI selection.
aubd_observe_gui_construction_effect = {
    if = { limit = { var:aubd_phase = 1 } debug_log = "AUBD: PRECOND phase1_yes" }
    else = { debug_log = "AUBD: PRECOND phase1_no" }
    if = { limit = { aubd_actual_duchy_owner_trigger = yes } debug_log = "AUBD: PRECOND actual_owner_yes" }
    else = { debug_log = "AUBD: PRECOND actual_owner_no" }
    if = { limit = { has_character_modifier = aubd_fast_district_preparation } debug_log = "AUBD: PRECOND preparation_modifier_yes" }
    else = { debug_log = "AUBD: PRECOND preparation_modifier_no" }
    if = { limit = { has_character_flag = enable_auto_build } debug_log = "AUBD: PRECOND auto_enabled_unexpected" }
    else = { debug_log = "AUBD: PRECOND auto_disabled_yes" }
    if = { limit = { gold >= 100 } debug_log = "AUBD: PRECOND product_gold_at_least100_yes" }
    else = { debug_log = "AUBD: PRECOND product_gold_at_least100_no" }
    province:2619 = {
        if = { limit = { aubd_preparation_inputs_trigger = yes } debug_log = "AUBD: PRECOND province_inputs_yes" }
        else = { debug_log = "AUBD: PRECOND province_inputs_no" }
        if = { limit = { can_have_duchy_capital_building_trigger = yes } debug_log = "AUBD: PRECOND duchy_capital_yes" }
        else = { debug_log = "AUBD: PRECOND duchy_capital_no" }
        culture = {
            if = { limit = { has_innovation = innovation_burhs } debug_log = "AUBD: PRECOND burhs_yes" }
            else = { debug_log = "AUBD: PRECOND burhs_no" }
            if = { limit = { has_innovation = innovation_castle_baileys } debug_log = "AUBD: PRECOND castle_baileys_yes" }
            else = { debug_log = "AUBD: PRECOND castle_baileys_no" }
        }
        if = { limit = { has_ongoing_holding_construction = yes } debug_log = "AUBD: PRECOND province_construction_active" }
        else = { debug_log = "AUBD: PRECOND province_construction_absent" }
        if = { limit = { has_building_or_higher = military_duchy_building_01 } debug_log = "AUBD: PRECOND primary01_already_present" }
        else = { debug_log = "AUBD: PRECOND primary01_absent" }
        holding_district = {
            if = { limit = { has_building_or_higher = castle_02 } debug_log = "AUBD: PRECOND holdingII_yes" }
            else = { debug_log = "AUBD: PRECOND holdingII_no" }
        }
    }
}
