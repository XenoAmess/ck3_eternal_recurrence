namespace = aubt

aubt.1 = {
	scope = none
	hidden = yes
	immediate = {
		global_var:aubt_province = {
			if = {
				limit = { has_building = barracks_01 }
				debug_log = "AUBT: OBSERVED target_building barracks_01"
			}
			else = { debug_log = "AUBT: OBSERVED target_building not_barracks_01" }
			if = {
				limit = { has_building = barracks_02 }
				debug_log = "AUBT: OBSERVED target_upgrade barracks_02"
			}
			else = { debug_log = "AUBT: OBSERVED target_upgrade not_barracks_02" }
			if = {
				limit = { has_ongoing_construction = no }
				debug_log = "AUBT: OBSERVED target_construction none"
			}
			else = { debug_log = "AUBT: OBSERVED target_construction ongoing" }
			if = {
				limit = {
					OR = {
						has_holding_type = castle_holding
						has_holding_type = city_holding
						has_holding_type = church_holding
					}
				}
				debug_log = "AUBT: OBSERVED target_holding eligible"
			}
			else = { debug_log = "AUBT: OBSERVED target_holding ineligible" }
			if = {
				limit = { barony = { is_leased_out = no } }
				debug_log = "AUBT: OBSERVED target_lease none"
			}
			else = { debug_log = "AUBT: OBSERVED target_lease leased" }
		}
		global_var:aubt_player = {
			if = {
				limit = { has_treasury = yes }
				debug_log = "AUBT: OBSERVED payer_has_treasury yes"
			}
			else = { debug_log = "AUBT: OBSERVED payer_has_treasury no" }
			if = {
				limit = { culture = { has_innovation = innovation_barracks } }
				debug_log = "AUBT: OBSERVED payer_innovation barracks"
			}
			else = { debug_log = "AUBT: OBSERVED payer_innovation missing_barracks" }
			if = {
				limit = { treasury < var:aubt_treasury_before }
				debug_log = "AUBT: OBSERVED payer_treasury decreased"
			}
			else = { debug_log = "AUBT: OBSERVED payer_treasury not_decreased" }
			if = {
				limit = { gold = var:aubt_gold_before }
				debug_log = "AUBT: OBSERVED payer_gold unchanged"
			}
			else = { debug_log = "AUBT: OBSERVED payer_gold changed"
			}
			if = {
				limit = { any_directly_owned_province = { has_building = barracks_02 } }
				debug_log = "AUBT: OBSERVED direct_iterator sees_barracks_02"
			}
			else = { debug_log = "AUBT: OBSERVED direct_iterator misses_barracks_02" }
		}
		if = {
			limit = { has_global_variable = aub_auto_build_loop_started }
			debug_log = "AUBT: OBSERVED global_loop active"
		}
		else = { debug_log = "AUBT: OBSERVED global_loop inactive" }
		if = {
			limit = {
				global_var:aubt_province = {
					has_building = barracks_02
					NOT = { has_building_or_higher = barracks_03 }
				}
				global_var:aubt_player = {
					has_treasury = yes
					treasury < var:aubt_treasury_before
				}
				has_global_variable = aub_auto_build_loop_started
			}
			debug_log = "AUBT: TEST PASS treasury_priority_one_tier"
		}
		else = { debug_log = "AUBT: TEST FAIL treasury_priority_one_tier" }
		global_var:aubt_player = {
			remove_character_flag = enable_auto_build
		}
		trigger_event = { id = aubt.2 days = 16 }
	}
}

aubt.2 = {
	scope = none
	hidden = yes
	immediate = {
		if = {
			limit = {
				global_var:aubt_province = { has_building = barracks_02 }
				NOT = { has_global_variable = aub_auto_build_loop_started }
			}
			debug_log = "AUBT: TEST PASS disabled_zero_side_effect"
		}
		else = { debug_log = "AUBT: TEST FAIL disabled_zero_side_effect" }
		global_var:aubt_player = {
			# Empty the live treasury while it is still available, then return to the
			# runner's original resource model for the explicit gold fallback.
			set_variable = { name = aubt_reset_treasury value = treasury }
			if = { limit = { var:aubt_reset_treasury > 0 } remove_treasury = var:aubt_reset_treasury }
			change_government = feudal_government
			set_variable = { name = aubt_reset_gold value = gold }
			if = { limit = { var:aubt_reset_gold > 0 } remove_short_term_gold = var:aubt_reset_gold }
			add_gold = 100000
			set_variable = { name = aubt_gold_before value = gold }
			capital_province = {
				remove_building = barracks_02
				add_building = barracks_01
			}
			add_character_flag = enable_auto_build
			aub_start_global_loop_effect = yes
		}
		trigger_event = { id = aubt.3 days = 3 }
	}
}

# Phase 2 representative matrix. All mutations live in the disposable -userdir
# fixture and run serially in one CK3 process.
aubt.10 = {
	type = character_event
	hidden = yes
	immediate = {
		save_scope_as = aub_payer
		save_scope_as = character
		add_gold = 100000
		add_prestige = 100000
		add_piety = 100000
		culture = {
			if = {
				limit = { NOT = { has_innovation = innovation_battlements } }
				add_innovation = innovation_battlements
			}
			if = {
				limit = { NOT = { has_innovation = innovation_castle_baileys } }
				add_innovation = innovation_castle_baileys
			}
			if = {
				limit = { NOT = { has_innovation = innovation_plenary_assemblies } }
				add_innovation = innovation_plenary_assemblies
			}
		}
		capital_province = { set_global_variable = { name = aubt_province value = this } }

		# Castle main building. The starting capital is already a castle; do not
		# reapply its holding type because CK3 rejects redundant conversions.
		global_var:aubt_province = {
			if = { limit = { has_building = castle_04 } remove_building = castle_04 }
			else_if = { limit = { has_building = castle_03 } remove_building = castle_03 }
			else_if = { limit = { has_building = castle_02 } remove_building = castle_02 }
			else_if = { limit = { has_building = castle_01 } remove_building = castle_01 }
			add_building = castle_01
			scope:aub_payer = { save_scope_as = holder }
			scope:aub_payer = { save_scope_as = build_owner }
			county = { save_scope_as = county }
			aub_upgrade_chain_castle_01_effect = yes
			if = {
				limit = { has_building = castle_02 NOT = { has_building_or_higher = castle_03 } }
				debug_log = "AUBT: TEST PASS main_castle"
			}
			else = { debug_log = "AUBT: TEST FAIL main_castle" }
		}

		# Duchy-capital building while the capital ownership and holding are still
		# untouched. Robert holds both the county and its de-jure duchy.
		global_var:aubt_province = {
			if = { limit = { has_building = archery_ranges_03 } remove_building = archery_ranges_03 }
			else_if = { limit = { has_building = archery_ranges_02 } remove_building = archery_ranges_02 }
			else_if = { limit = { has_building = archery_ranges_01 } remove_building = archery_ranges_01 }
			add_building = archery_ranges_01
			scope:aub_payer = { save_scope_as = holder }
			scope:aub_payer = { save_scope_as = build_owner }
			county = { save_scope_as = county }
			aub_upgrade_chain_archery_ranges_01_effect = yes
			if = {
				limit = { has_building = archery_ranges_02 }
				debug_log = "AUBT: TEST PASS duchy_capital"
			}
			else = { debug_log = "AUBT: TEST FAIL duchy_capital" }
		}

		# Exact negative gate: Apulia's plains terrain cannot upgrade outposts.
		# The source is materialized first, then all three player resources must
		# remain unchanged when the generated vanilla target gate rejects it.
		global_var:aubt_province = {
			# The core matrix has already verified barracks_01 -> 02; release that
			# regular slot before installing the deliberately terrain-invalid source.
			if = { limit = { has_building = barracks_02 } remove_building = barracks_02 }
			else_if = { limit = { has_building = barracks_01 } remove_building = barracks_01 }
			if = { limit = { has_building = outposts_08 } remove_building = outposts_08 }
			else_if = { limit = { has_building = outposts_07 } remove_building = outposts_07 }
			else_if = { limit = { has_building = outposts_06 } remove_building = outposts_06 }
			else_if = { limit = { has_building = outposts_05 } remove_building = outposts_05 }
			else_if = { limit = { has_building = outposts_04 } remove_building = outposts_04 }
			else_if = { limit = { has_building = outposts_03 } remove_building = outposts_03 }
			else_if = { limit = { has_building = outposts_02 } remove_building = outposts_02 }
			else_if = { limit = { has_building = outposts_01 } remove_building = outposts_01 }
			add_building = outposts_01
			scope:aub_payer = { save_scope_as = holder }
			scope:aub_payer = { save_scope_as = build_owner }
			county = { save_scope_as = county }
		}
		set_variable = { name = aubt_gold_before_gate value = gold }
		set_variable = { name = aubt_prestige_before_gate value = prestige }
		set_variable = { name = aubt_piety_before_gate value = piety }
		global_var:aubt_province = { aub_upgrade_chain_outposts_01_effect = yes }
		if = {
			limit = {
				global_var:aubt_province = {
					has_building = outposts_01
					NOT = { has_building = outposts_02 }
					NOT = { aub_can_upgrade_to_outposts_02_trigger = yes }
				}
				gold = var:aubt_gold_before_gate
				prestige = var:aubt_prestige_before_gate
				piety = var:aubt_piety_before_gate
			}
			debug_log = "AUBT: TEST PASS vanilla_gate_rejection"
		}
		else = { debug_log = "AUBT: TEST FAIL vanilla_gate_rejection" }

		# Tribal main building and its gold+prestige cost route.
		# Vanilla's building_requirement_tribal checks holder government, not only
		# holding type, so the disposable payer must be tribal for this sample.
		change_government = tribal_government
		global_var:aubt_province = {
			set_holding_type = tribal_holding
			if = { limit = { has_building = tribe_02 } remove_building = tribe_02 }
			else_if = { limit = { has_building = tribe_01 } remove_building = tribe_01 }
			add_building = tribe_01
			scope:aub_payer = { save_scope_as = holder }
			scope:aub_payer = { save_scope_as = build_owner }
			county = { save_scope_as = county }
		}
		set_variable = { name = aubt_prestige_before value = prestige }
		global_var:aubt_province = { aub_upgrade_chain_tribe_01_effect = yes }
		if = {
			limit = {
				global_var:aubt_province = { has_building = tribe_02 }
				prestige < var:aubt_prestige_before
			}
			debug_log = "AUBT: TEST PASS main_tribal_mixed_cost"
		}
		else = { debug_log = "AUBT: TEST FAIL main_tribal_mixed_cost" }

		# Ordinary tribal building.
		global_var:aubt_province = {
			if = { limit = { has_building = longhouses_02 } remove_building = longhouses_02 }
			else_if = { limit = { has_building = longhouses_01 } remove_building = longhouses_01 }
			add_building = longhouses_01
			scope:aub_payer = { save_scope_as = holder }
			scope:aub_payer = { save_scope_as = build_owner }
			county = { save_scope_as = county }
			aub_upgrade_chain_longhouses_01_effect = yes
			if = {
				limit = { has_building = longhouses_02 }
				debug_log = "AUBT: TEST PASS regular_tribal"
			}
			else = { debug_log = "AUBT: TEST FAIL regular_tribal" }
		}
		change_government = feudal_government

		# Temple-citadel main building. Supplying scope:character verifies the
		# vanilla on_complete payer context that immediate add_building needs.
		global_var:aubt_province = {
			set_holding_type = temple_citadel_holding
			if = { limit = { has_building = temple_citadel_04 } remove_building = temple_citadel_04 }
			else_if = { limit = { has_building = temple_citadel_03 } remove_building = temple_citadel_03 }
			else_if = { limit = { has_building = temple_citadel_02 } remove_building = temple_citadel_02 }
			else_if = { limit = { has_building = temple_citadel_01 } remove_building = temple_citadel_01 }
			add_building = temple_citadel_01
			scope:aub_payer = { save_scope_as = holder }
			scope:aub_payer = { save_scope_as = build_owner }
			county = { save_scope_as = county }
			aub_upgrade_chain_temple_citadel_01_effect = yes
		}
		if = {
			limit = { global_var:aubt_province = { has_building = temple_citadel_02 } }
			debug_log = "AUBT: TEST PASS main_temple_citadel"
		}
		else = { debug_log = "AUBT: TEST FAIL main_temple_citadel" }

		# Temple-citadel unique ordinary building.
		global_var:aubt_province = {
			if = { limit = { has_building = citadel_shrine_02 } remove_building = citadel_shrine_02 }
			else_if = { limit = { has_building = citadel_shrine_01 } remove_building = citadel_shrine_01 }
			add_building = citadel_shrine_01
			scope:aub_payer = { save_scope_as = holder }
			scope:aub_payer = { save_scope_as = build_owner }
			county = { save_scope_as = county }
			aub_upgrade_chain_citadel_shrine_01_effect = yes
			if = {
				limit = { has_building = citadel_shrine_02 }
				debug_log = "AUBT: TEST PASS temple_citadel_unique"
			}
			else = { debug_log = "AUBT: TEST FAIL temple_citadel_unique" }
		}

		# Scripted costs: one prestige route and one piety route.
		global_var:aubt_province = {
			if = { limit = { has_building = oath_poor_building_02 } remove_building = oath_poor_building_02 }
			else_if = { limit = { has_building = oath_poor_building_01 } remove_building = oath_poor_building_01 }
			add_building = oath_poor_building_01
			scope:aub_payer = { save_scope_as = holder }
			scope:aub_payer = { save_scope_as = build_owner }
			county = { save_scope_as = county }
			scope:aub_payer = { set_variable = { name = aubt_prestige_before value = prestige } }
			scope:aub_payer = { set_variable = { name = aubt_piety_before value = piety } }
			aub_upgrade_chain_oath_poor_building_01_effect = yes
			if = { limit = { has_building = oath_faithful_building_02 } remove_building = oath_faithful_building_02 }
			else_if = { limit = { has_building = oath_faithful_building_01 } remove_building = oath_faithful_building_01 }
			add_building = oath_faithful_building_01
			aub_upgrade_chain_oath_faithful_building_01_effect = yes
		}
		if = {
			limit = {
				global_var:aubt_province = {
					has_building = oath_poor_building_02
					has_building = oath_faithful_building_02
				}
				prestige < var:aubt_prestige_before
				piety < var:aubt_piety_before
			}
			debug_log = "AUBT: TEST PASS scripted_cost_resources"
		}
		else = { debug_log = "AUBT: TEST FAIL scripted_cost_resources" }

		# Real map special-building slot: Zhouzhi (province 10148) starts with the
		# Chang'an market in 1066. This avoids inventing a special slot in Apulia.
		set_variable = { name = aubt_gold_before_special value = gold }
		province:10148 = {
			if = {
				limit = { has_building = changan_market_02 }
				remove_building = changan_market_02
				scope:aub_payer = { save_scope_as = character }
				add_special_building = changan_market_01
			}
			else_if = {
				limit = { NOT = { has_building = changan_market_01 } }
				scope:aub_payer = { save_scope_as = character }
				add_special_building = changan_market_01
			}
			province_owner = { save_scope_as = holder }
			province_owner = { save_scope_as = build_owner }
			county = { save_scope_as = county }
			aub_upgrade_chain_changan_market_01_effect = yes
		}
		if = {
			limit = {
				province:10148 = { has_building = changan_market_02 }
				gold < var:aubt_gold_before_special
			}
			debug_log = "AUBT: TEST PASS special_gold"
		}
		else = { debug_log = "AUBT: TEST FAIL special_gold" }

		# City main building after ownership-sensitive castle/duchy checks.
		global_var:aubt_province = {
			set_holding_type = city_holding
			if = { limit = { has_building = city_04 } remove_building = city_04 }
			else_if = { limit = { has_building = city_03 } remove_building = city_03 }
			else_if = { limit = { has_building = city_02 } remove_building = city_02 }
			else_if = { limit = { has_building = city_01 } remove_building = city_01 }
			add_building = city_01
			scope:aub_payer = { save_scope_as = holder }
			scope:aub_payer = { save_scope_as = build_owner }
			county = { save_scope_as = county }
			aub_upgrade_chain_city_01_effect = yes
			if = {
				limit = { has_building = city_02 NOT = { has_building_or_higher = city_03 } }
				debug_log = "AUBT: TEST PASS main_city"
			}
			else = { debug_log = "AUBT: TEST FAIL main_city" }
		}

		# Church main building is last among governed holdings because changing a
		# capital to a leased religious holding can alter its barony owner.
		global_var:aubt_province = {
			set_holding_type = church_holding
			if = { limit = { has_building = temple_04 } remove_building = temple_04 }
			else_if = { limit = { has_building = temple_03 } remove_building = temple_03 }
			else_if = { limit = { has_building = temple_02 } remove_building = temple_02 }
			else_if = { limit = { has_building = temple_01 } remove_building = temple_01 }
			add_building = temple_01
			scope:aub_payer = { save_scope_as = holder }
			scope:aub_payer = { save_scope_as = build_owner }
			county = { save_scope_as = county }
			aub_upgrade_chain_temple_01_effect = yes
			if = {
				limit = { has_building = temple_02 NOT = { has_building_or_higher = temple_03 } }
				debug_log = "AUBT: TEST PASS main_church"
			}
			else = { debug_log = "AUBT: TEST FAIL main_church" }
		}

		# Nomad and herder holdings have no applicable province-building edge.
		set_variable = { name = aubt_gold_before_na value = gold }
		global_var:aubt_province = {
			set_holding_type = nomad_holding
			aub_upgrade_all_supported_buildings_effect = yes
			set_holding_type = herder_holding
			aub_upgrade_all_supported_buildings_effect = yes
		}
		if = {
			limit = {
				global_var:aubt_province = { has_holding_type = herder_holding }
				gold = var:aubt_gold_before_na
			}
			debug_log = "AUBT: TEST PASS nomad_herder_na"
		}
		else = { debug_log = "AUBT: TEST FAIL nomad_herder_na" }

		debug_log = "AUBT: TEST DONE source-live"
		trigger_event = aubt.900
	}
}
