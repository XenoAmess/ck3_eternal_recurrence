on_game_start_after_lobby = {
	on_actions = {
		aubt_start
	}
}

aubt_start = {
	effect = {
		random_player = {
			# Defer all mutations until one game day has elapsed so the runner can
			# inspect the real, still-available production decision while paused.
			trigger_event = { id = aubt.0 days = 1 }
		}
	}
}
