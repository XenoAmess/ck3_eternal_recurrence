				}
			}
			debug_log = "RQA: TEST PASS pre_chaos_movement_identity_prepared"
		}
		else = { debug_log = "RQA: TEST FAIL pre_chaos_movement_identity_prepared" }
		# Populate the full vanilla outgoing-phase contract for every current
		# participant in the emperor's realm, including direct vassals which are not
		# part of the two focused assertions below.

			every_situation_participant = {
				limit = { top_liege = root }
				top_participant_group:dynastic_cycle = {
					save_scope_as = rqa_current_movement
				}
				set_variable = {
					name = former_movement_member
					value = scope:rqa_current_movement
				}
			}
		}
		situation:dynastic_cycle = {
			situation_top_sub_region = {
				change_phase = { phase = situation_dynastic_cycle_phase_chaos }
			}
		}
		if = {
			limit = {
				situation:dynastic_cycle = {
					situation_current_phase = situation_dynastic_cycle_phase_chaos
				}
			}
			debug_log = "RQA: TEST PASS chaos_phase_transition_applied"
			# Organic instability-to-Chaos transitions create these variables in the

				set_variable = {
					name = former_movement_member
					value = scope:rqa_loyal_movement
				}
			}
			var:rqa_defecting_direct = {
				set_variable = {
					name = former_movement_member
					value = scope:rqa_loyal_movement
				}
			}
			var:rqa_control_ruler = {
				set_variable = {
					name = former_movement_member
					value = scope:rqa_control_movement
				}
			}
			if = {
				limit = {
					exists = var:rqa_loyal_direct
					var:rqa_loyal_direct.var:former_movement_member = {
						participant_group_type = pro_hegemon_movement

					exists = var:rqa_defecting_direct

rqa_prepare_threshold_effect = {
	if = {
		limit = { has_character_flag = rqa_threshold_pending }
	remove_character_flag = rqa_threshold_pending
	remove_character_flag = rqa_ready_for_threshold
	if = {
		limit = {
			exists = var:rqa_control_ruler
			var:rqa_control_ruler = { is_independent_ruler = yes }
			rmtm_holds_restoration_hegemony_trigger = yes
		}
		title:h_china = {
			every_de_jure_county = {
				limit = { NOT = { holder = root.var:rqa_control_ruler } }
				save_scope_as = rqa_county_to_transfer
				root = { rqa_transfer_county_to_control_effect = yes }
			}
		}
		if = {
			limit = {
				NOT = { rqa_controls_half_of_china_trigger = yes }
				NOT = { rqa_controls_claim_mandate_share_trigger = yes }
			}
			debug_log = "RQA: TEST PASS china_control_reset"
		}
		else = { debug_log = "RQA: TEST FAIL china_control_reset" }
		while = {
			limit = { NOT = { rqa_controls_half_of_china_trigger = yes } }
			title:h_china = {
				ordered_de_jure_county = {
					limit = { NOT = { holder.top_liege = root } }
					order_by = development_level
					save_scope_as = rqa_county_to_transfer
					root = { rqa_transfer_county_to_root_effect = yes }
				}
			}
		}
		if = {
			limit = {
				rqa_controls_half_of_china_trigger = yes
				NOT = { rqa_controls_claim_mandate_share_trigger = yes }
			}
			debug_log = "RQA: TEST PASS fifty_percent_below_claim_threshold"
		}
		else = { debug_log = "RQA: TEST FAIL fifty_percent_below_claim_threshold" }
		while = {
			limit = { NOT = { rqa_controls_claim_mandate_share_trigger = yes } }
			title:h_china = {
				ordered_de_jure_county = {
					limit = { NOT = { holder.top_liege = root } }
					order_by = development_level
					save_scope_as = rqa_county_to_transfer
					root = { rqa_transfer_county_to_root_effect = yes }
				}
			}
		}
		if = {
			limit = { rqa_controls_claim_mandate_share_trigger = yes }
			debug_log = "RQA: TEST PASS original_fifty_one_percent_threshold_reached"
			add_character_flag = rqa_waiting_for_restoration
			debug_log = "RQA: TEST PASS restoration_decision_ready"
		}
		else = { debug_log = "RQA: TEST FAIL original_fifty_one_percent_threshold_reached" }
	}
	else = { debug_log = "RQA: TEST FAIL threshold_preconditions" }
	}
}
