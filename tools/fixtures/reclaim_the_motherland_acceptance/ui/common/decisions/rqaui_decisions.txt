rqaui_verify_actual_restoration_decision = {
    picture = { reference = "gfx/interface/illustrations/decisions/decision_realm.dds" }
    decision_group_type = rqa_acceptance
    sort_order = 4
    ai_check_interval = 0
    desc = rqaui_verify_actual_restoration_decision_desc
    selection_tooltip = rqaui_verify_actual_restoration_decision_tooltip
    is_shown = { is_ai = no has_character_flag = rqaui_fiftyone_ready NOT = { has_character_flag = rqaui_finished } }
    is_valid = { has_title = title:h_china title:h_china = { holder = root } has_character_flag = claimed_the_mandate_of_heaven NOT = { rmtm_holds_restoration_hegemony_trigger = yes } }
    effect = {
        if = {
            limit = {
                has_title = title:h_china
                title:h_china = { holder = root }
                has_character_flag = claimed_the_mandate_of_heaven
                NOT = { rmtm_holds_restoration_hegemony_trigger = yes }
                NOT = { exists = var:rqa_later_title.holder }
            }
            debug_log = "RQAUI: TEST PASS actual_production_restoration_effect_observed"
            debug_log = "RQAUI: TEST DONE threshold_ui"
            add_character_flag = rqaui_finished
        }
        else = { debug_log = "RQAUI: TEST FAIL actual_production_restoration_effect_observed" }
    }
    ai_potential = { always = no }
}
