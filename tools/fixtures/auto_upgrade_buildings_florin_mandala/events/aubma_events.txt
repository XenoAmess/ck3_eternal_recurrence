# This partial scene is fixture preparation, never product behavior or GUI credit.
aubma.1 = {
    type = character_event
    hidden = yes
    immediate = {
        debug_log = "AUBMA: TEST BEGIN mandala_matrix"
        if = {
            limit = {
                this = character:1128
                is_ai = no
                character:burman002 = { is_alive = yes is_adult = yes is_landed = no }
                title:c_pagan = { holder = character:burman001 }
            }
            remove_character_flag = enable_auto_build
            remove_global_variable = aub_auto_build_loop_started
            character:burman002 = { save_scope_as = aubma_recipient }
            # Stock granted title transaction, then switch from the old player.
            create_title_and_vassal_change = {
                type = granted
                save_scope_as = aubma_title_change
            }
            title:c_pagan = {
                change_title_holder = {
                    holder = scope:aubma_recipient
                    change = scope:aubma_title_change
                }
            }
            resolve_title_and_vassal_change = scope:aubma_title_change
            set_player_character = scope:aubma_recipient
            scope:aubma_recipient = {
                remove_character_flag = enable_auto_build
                remove_character_flag = aub_funding_treasury_only
                remove_character_flag = aub_pause_when_over_domain_limit
                add_character_flag = aub_funding_personal_only
                # All four target costs are gold. Resources precede Native
                # target queries and get a day/event boundary to settle.
                add_gold = 100000
                change_government = mandala_government
                # R13/R15 actually iterated Pagan9550 and Popa9551. Keep the
                # latter explicitly owned, but exclude its holding through the
                # same native herder preparation exercised by R14. A granted
                # transaction ACK alone never proves the resulting scan domain.
                create_title_and_vassal_change = {
                    type = granted
                    save_scope_as = aubma_popa_owned_change
                }
                title:b_popa = {
                    change_title_holder = {
                        holder = scope:aubma_recipient
                        change = scope:aubma_popa_owned_change
                    }
                }
                resolve_title_and_vassal_change = scope:aubma_popa_owned_change
                province:9551 = { set_holding_type = herder_holding }
                # Initialize the prepared quota input; 0004 must still reset it.
                set_variable = { name = aub_upgrades_this_scan value = 0 }
                debug_log = "AUBMA: OBSERVE immediate_owned_domain_begin"
                aubma_read_pagan_inventory_effect = yes
                aubma_observe_scan_domain_effect = yes
                trigger_event = { id = aubma.2 days = 1 }
            }
        }
        else = {
            debug_log = "AUBMA: TEST FAIL original_actor_or_recipient"
            random_player = { trigger_event = aubma.801 }
        }
    }
}

                exists = scope:aubma_holding

                    has_building = pb_mandala_03
                    has_building = sacred_pool_1
                    has_building = temple_viharas_2
                }
                scope:aubma_great = { has_building = mandala_capital_02 NOT = { has_building_or_higher = mandala_capital_03 } }
            }
            debug_log = "AUBMA: READY actual_player_owned_pagan_native_primaries"
            save_scope_as = character

                province_owner = { save_scope_as = holder }
                province_owner = { save_scope_as = build_owner }
                county = { save_scope_as = county }
                # Province culture, not payer culture. No late innovations.
                culture = {
                    if = {
                        limit = { NOT = { has_cultural_era_or_later = culture_era_high_medieval } }
                        join_era = culture_era_high_medieval
                    }
                }
                # Stock remove_district_type, preserving holding and Mandala.
                remove_district_type = ceramics_district
                remove_district_type = walls_district
                remove_district_type = urban_housing_district
                remove_district_type = temple_district
                remove_district_type = patrician_district
            }

            trigger_event = { id = aubma.3 days = 1 }
        }
        else = {

            random_player = { trigger_event = aubma.801 }
        }
    }
}
