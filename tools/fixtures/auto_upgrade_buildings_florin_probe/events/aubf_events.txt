namespace = aubf

# Capability probe only. This does not replace the original AUB business matrix.
aubf.1 = {
	type = character_event
	hidden = yes
	immediate = {
		debug_log = "AUBF: TEST BEGIN district_native_probe"
		if = {
			limit = {
				this = character:1128
				capital_province = { has_holding_type = castle_holding }
			}
			debug_log = "AUBF: TEST PASS controlled_robert_castle"
			remove_character_flag = enable_auto_build
			remove_character_flag = aub_funding_treasury_only
			add_character_flag = aub_funding_personal_only
			save_scope_as = aub_payer
			save_scope_as = character
			# Florin holding primaries use the province culture's era, not
			# the payer culture's innovations. This isolated fixture advances
			# that shared culture only to make the next-next tier eligible.
			capital_province = {
				culture = {
					if = {
						limit = { NOT = { has_cultural_era_or_later = culture_era_high_medieval } }
						join_era = culture_era_high_medieval
					}
				}
				if = {
					limit = { culture = { has_cultural_era_or_later = culture_era_high_medieval } }
					debug_log = "AUBF: TEST PASS target_province_high_medieval"
				}
				else = { debug_log = "AUBF: TEST FAIL target_province_high_medieval" }
			}
			add_gold = 100000
			set_variable = { name = aub_upgrades_this_scan value = 0 }
			set_variable = {
				name = aubf_expected_gold
				value = { value = gold subtract = main_building_tier_2_cost }
			}
			capital_province = {
				province_owner = { save_scope_as = holder }
				county = { save_scope_as = county }
				holding_district = {
					if = { limit = { has_building = castle_04 } downgrade_building_effect = castle_04 }
					if = { limit = { has_building = castle_03 } downgrade_building_effect = castle_03 }
					if = { limit = { has_building = castle_02 } downgrade_building_effect = castle_02 }
					if = {
						limit = { has_building = castle_01 can_start_building_construction_for_owner = castle_02 }
						debug_log = "AUBF: TEST PASS native_district_eligibility"
					}
					else = { debug_log = "AUBF: TEST FAIL native_district_eligibility" }
					aub_upgrade_chain_castle_01_effect = yes
					if = {
						limit = { has_building = castle_02 NOT = { has_building = castle_03 } }
						debug_log = "AUBF: TEST PASS native_district_upgrade_one_tier"
					}
					else = { debug_log = "AUBF: TEST FAIL native_district_upgrade_one_tier" }
				}
			}
			if = {
				limit = { gold = var:aubf_expected_gold var:aub_upgrades_this_scan = 1 }
				debug_log = "AUBF: TEST PASS exact_personal_charge_and_count"
			}
			else = { debug_log = "AUBF: TEST FAIL exact_personal_charge_and_count" }
			capital_province = {
				holding_district = {
					if = { limit = { has_building = castle_02 } downgrade_building_effect = castle_02 }
				}
			}
			set_variable = { name = aub_upgrades_this_scan value = 0 }
			capital_province = { aub_upgrade_all_supported_buildings_effect = yes }
			capital_province = {
				holding_district = {
					if = { limit = { has_building = castle_01 } debug_log = "AUBF: OBSERVE full_scan_castle01" }
					if = { limit = { has_building = castle_02 } debug_log = "AUBF: OBSERVE full_scan_castle02" }
					if = { limit = { has_building = castle_03 } debug_log = "AUBF: OBSERVE full_scan_castle03" }
					if = { limit = { has_building = castle_04 } debug_log = "AUBF: OBSERVE full_scan_castle04" }
					if = {
						limit = { can_start_building_construction_for_owner = castle_03 }
						debug_log = "AUBF: OBSERVE full_scan_castle03_eligible"
					}
					else = { debug_log = "AUBF: OBSERVE full_scan_castle03_blocked" }
				}
			}
			if = { limit = { var:aub_upgrades_this_scan = 0 } debug_log = "AUBF: OBSERVE full_scan_count00" }
			if = { limit = { var:aub_upgrades_this_scan = 1 } debug_log = "AUBF: OBSERVE full_scan_count01" }
			if = { limit = { var:aub_upgrades_this_scan = 2 } debug_log = "AUBF: OBSERVE full_scan_count02" }
			if = { limit = { var:aub_upgrades_this_scan = 3 } debug_log = "AUBF: OBSERVE full_scan_count03" }
			if = { limit = { var:aub_upgrades_this_scan = 4 } debug_log = "AUBF: OBSERVE full_scan_count04" }
			if = { limit = { var:aub_upgrades_this_scan = 5 } debug_log = "AUBF: OBSERVE full_scan_count05" }
			if = { limit = { var:aub_upgrades_this_scan = 6 } debug_log = "AUBF: OBSERVE full_scan_count06" }
			if = { limit = { var:aub_upgrades_this_scan = 7 } debug_log = "AUBF: OBSERVE full_scan_count07" }
			if = { limit = { var:aub_upgrades_this_scan = 8 } debug_log = "AUBF: OBSERVE full_scan_count08" }
			if = { limit = { var:aub_upgrades_this_scan = 9 } debug_log = "AUBF: OBSERVE full_scan_count09" }
			if = { limit = { var:aub_upgrades_this_scan = 10 } debug_log = "AUBF: OBSERVE full_scan_count10" }
			if = { limit = { var:aub_upgrades_this_scan = 11 } debug_log = "AUBF: OBSERVE full_scan_count11" }
			if = { limit = { var:aub_upgrades_this_scan = 12 } debug_log = "AUBF: OBSERVE full_scan_count12" }
			if = { limit = { var:aub_upgrades_this_scan = 13 } debug_log = "AUBF: OBSERVE full_scan_count13" }
			if = { limit = { var:aub_upgrades_this_scan = 14 } debug_log = "AUBF: OBSERVE full_scan_count14" }
			if = { limit = { var:aub_upgrades_this_scan = 15 } debug_log = "AUBF: OBSERVE full_scan_count15" }
			if = { limit = { var:aub_upgrades_this_scan > 15 } debug_log = "AUBF: TEST FAIL count_above_cap" }
			if = {
				limit = {
					capital_province = {
						holding_district = {
							has_building = castle_02
							NOT = { has_building_or_higher = castle_03 }
							can_start_building_construction_for_owner = castle_03
						}
					}
					var:aub_upgrades_this_scan > 0
					var:aub_upgrades_this_scan < 15
				}
				debug_log = "AUBF: TEST PASS primary_mutation_no_repeat_before_cap"
			}
			else = { debug_log = "AUBF: TEST FAIL primary_mutation_no_repeat_before_cap" }
			capital_province = {
				holding_district = {
					if = { limit = { has_building = castle_04 } downgrade_building_effect = castle_04 }
					if = { limit = { has_building = castle_03 } downgrade_building_effect = castle_03 }
					if = { limit = { has_building = castle_02 } downgrade_building_effect = castle_02 }
				}
			}
			set_variable = { name = aubf_all_gold value = gold }
			remove_short_term_gold = var:aubf_all_gold
			set_variable = { name = aub_upgrades_this_scan value = 0 }
			capital_province = { holding_district = { aub_upgrade_chain_castle_01_effect = yes } }
			if = {
				limit = {
					gold = 0
					var:aub_upgrades_this_scan = 0
					capital_province = {
						holding_district = { has_building = castle_01 NOT = { has_building = castle_02 } }
					}
				}
				debug_log = "AUBF: TEST PASS insufficient_personal_zero_change"
			}
			else = { debug_log = "AUBF: TEST FAIL insufficient_personal_zero_change" }
		}
		else = { debug_log = "AUBF: TEST FAIL controlled_robert_castle" }
		debug_log = "AUBF: TEST DONE district_native_probe"
	}
}
