				recipient = var:zqa_release_hc_target

			}

			} }

				recipient = var:zqa_release_rc_target
				interaction = xqol_release_hook_recruit_interaction

				recipient = var:zqa_release_rc_target

				recipient = var:zqa_release_h_target

				recipient = var:zqa_release_h_target

			} }
			NOT = { is_character_interaction_potentially_accepted = {
				recipient = var:zqa_release_h_target
				interaction = xqol_release_recruit_conversion_interaction
			} }
			is_character_interaction_potentially_accepted = {
				recipient = var:zqa_release_h_target
				interaction = xqol_release_hook_interaction
			}

			# R: all earlier branches and H are unavailable, then R accepts.
			NOT = { is_character_interaction_potentially_accepted = {

zqa_verify_release_matrix_effect = {
	if = {
		limit = { has_character_flag = zqa_await_release_matrix }
		remove_character_flag = zqa_await_release_matrix
		if = {
			limit = {
				var:xqol_last_bulk_release_count = 7
				var:zqa_release_hrc_target = { is_imprisoned = no }
				var:zqa_release_hr_target = {
					is_imprisoned = no
					faith = faith:catholic
				}
				var:zqa_release_hc_target = { is_imprisoned = no }
				var:zqa_release_rc_target = { is_imprisoned = no }
				var:zqa_release_h_target = { is_imprisoned = no }
				var:zqa_release_r_target = { is_imprisoned = no }
				var:zqa_release_c_target = { is_imprisoned = no }
				var:zqa_release_reject_target = {
					is_imprisoned_by = root
					faith = faith:catholic
				}

				# Hook-bearing outcomes (all four started without a hook).
				has_hook = var:zqa_release_hrc_target
				has_hook = var:zqa_release_hr_target
				has_hook = var:zqa_release_hc_target
				has_hook = var:zqa_release_h_target

				# Recruit-bearing outcomes.
				var:zqa_release_hrc_target = { is_courtier_of = root }
				var:zqa_release_hr_target = { is_courtier_of = root }
				var:zqa_release_rc_target = { is_courtier_of = root }
				var:zqa_release_r_target = { is_courtier_of = root }

				# Conversion-bearing outcomes.  HR remaining Catholic proves the
				# equal-size HR branch beat the later HC and RC branches.
				var:zqa_release_hrc_target = { faith = root.faith }
				var:zqa_release_hc_target = { faith = root.faith }
				var:zqa_release_rc_target = { faith = root.faith }
				var:zqa_release_c_target = { faith = root.faith }
				var:zqa_release_h_target = { faith = root.faith }
				var:zqa_release_r_target = { faith = root.faith }
			}
			debug_log = "ZQA: TEST PASS release_priority_matrix"
			# Unrelated whole phase2 completion marker excluded; only release/Rite is exercised.
		}
		else = {
			debug_log = "ZQA: TEST FAIL release_priority_matrix"
		}


        if = {
            limit = {
                var:xqol_last_bulk_release_count = 7
                var:zqa_release_hrc_target = { rite = root.rite }
                var:zqa_release_hc_target = { rite = root.rite }
                var:zqa_release_rc_target = { rite = root.rite }
                var:zqa_release_c_target = { rite = root.rite }
            }
            debug_log = "ZQA120: TEST PASS release_conversion_rite_matched"
        }
        else = { debug_log = "ZQA120: TEST FAIL release_conversion_rite_matched" }
		remove_variable = zqa_release_hrc_target
		remove_variable = zqa_release_hr_target
		remove_variable = zqa_release_hc_target
		remove_variable = zqa_release_rc_target
		remove_variable = zqa_release_h_target
		remove_variable = zqa_release_r_target
		remove_variable = zqa_release_c_target
		remove_variable = zqa_release_reject_target
		zqa120_accepted_rite_probe_effect = yes
	}
}
