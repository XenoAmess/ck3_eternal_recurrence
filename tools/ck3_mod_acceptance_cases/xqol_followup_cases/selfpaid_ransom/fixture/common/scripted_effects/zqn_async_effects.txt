# Extracted from zqa ransom setup/full/any; no conversion/release-terms matrix.
# Test-only wallet construction writes only the two real self-payers' own gold.
# No callback, transfer, release, hidden payer or doctrinal reward is fabricated.
zqnr_setup_self_ransom_effect = {
 save_scope_as = actor
 save_scope_as = puppet_or_actor
 random_vassal = {
  limit = { is_ai = yes is_alive = yes is_adult = yes is_landed = yes is_imprisoned = no NOT = { is_at_war_with = scope:zqnr_actual_actor } }
  save_scope_as = zqnr_full
 }
 random_vassal = {
  limit = { is_ai = yes is_alive = yes is_adult = yes is_landed = yes is_imprisoned = no NOT = { this = scope:zqnr_full } NOT = { is_at_war_with = scope:zqnr_actual_actor } }
  save_scope_as = zqnr_any
 }
 if = {
  limit = { exists = scope:zqnr_full exists = scope:zqnr_any NOT = { dynasty ?= { has_dynasty_perk = fp1_pillage_legacy_3 } } }
  every_prisoner = { release_from_prison = yes }
  imprison = { target = scope:zqnr_full type = dungeon }
  imprison = { target = scope:zqnr_any type = dungeon }
  set_variable = { name = zqnr_full_target value = scope:zqnr_full }
  set_variable = { name = zqnr_any_target value = scope:zqnr_any }
  scope:zqnr_full = { remove_long_term_gold = gold add_gold = normal_ransom_cost_value }
  scope:zqnr_any = { remove_long_term_gold = gold add_gold = 1 }
  if = {
   limit = {
    is_character_interaction_valid = { recipient = scope:zqnr_full interaction = ransom_interaction }
    is_character_interaction_potentially_accepted = { recipient = scope:zqnr_full interaction = ransom_interaction }
    is_character_interaction_valid = { recipient = scope:zqnr_any interaction = ransom_interaction }
    is_character_interaction_potentially_accepted = { recipient = scope:zqnr_any interaction = ransom_interaction }
    scope:zqnr_full = { is_ruler = yes is_imprisoned_by = scope:zqnr_actual_actor xqol_ransom_payer_can_pay_full_trigger = yes normal_ransom_cost_value > 1 gold >= normal_ransom_cost_value NOT = { gold > normal_ransom_cost_value } }
    scope:zqnr_any = { is_ruler = yes is_imprisoned_by = scope:zqnr_actual_actor xqol_ransom_payer_has_gold_trigger = yes NOT = { xqol_ransom_payer_can_pay_full_trigger = yes } normal_ransom_cost_value > 1 gold >= 1 NOT = { gold > 1 } }
    NOT = { any_prisoner = { NOR = { this = scope:zqnr_actual_actor.var:zqnr_full_target this = scope:zqnr_actual_actor.var:zqnr_any_target } } }
   }
   debug_log = "ZQRS120: TEST READY normal_full_self_quote"
   debug_log = "ZQRS120: TEST READY one_gold_self_quote"
   debug_log = "ZQRS120: SCOPE BEGIN actual_self_payers_D0"
   debug_log_scopes = yes
   debug_log = "ZQRS120: SCOPE END actual_self_payers_D0"
   add_character_flag = zqnr_dispatch_full_once
   trigger_event = { id = zqn120.2 days = 1 }
  }
  else = { debug_log = "ZQRS120: TEST FAIL actual_self_ransom_validity_quote_wallet" }
 }
 else = { debug_log = "ZQRS120: TEST FAIL self_payers_missing_or_pillage_branch" }
}

zqnr_dispatch_full_effect = {
 if = {
  limit = { has_character_flag = zqnr_dispatch_full_once }
  remove_character_flag = zqnr_dispatch_full_once
  save_scope_as = actor
  save_scope_as = puppet_or_actor
  var:zqnr_full_target = { save_scope_as = recipient }
  if = {
   limit = {
    var:zqnr_full_target = { is_alive = yes is_ruler = yes is_imprisoned_by = scope:zqnr_actual_actor xqol_ransom_payer_can_pay_full_trigger = yes gold >= normal_ransom_cost_value NOT = { gold > normal_ransom_cost_value } }
    var:zqnr_any_target = { is_alive = yes is_ruler = yes is_imprisoned_by = scope:zqnr_actual_actor gold >= 1 NOT = { gold > 1 } NOT = { xqol_ransom_payer_can_pay_full_trigger = yes } }
    is_character_interaction_potentially_accepted = { recipient = var:zqnr_full_target interaction = ransom_interaction }
   }
   set_variable = { name = zqnr_full_quote value = var:zqnr_full_target.normal_ransom_cost_value }
   set_variable = { name = zqnr_full_wallet_before value = var:zqnr_full_target.gold }
   set_variable = { name = zqnr_full_piety_before value = var:zqnr_full_target.piety }
   set_variable = { name = zqnr_captor_full_before value = gold }
   set_variable = { name = zqnr_expected_captor value = gold }
   change_variable = { name = zqnr_expected_captor add = var:zqnr_full_target.normal_ransom_cost_value }
   var:zqnr_full_target = {
    if = { limit = { has_personal_tenet_flag = tenet_philanthropia_ransom_captives } debug_log = "ZQRS120: OBSERVED full_personal_philanthropia_yes" }
    else = { debug_log = "ZQRS120: OBSERVED full_personal_philanthropia_no" }
   }
   set_variable = { name = zqnr_wait_days value = 0 }
   add_character_flag = zqnr_wait_full
   debug_log = "ZQRS120: SCOPE BEGIN full_quote_and_real_self_wallet_before"
   debug_log_scopes = yes
   debug_log = "ZQRS120: SCOPE END full_quote_and_real_self_wallet_before"
   xqol_bulk_ransom_full_effect = yes
   trigger_event = { id = zqn120.3 days = 1 }
  }
  else = { debug_log = "ZQRS120: TEST FAIL full_dispatch_fresh_quote_wallet_guard" }
 }
}
