ervc_cc_toggle_trait_gui = {
	scope = character
	saved_scopes = { ervc_trait }
	is_shown = { ervc_cc_ui_access_trigger = yes }
	is_valid = {
		ervc_cc_ui_access_trigger = yes
		exists = scope:ervc_trait
		OR = {
			ervc_cc_trait_is_selected_trigger = { ervc_trait = scope:ervc_trait }
			AND = {
				scope:ervc_trait = { save_temporary_scope_as = ervc_cc_candidate_trait }
				ervc_cc_candidate_trait_metadata_valid_trigger = yes
				OR = {
					is_target_in_variable_list = {
						name = ervc_cc_catalog_education
						target = scope:ervc_cc_candidate_trait
					}
					AND = {
						is_target_in_variable_list = {
							name = ervc_cc_catalog_commander
							target = scope:ervc_cc_candidate_trait
						}
						OR = {
							AND = {
								var:ervc_cc_commander_count < 2
								OR = {
									NOT = { has_variable_list = ervc_cc_selected_commander }
									variable_list_size = {
										name = ervc_cc_selected_commander
										value < 2
									}
								}
							}
							ervc_cc_candidate_replaces_selected_trait_trigger = yes
						}
					}
					is_target_in_variable_list = {
						name = ervc_cc_catalog_physical
						target = scope:ervc_cc_candidate_trait
					}
					AND = {
						is_target_in_variable_list = {
							name = ervc_cc_catalog_personality
							target = scope:ervc_cc_candidate_trait
						}
						OR = {
							AND = {
								var:ervc_cc_personality_count < 3
								OR = {
									NOT = { has_variable_list = ervc_cc_selected_personality }
									variable_list_size = {
										name = ervc_cc_selected_personality
										value < 3
									}
								}
							}
							ervc_cc_candidate_replaces_selected_trait_trigger = yes
						}
					}
					is_target_in_variable_list = {
						name = ervc_cc_catalog_other
						target = scope:ervc_cc_candidate_trait
					}
				}
			}
		}
	}
	effect = {
		if = {
			limit = {
				ervc_cc_ui_access_trigger = yes
				exists = scope:ervc_trait
			}
			scope:ervc_trait = { save_scope_as = ervc_cc_candidate_trait }
			if = {
				limit = {
					ervc_cc_trait_is_selected_trigger = {
						ervc_trait = scope:ervc_cc_candidate_trait
					}
				}
				ervc_cc_remove_selected_trait_effect = {
					ervc_trait = scope:ervc_cc_candidate_trait
				}
			}
			else_if = {
				limit = { ervc_cc_candidate_trait_metadata_valid_trigger = yes }
				ervc_cc_remove_selected_trait_conflicts_effect = yes
				if = {
					limit = {
						is_target_in_variable_list = {
							name = ervc_cc_catalog_education
							target = scope:ervc_cc_candidate_trait
						}
					}
					clear_variable_list = ervc_cc_selected_education
					add_to_variable_list = {
						name = ervc_cc_selected_education
						target = scope:ervc_cc_candidate_trait
					}
				}
				else_if = {
					limit = {
						is_target_in_variable_list = {
							name = ervc_cc_catalog_commander
							target = scope:ervc_cc_candidate_trait
						}
						var:ervc_cc_commander_count < 2
						OR = {
							NOT = { has_variable_list = ervc_cc_selected_commander }
							variable_list_size = {
								name = ervc_cc_selected_commander
								value < 2
							}
						}
					}
					add_to_variable_list = {
						name = ervc_cc_selected_commander
						target = scope:ervc_cc_candidate_trait
					}
					change_variable = { name = ervc_cc_commander_count add = 1 }
				}
				else_if = {
					limit = {
						is_target_in_variable_list = {
							name = ervc_cc_catalog_physical
							target = scope:ervc_cc_candidate_trait
						}
					}
					add_to_variable_list = {
						name = ervc_cc_selected_physical
						target = scope:ervc_cc_candidate_trait
					}
				}
				else_if = {
					limit = {
						is_target_in_variable_list = {
							name = ervc_cc_catalog_personality
							target = scope:ervc_cc_candidate_trait
						}
						var:ervc_cc_personality_count < 3
						OR = {
							NOT = { has_variable_list = ervc_cc_selected_personality }
							variable_list_size = {
								name = ervc_cc_selected_personality
								value < 3
							}
						}
					}
					add_to_variable_list = {
						name = ervc_cc_selected_personality
						target = scope:ervc_cc_candidate_trait
					}
					change_variable = { name = ervc_cc_personality_count add = 1 }
				}
				else_if = {
					limit = {
						is_target_in_variable_list = {
							name = ervc_cc_catalog_other
							target = scope:ervc_cc_candidate_trait
						}
					}
					add_to_variable_list = {
						name = ervc_cc_selected_other
						target = scope:ervc_cc_candidate_trait
					}
				}
			}
		}
	}
}
