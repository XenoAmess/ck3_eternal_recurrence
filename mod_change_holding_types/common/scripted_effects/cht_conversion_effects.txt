# Character scope; scope:barony is supplied by the native title selector.

cht_convert_to_castle_effect = {
    if = {
        limit = {
            is_ai = no
            exists = scope:barony
            scope:barony = { ch_barony_is_valid_for_castle_trigger = { CHARACTER = root } }
        }
        scope:barony.title_province = { set_holding_type = castle_holding }
    }
}
