	}

		name = zg361.4.b
		trigger = {
			influence >= 150
			has_variable = zg361_result_appeal_open
			var:zg361_result_appeal_open = 1
			has_variable = zg361_result_case_state
			var:zg361_result_case_state = 3
		}

		zg361_b2_on_appeal_filed_effect = yes
		# An actually independent B2 reviewer is consumed by the real appeal
		# roll. It improves review quality without changing the sanction ceiling.
		if = {
			limit = { has_variable = zg361_b2_m077_quality_bonus var:zg361_b2_m077_quality_bonus = 10 }
			random = {
				chance = 40
				set_variable = { name = zg361_appeal_won value = 1 }
			}
		}
		else = {
			random = {
				chance = 30
				set_variable = { name = zg361_appeal_won value = 1 }
			}
		}
		if = {
			limit = { has_variable = zg361_appeal_won }
			remove_variable = zg361_appeal_won
			zg361_appeal_regrade_to_35_effect = yes
			debug_log = "ZG361: player appeal won, regraded to 3.5"
		}
		else = {
			zg361_b2_on_appeal_upheld_effect = yes
			var:zg361_result_case_owner = {
				if = {
					limit = { is_alive = yes }
					add_opinion = { modifier = zg361_annoyed target = root }
				}
			}
			debug_log = "ZG361: player appeal rejected"
		}
	}

	option = { # 摆烂
		name = zg361.4.c
		add_stress = medium_stress_loss
		set_variable = { name = zg361_slacking value = 1 }
	}

	option = { # 奋发
		name = zg361.4.d
		add_stress = minor_stress_gain
		set_variable = { name = zg361_striving value = 1 }
	}
