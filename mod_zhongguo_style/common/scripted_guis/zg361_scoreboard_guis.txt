# ZhongGuo 361 Style — scoreboard GUI predicates

zg361_review_now_bridge_gui = {
	scope = character

	is_shown = {
		is_ai = no
		has_character_flag = zg361_review_now_pending
	}

	effect = {
		if = {
			limit = { has_character_flag = zg361_review_now_pending }
			remove_character_flag = zg361_review_now_pending
			zg361_run_review_effect = yes
		}
	}
}

zg361_view_result_statement_bridge_gui = {
	scope = character

	is_shown = {
		is_ai = no
		has_character_flag = zg361_view_result_statement_pending
	}

	effect = {
		if = {
			limit = { has_character_flag = zg361_view_result_statement_pending }
			remove_character_flag = zg361_view_result_statement_pending
			trigger_event = zg361.53
		}
	}
}
