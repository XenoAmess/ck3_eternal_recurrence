			}
			triggered_desc = {
				trigger = { zg361_ratio_policy_relaxed_current_trigger = yes }
				desc = zg361_ratio_policy_current_relaxed
			}
			triggered_desc = {
				trigger = { zg361_ratio_policy_off_current_trigger = yes }
				desc = zg361_ratio_policy_current_off
			}
			desc = zg361_ratio_policy_current_strict
		}
	}

	is_shown = {
		zg361_is_celestial_liege_trigger = yes
		has_game_rule = zg361_on
	}
	is_valid = {
		prestige >= 75

			text = zg361_ratio_policy_decision_ready

		}
	}
	is_valid_showing_failures_only = {
		prestige >= 75
		custom_description = {
			text = zg361_ratio_policy_decision_ready

		add_character_flag = zg361_ratio_policy_pending
	}
}
