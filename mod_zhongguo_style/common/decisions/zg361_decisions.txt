	decision_group_type = zg361
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_realm.dds"
	}
	ai_check_interval = 0
	cooldown = { years = 1 }

	is_shown = {
		zg361_is_celestial_liege_trigger = yes
		has_game_rule = zg361_on
	}
	is_valid = {
		zg361_review_now_business_valid_trigger = yes
	}
	is_valid_showing_failures_only = {
		zg361_review_now_business_valid_trigger = yes
	}

	cost = { prestige = 150 }

	# 决议 effect 会被 UI 预评估：这里只写一次性旗标，由注册 GUI 在真实点击后消费。
	effect = {
		add_character_flag = zg361_review_now_pending
	}
}

	}
}

# 末位比例：刚性 361（10%）
zg361_ratio_strict_decision = {
	decision_group_type = zg361
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_realm.dds"
	}
	ai_check_interval = 0
	cooldown = { years = 1 }

	is_shown = {
		zg361_is_celestial_liege_trigger = yes
		has_game_rule = zg361_on
		trigger_if = {
			limit = { has_variable = zg361_ratio_override }
			NOT = { var:zg361_ratio_override = 10 }
		}
	}
	is_valid = { prestige >= 75 }
	is_valid_showing_failures_only = { prestige >= 75 }

	cost = { prestige = 75 }

	effect = {
		set_variable = { name = zg361_ratio_override value = 10 }
		debug_log = "ZG361: liege set bottom ratio override to strict (10%)"
	}
}
