# ZhongGuo 361 Style — decisions
# 末位比例三决议 + 立即开榜。AI 不主动使用（ai_check_interval = 0），AI 侧由年度 tick 静默覆盖。

# 立即开展绩效考核（提前开榜）
zg361_review_now_decision = {
	decision_group_type = zg361
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_realm.dds"
	}
	ai_check_interval = 0
	cooldown = { years = 1 }

	is_shown = {
		zg361_is_celestial_liege_trigger = yes
		has_game_rule = zg361_on
	}
	is_valid = {
		NOT = { has_character_flag = zg361_review_in_progress }
		trigger_if = {
			limit = { has_variable = zg361_last_settled_year }
			NOT = { var:zg361_last_settled_year = current_year }
		}
		any_vassal = {
			count >= 1
			zg361_is_reviewable_vassal_trigger = yes
		}
	}
	is_valid_showing_failures_only = {
		NOT = { has_character_flag = zg361_review_in_progress }
		trigger_if = {
			limit = { has_variable = zg361_last_settled_year }
			NOT = { var:zg361_last_settled_year = current_year }
		}
		any_vassal = {
			count >= 1
			zg361_is_reviewable_vassal_trigger = yes
		}
	}

	cost = { prestige = 150 }

	# 决议 effect 会被 UI 预评估：这里只写一次性旗标，由注册 GUI 在真实点击后消费。
	effect = {
		add_character_flag = zg361_review_now_pending
	}
}
