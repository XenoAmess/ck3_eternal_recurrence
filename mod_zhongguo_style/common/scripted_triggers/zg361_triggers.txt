# ZhongGuo 361 Style — scripted triggers

# 考核执行者：所有公爵及以上天朝制在任领主；每人只考核自己的直属官员
zg361_is_celestial_liege_trigger = {
	government_has_flag = government_is_celestial
	highest_held_title_tier >= tier_duchy
	is_landed = yes
	is_alive = yes
}

# 被考核对象：合格领主的全部直属在任有地官员，不以本人的政府或爵位再筛一次。
# 伯爵/男爵只能作为受评者，因为考核执行者 trigger 另有 tier_duchy 下限。
zg361_is_reviewable_vassal_trigger = {
	is_alive = yes
	is_landed = yes
	liege = { zg361_is_celestial_liege_trigger = yes }
}

# “立即开展绩效考核”决议与原生自动玩家入口共用同一业务合法性。
# prestige 由各入口各自按同一个 150 成本核对；这里仅冻结业务前置，避免
# 决议与无坐标 named-widget 入口日后发生语义漂移。
zg361_review_now_business_valid_trigger = {
	NOT = { has_character_flag = zg361_review_in_progress }
	trigger_if = {
		limit = { has_variable = zg361_last_settled_year }
		NOT = { var:zg361_last_settled_year = current_year }
	}
	any_vassal = {
		count >= 1
		zg361_is_reviewable_vassal_trigger = yes
	}
}

# 京察合法来宾：命令签发、活动开办与逾期追责必须共用这一口径。
# 直属关系由调用处的 any_vassal / scope:host 约束，这里只定义人物资格。
zg361_is_jingcha_eligible_guest_trigger = {
	zg361_is_reviewable_vassal_trigger = yes
	age >= 16
	is_available = yes
	is_imprisoned = no
}

# “踩一个人”只能在真实存在一名非新人 3.5 与一名 3.25 时开放。
# 这样校准永远是等额交换，不会凭空制造末位，也不会击穿新人保护。
zg361_can_calibrate_demote_trigger = {
	any_vassal = {
		zg361_is_current_liege_review_record_trigger = yes
		# 同一 trigger block 不保证短路；未入考核池的直属封臣没有
		# pending_grade，必须把数值读取放进存在性门内，避免事件 tooltip
		# 预求值时向 error.log 刷 unset-variable。
		trigger_if = {
			limit = { has_variable = zg361_pending_grade }
			var:zg361_pending_grade = 2
		}
		trigger_else = { always = no }
		NOT = { has_character_flag = zg361_newcomer_this_cycle }
	}
	any_vassal = {
		zg361_is_current_liege_review_record_trigger = yes
		trigger_if = {
			limit = { has_variable = zg361_pending_grade }
			var:zg361_pending_grade = 1
		}
		trigger_else = { always = no }
	}
}

# 背靠背互评只有在本直属考核单元达到强制分布门槛时才有下期 KPI 意义。
zg361_has_ranked_peer_cohort_trigger = {
	zg361_is_reviewable_vassal_trigger = yes
	liege = {
		any_vassal = {
			count >= 3
			zg361_is_reviewable_vassal_trigger = yes
		}
	}
}

# B1 互评提交者与对象。所有数值读取都放在存在性 trigger_if 后，避免
# interaction 列表枚举普通角色时产生 unset-variable 错误。
zg361_b1_peer_submission_actor_trigger = {
	trigger_if = {
		limit = {
			exists = liege
			has_variable = zg361_b1_case_owner
			has_variable = zg361_b1_cycle_serial
			has_variable = zg361_b1_case_state
			has_variable = zg361_b1_peer_used
			has_variable = zg361_b1_peer_cap
		}
		var:zg361_b1_case_state = 3
		var:zg361_b1_peer_used < var:zg361_b1_peer_cap
		var:zg361_b1_case_owner = liege
	}
	trigger_else = { always = no }
}

zg361_b1_peer_submission_recipient_trigger = {
	trigger_if = {
		limit = {
			has_variable = zg361_b1_case_owner
			has_variable = zg361_b1_cycle_serial
			has_variable = zg361_b1_case_state
			has_variable = zg361_b1_peer_slot_1_filled
			has_variable = zg361_b1_peer_slot_2_filled
			has_variable = zg361_b1_peer_slot_3_filled
			scope:actor = {
				has_variable = zg361_b1_case_owner
				has_variable = zg361_b1_cycle_serial
			}
		}
		var:zg361_b1_case_owner = scope:actor.var:zg361_b1_case_owner
		var:zg361_b1_cycle_serial = scope:actor.var:zg361_b1_cycle_serial
		var:zg361_b1_case_state = 3
		OR = {
			var:zg361_b1_peer_slot_1_filled = 0
			var:zg361_b1_peer_slot_2_filled = 0
			var:zg361_b1_peer_slot_3_filled = 0
		}
	}
	trigger_else = { always = no }
}

# 本周期淘汰候选人：连续 3.25 且本周期仍 3.25（用于玩家领主事件内重新推导名单）
# 必须仍属于做出该记录的直属领主，且记录来自该领主的最新考核轮次。
zg361_is_current_liege_review_record_trigger = {
	# any_vassal 会枚举未参加本轮考核的直属封臣；AND 条件不短路，
	# 所以不能把 has_variable 与 var: 读取平铺在同一层。
	trigger_if = {
		limit = {
			has_variable = zg361_last_reviewer
			has_variable = zg361_last_review_serial
			liege = { has_variable = zg361_review_serial }
		}
		var:zg361_last_reviewer = liege
		var:zg361_last_review_serial = liege.var:zg361_review_serial
	}
	trigger_else = { always = no }
}

zg361_is_elimination_candidate_trigger = {
	zg361_is_reviewable_vassal_trigger = yes
	zg361_is_current_liege_review_record_trigger = yes
	has_character_modifier = zg361_grade_325
	has_variable = zg361_streak_bottom
	var:zg361_streak_bottom >= 2
}
