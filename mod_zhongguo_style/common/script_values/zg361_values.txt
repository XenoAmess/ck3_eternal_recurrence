# ============================================================
# KPI 分项证据（当前作用域 = 被考核封臣；调用链 ROOT 可能仍是考核领主）
#
# 二期首纵切把同一公式拆成八个可冻结分项。zg361_kpi_value 只做代数求和，
# 不再维护第二份平行公式；个人清算单、申诉和后续 PIP 都读取结算时冻结值。
# ============================================================

# 1. 治理产出：总督效率（-50~+50）×2；直属男爵无总督效率，记 0。
zg361_kpi_governance_evidence_value = {
	value = 0
	if = {
		limit = { is_governor = yes }
		add = {
			value = governor_efficiency_presented
			multiply = 2
		}
	}
	if = {
		limit = { has_variable = zg361_b1_evidence_late_frozen }
		if = {
			limit = { var:zg361_b1_evidence_late_frozen = 1 }
			add = zg361_b1_goal_adjustment_value
		}
	}
}

# D+300 goal/baseline consumer for #002/#003/#005/#006/#047. The three
# evidence windows remain immutable variables; only their bounded delta enters
# the governance component, so the authoritative KPI is still exactly eight
# components and a conservative target cannot buy 3.75 by itself.
zg361_b1_goal_adjustment_value = {
	value = 0
	if = {
		limit = {
			has_variable = zg361_b1_evidence_early
			has_variable = zg361_b1_evidence_late
		}
		add = {
			value = var:zg361_b1_evidence_late
			subtract = var:zg361_b1_evidence_early
			multiply = var:zg361_b1_goal_weight
			divide = 100
			max = 20
			min = -20
		}
	}
	if = {
		limit = { has_variable = zg361_b1_difficulty_adjustment }
		add = var:zg361_b1_difficulty_adjustment
	}
}

# 2. 当前职级/贤能品级。
zg361_kpi_capability_evidence_value = {
	value = 0
	if = {
		limit = { government_has_flag = government_has_merit }
		add = {
			value = merit_level
			multiply = 8
		}
	}
}

# 3. 相对上一轮的成长；首次建账无快照时按 0。
zg361_kpi_growth_evidence_value = {
	value = 0
	if = {
		limit = {
			government_has_flag = government_has_merit
			has_variable = zg361_prev_merit_level
		}
		if = {
			limit = { merit_level >= var:zg361_prev_merit_level }
			add = {
				value = merit_level
				subtract = var:zg361_prev_merit_level
				multiply = 10
				max = 20
			}
		}
		else = {
			add = {
				value = merit_level
				subtract = var:zg361_prev_merit_level
				multiply = 10
				min = -20
			}
		}
	}
}

		add = -20
	}
	else_if = {
		limit = { liege = { opinion = { target = prev value <= -20 } } }
		add = -10
	}
}

# 6. 上期奋斗/摆烂与 B1 密封同僚互评信号。

	value = 0
	if = {
		limit = { has_variable = zg361_striving }
		add = zg361_kpi_striving_bonus_value
	}
	if = {
		limit = { has_variable = zg361_slacking }
		add = {
			value = 0
			subtract = zg361_kpi_slacking_malus_value
		}
	}
	# B1 peer records are sealed evaluator/subject/cycle facts. Their bounded

	# collaboration consumer and remains capped at +10/-15.

		limit = {
			has_variable = zg361_b1_peer_normalized_score
			has_variable = zg361_b1_peer_use_mode
		}
		if = {
			limit = { var:zg361_b1_peer_use_mode = 2 }
			add = var:zg361_b1_peer_normalized_score
		}
	}
}

zg361_values_score_value = {
	value = 0

	# 忠诚：封臣对领主好感
	if = {
		limit = { opinion = { target = liege value >= 50 } }
		add = 20
	}
	else_if = {
		limit = { opinion = { target = liege value >= 20 } }
		add = 10
	}
	else_if = {
		limit = { opinion = { target = liege value <= -50 } }
		add = -30
	}
	else_if = {
		limit = { opinion = { target = liege value <= -20 } }
		add = -15
	}

	# 品性
	if = {
		limit = {
			OR = {
				has_trait = honest
				has_trait = trusting
				has_trait = loyal
			}
		}
		add = 10
	}
	if = {
		limit = {
			OR = {
				has_trait = deceitful
				has_trait = arbitrary
				has_trait = callous
			}
		}
		add = -10
	}
	if = {
		limit = {
			OR = {
				has_trait = murderer
				has_trait = cannibal
				has_trait = witch
				has_trait = deviant
			}
		}
		add = -30
	}

	# 行为红线
	if = {
		limit = { is_at_war_with_liege = yes }
		add = -50
	}
	if = {
		limit = { exists = joined_faction }
		add = -20
	}
}

# 分档阈值（root = 领主；读取领主变量 zg361_cohort_n）

# 头部名额：cohort × 30%，四舍五入，至少 1
zg361_top_slots_value = {
	value = 0
	add = var:zg361_cohort_n
	multiply = 0.3
	round = yes
	min = 1
}

# 末位名额：领主决议覆盖 > 游戏规则；刚性模式 cohort >= 5 保底 1
zg361_bottom_slots_value = {
	value = 0

	# 领主决议覆盖（比例决议写入 var:zg361_ratio_override：10/5/0）
	if = {
		limit = {
			has_variable = zg361_ratio_override
			var:zg361_ratio_override = 0
		}
		value = 0
	}
	else_if = {
		limit = {
			has_variable = zg361_ratio_override
			var:zg361_ratio_override = 5
		}
		add = {
			value = var:zg361_cohort_n
			multiply = 0.05
			floor = yes
		}
	}
	else_if = {
		limit = {
			has_variable = zg361_ratio_override
			var:zg361_ratio_override = 10
		}
		add = {
			value = var:zg361_cohort_n
			multiply = 0.1
			floor = yes
		}
		if = {
			limit = { var:zg361_cohort_n >= 5 }
			min = 1
		}
	}
	# 游戏规则兜底
	else_if = {
		limit = { has_game_rule = zg361_ratio_relaxed }
		add = {
			value = var:zg361_cohort_n
			multiply = 0.05
			floor = yes
		}
	}
	else_if = {
		limit = { has_game_rule = zg361_ratio_strict }
		add = {
			value = var:zg361_cohort_n
			multiply = 0.1
			floor = yes
		}
		if = {
			limit = { var:zg361_cohort_n >= 5 }
			min = 1
		}
	}
}

	}
	else_if = {
		limit = { liege = { opinion = { target = prev value >= 50 } } }
		add = -100
	}
	else_if = {
		limit = { liege = { opinion = { target = prev value >= 20 } } }
		add = -40
