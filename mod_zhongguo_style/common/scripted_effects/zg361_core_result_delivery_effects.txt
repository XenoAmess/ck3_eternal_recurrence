			has_variable = zg361_pending_grade
			OR = {
				zg361_is_current_liege_review_record_trigger = yes
				AND = {
					has_variable = zg361_b1_case_owner
					has_variable = zg361_b1_case_subject
					has_variable = zg361_b1_cycle_serial
					has_variable = zg361_b1_case_serial

					has_variable = zg361_b1_case_active

					var:zg361_b1_roster_employment_state = 2
					OR = { var:zg361_b1_leaver_route = 1 var:zg361_b1_leaver_route = 2 }
				}
			}
		}
		root = {
			if = {
				limit = { NOT = { has_variable = zg361_result_case_serial_cursor } }
				set_variable = { name = zg361_result_case_serial_cursor value = 0 }
			}
			if = {
				limit = { var:zg361_result_case_serial_cursor >= 999999 }
				set_variable = { name = zg361_result_case_serial_cursor value = 0 }
			}
			change_variable = { name = zg361_result_case_serial_cursor add = 1 }
		}

		set_variable = { name = zg361_result_case_owner value = root }
		set_variable = { name = zg361_result_cycle_serial value = root.var:zg361_review_serial }
		set_variable = { name = zg361_result_case_serial value = root.var:zg361_result_case_serial_cursor }
		set_variable = { name = zg361_result_grade value = var:zg361_pending_grade }
		set_variable = { name = zg361_result_absolute_grade value = var:zg361_absolute_grade }
		set_variable = { name = zg361_result_kpi_frozen value = var:zg361_kpi }
		set_variable = { name = zg361_result_rank_frozen value = var:zg361_rank }
		set_variable = { name = zg361_result_cohort_n_frozen value = root.var:zg361_cohort_n }

		set_variable = { name = zg361_result_evidence_superior value = var:zg361_evidence_superior }
		set_variable = { name = zg361_result_evidence_values value = var:zg361_evidence_values }
		set_variable = { name = zg361_result_evidence_collaboration value = var:zg361_evidence_collaboration }
		set_variable = { name = zg361_result_evidence_jingcha value = var:zg361_evidence_jingcha }
		set_variable = { name = zg361_result_evidence_organization value = var:zg361_evidence_organization }

		# reason: 0=事实与终档一致；1–4=四类校准交换；5=强制分布下调；
		# 6=新人保护抬升；7=横向配额抬升；8=小队列中性化；
		# 9=#040 灰色离任占用既有 C；10=原 C 承担者按离任者应得档对调。事实档永不被 reason 反写。
		set_variable = { name = zg361_result_grade_reason value = 0 }

			limit = { has_variable = zg361_calibration_reason }
			set_variable = { name = zg361_result_grade_reason value = var:zg361_calibration_reason }
		}
		else_if = {
			limit = { has_variable = zg361_grade_reason_override }
			set_variable = { name = zg361_result_grade_reason value = var:zg361_grade_reason_override }
		}
		else_if = {
			limit = {
				var:zg361_pending_grade < var:zg361_absolute_grade
			}
			set_variable = { name = zg361_result_grade_reason value = 5 }
		}

			limit = {
				has_character_flag = zg361_newcomer_this_cycle
				var:zg361_pending_grade > var:zg361_absolute_grade
			}
			set_variable = { name = zg361_result_grade_reason value = 6 }
		}

			limit = { var:zg361_pending_grade > var:zg361_absolute_grade }
			set_variable = { name = zg361_result_grade_reason value = 7 }
		}

		set_variable = { name = zg361_result_gold_paid value = 0 }
		set_variable = { name = zg361_result_merit_paid value = 0 }
		set_variable = { name = zg361_result_salary_cut_active value = 0 }
		set_variable = { name = zg361_result_appeal_open value = 0 }
		set_variable = { name = zg361_result_delivery_method value = 0 }

		# evaluation speculatively walks the delivery/refund effects and does not
		# short-circuit a has_variable guard before comparing the serial. Keeping
		# both receipts initialized avoids unset-scope errors without posting or
		# weakening either idempotence gate; real case serials are always 1..999999.
		set_variable = { name = zg361_result_settlement_posted_serial value = 0 }
		set_variable = { name = zg361_result_refund_posted_serial value = 0 }
		set_variable = { name = zg361_result_treasury_refunded value = 0 }
		set_variable = { name = zg361_result_gold_refunded value = 0 }
		set_variable = { name = zg361_result_merit_refunded value = 0 }
		set_variable = { name = zg361_result_delivered_year value = 0 }

		remove_variable = zg361_result_delivery_witness
		remove_variable = zg361_result_delivery_witness_receipt
		remove_variable = zg361_result_appeal_outcome
		if = {
			limit = { var:zg361_pending_grade = 1 }
			set_variable = { name = zg361_result_case_state value = 1 } # prepared
		}
		else = {
			set_variable = { name = zg361_result_case_state value = 3 } # delivered/settled
			set_variable = { name = zg361_result_delivery_method value = 4 } # automatic publication
			set_variable = { name = zg361_result_delivered_year value = current_year }
			set_variable = { name = zg361_result_settlement_posted_serial value = var:zg361_result_case_serial }
			# The signed Workforce slot is stable before this settlement chain.
			# Publish reads the new result only from a subject-root D+1 relay.
			if = {
				limit = {
					has_variable = zg361_we_m269_outcome_pending
					has_variable = zg361_we_m269_outcome_settled
					has_variable = zg361_we_m269_receipt_choice
					has_variable = zg361_we_m269_write_owner
					has_variable = zg361_we_m269_write_subject
					has_variable = zg361_we_m269_write_cycle
					has_variable = zg361_we_m269_write_case
					var:zg361_we_m269_outcome_pending = 1
					var:zg361_we_m269_outcome_settled = 0
					var:zg361_we_m269_write_subject = this
					OR = { var:zg361_we_m269_receipt_choice = 1 var:zg361_we_m269_receipt_choice = 2 }
					has_variable = zg361_workforce_attribution_fact_signature_committed
					has_variable = zg361_workforce_attribution_fact_state
					has_variable = zg361_workforce_attribution_fact_consumed
					has_variable = zg361_workforce_attribution_fact_subject
					has_variable = zg361_workforce_attribution_fact_owner
					has_variable = zg361_workforce_attribution_fact_cycle
					has_variable = zg361_workforce_attribution_fact_case
					var:zg361_workforce_attribution_fact_signature_committed = 1
					var:zg361_workforce_attribution_fact_state = 2
					var:zg361_workforce_attribution_fact_consumed = 0
					var:zg361_workforce_attribution_fact_subject = this
					var:zg361_workforce_attribution_fact_owner = var:zg361_we_m269_write_owner
					var:zg361_workforce_attribution_fact_cycle = var:zg361_we_m269_write_cycle
					var:zg361_workforce_attribution_fact_case = var:zg361_we_m269_write_case
					trigger_if = {
						limit = { has_variable = zg361_we_m269_result_relay_queued }
						var:zg361_we_m269_result_relay_queued = 0
					}
					trigger_else = { always = yes }
				}
				trigger_event = { id = zg361we.5378 days = 1 }
				set_variable = { name = zg361_we_m269_result_relay_queued value = 1 }
			}
		}
		# A departed old-owner row is informational and must not open PIP,
		# retaliation, or later-employer B2 state merely because its frozen grade
		# was posted. Current employment continues through the normal hook.

			limit = {
				NOT = {
					AND = {
						has_variable = zg361_b1_roster_employment_state
						has_variable = zg361_b1_leaver_route

						OR = { var:zg361_b1_leaver_route = 1 var:zg361_b1_leaver_route = 2 }
					}
				}
			}
			zg361_b2_on_result_frozen_effect = yes
		}
		debug_log = "ZG361P2: result case frozen"
	}
	else = {
		debug_log = "ZG361P2: result case freeze rejected by owner/subject/serial guard"
	}
}
