# ZhongGuo 361 Style — character interactions
# 申诉/绩效沟通/举荐/攻讦/逐人夺爵/逐人致仕。
# AI 不主动使用这些互动（ai_will_do = 0）：AI 侧奖惩由年度考核自动裁决，避免与淘汰链双重执行。

# ============================================================
# 绩效申诉（封臣 → 领主）：被打 3.25 / 背上 PIP 时翻案
# ============================================================
zg361_appeal_interaction = {
	icon = icon_contract_modification_single
	category = interaction_category_vassal
	desc = zg361_appeal_interaction_desc

	cooldown = { years = 1 }
	cooldown_against_recipient = { years = 1 }
	auto_accept = no

	is_shown = {
		scope:actor = {
			OR = {
				has_character_modifier = zg361_grade_325
				has_character_modifier = zg361_pip
			}
			liege = scope:recipient
		}
		scope:recipient = {
			zg361_is_celestial_liege_trigger = yes
		}
	}

	is_valid = {
		scope:actor = { influence >= 150 }
	}
	is_valid_showing_failures_only = {
		scope:actor = { influence >= 150 }
	}

	ai_accept = {
		base = 30
		modifier = {
			add = 20
			scope:recipient = { opinion = { target = scope:actor value >= 20 } }
		}
		modifier = {
			add = 15
			scope:actor = { merit_level >= 5 }
		}
		modifier = {
			add = -100
			scope:actor = { is_at_war_with_liege = yes }
		}
	}

	ai_will_do = { base = 0 }
	ai_frequency_by_tier = { }

	on_accept = {
		scope:actor = {
			change_influence = { value = 0 subtract = zg361_appeal_influence_cost_value }
			# 翻案：改判 3.5
			remove_character_modifier = zg361_grade_325
			remove_character_modifier = zg361_pip
			set_variable = { name = zg361_streak_bottom value = 0 }
			zg361_grade_35_apply_effect = yes
			# 领主侧计数修正
			liege = {
				if = {
					limit = { has_variable = zg361_last_325_n }
					change_variable = { name = zg361_last_325_n add = -1 }
					change_variable = { name = zg361_last_35_n add = 1 }
				}
			}
			debug_log = "ZG361: appeal accepted, regraded to 3.5"
		}
	}

	on_decline = {
		scope:actor = {
			change_influence = { value = 0 subtract = zg361_appeal_influence_cost_value }
			add_opinion = { modifier = zg361_appeal_rejected target = scope:recipient }
			debug_log = "ZG361: appeal rejected"
		}
	}
}

# 攻讦（同侪封臣 → 同侪封臣）：七成让对方下期 KPI -15，三成反噬
# ============================================================
zg361_slander_interaction = {
	icon = icon_damage_efficiency
	category = interaction_category_hostile
	desc = zg361_slander_interaction_desc

	cost = { prestige = 50 }

	cooldown = { years = 1 }
	cooldown_against_recipient = { years = 2 }
	auto_accept = yes

	is_shown = {
		scope:actor = {
			zg361_is_reviewable_vassal_trigger = yes
		}
		scope:recipient = {
			zg361_is_reviewable_vassal_trigger = yes
			NOT = { this = scope:actor }
			top_liege ?= scope:actor.top_liege
		}
	}

	ai_will_do = { base = 0 }
	ai_frequency_by_tier = { }

	on_accept = {
		scope:actor = {
			random = {
				chance = 70
				set_variable = { name = zg361_slander_hit value = 1 }
			}
			if = {
				limit = { has_variable = zg361_slander_hit }
				remove_variable = zg361_slander_hit
				scope:recipient = {
					set_variable = { name = zg361_slandered value = 1 }
				}
				debug_log = "ZG361: slander landed"
			}
			else = {
				set_variable = { name = zg361_slander_backfire value = 1 }
				scope:recipient = {
					add_opinion = { modifier = zg361_slandered_me target = scope:actor }
				}
				debug_log = "ZG361: slander backfired"
			}
		}
	}
}
