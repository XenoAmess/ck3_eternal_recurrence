# ZhongGuo 361 Style — character interactions
# 申诉/绩效沟通/举荐/攻讦/逐人夺爵/逐人致仕。
# AI 不主动使用这些互动（ai_will_do = 0）：AI 侧奖惩由年度考核自动裁决，避免与淘汰链双重执行。

# ============================================================
# 绩效申诉（封臣 → 领主）：被打 3.25 / 背上 PIP 时翻案
# ============================================================
zg361_appeal_interaction = {
	icon = icon_contract_modification_single
	category = interaction_category_vassal
	desc = zg361_appeal_interaction_desc

	cooldown = { years = 1 }
	cooldown_against_recipient = { years = 1 }
	auto_accept = no

	is_shown = {
		has_game_rule = zg361_on
		scope:actor = {
			is_ai = no
			has_character_modifier = zg361_grade_325
			has_variable = zg361_result_case_owner
			var:zg361_result_case_owner = scope:recipient
			has_variable = zg361_result_case_state
			var:zg361_result_case_state = 3
			has_variable = zg361_result_appeal_open
			var:zg361_result_appeal_open = 1
		}
		scope:recipient = {
			zg361_is_celestial_liege_trigger = yes
		}
	}

	is_valid = {
		scope:actor = { influence >= 150 }
	}
	is_valid_showing_failures_only = {
		scope:actor = { influence >= 150 }
	}

	ai_accept = {
		base = 30
		modifier = {
			add = 20
			scope:recipient = { opinion = { target = scope:actor value >= 20 } }
		}
		modifier = {
			add = 15
			scope:actor = {
				government_has_flag = government_has_merit
				merit_level >= 5
			}
		}
		modifier = {
			add = -100
			scope:actor = { is_at_war_with_liege = yes }
		}
	}

	ai_will_do = { base = 0 }

	on_accept = {
		scope:actor = {
			change_influence = { value = 0 subtract = zg361_appeal_influence_cost_value }
			zg361_b2_on_appeal_filed_effect = yes
			# 翻案：改判 3.5
			zg361_appeal_regrade_to_35_effect = yes
			debug_log = "ZG361: appeal accepted, regraded to 3.5"
		}
	}

	on_decline = {
		scope:actor = {
			change_influence = { value = 0 subtract = zg361_appeal_influence_cost_value }
			zg361_b2_on_appeal_filed_effect = yes
			zg361_b2_on_appeal_upheld_effect = yes
			add_opinion = { modifier = zg361_appeal_rejected target = scope:recipient }
			debug_log = "ZG361: appeal rejected"
		}
	}
}

zg361_review_talk_interaction = {
	icon = icon_acknowledge_governor
	category = interaction_category_friendly
	desc = zg361_review_talk_interaction_desc

	cooldown_against_recipient = { years = 1 }
	auto_accept = yes

	is_shown = {
		has_game_rule = zg361_on
		scope:actor = {
			zg361_is_celestial_liege_trigger = yes
		}
		scope:recipient = {
			liege = scope:actor
			OR = {
				has_character_modifier = zg361_grade_375
				has_character_modifier = zg361_grade_35
				has_character_modifier = zg361_grade_325
			}
		}
	}

	ai_will_do = { base = 0 }

	on_accept = {
		scope:recipient = {
			add_stress = minor_stress_loss
			add_opinion = { modifier = zg361_talked target = scope:actor }
		}
		debug_log = "ZG361: performance talk held"
	}
}
