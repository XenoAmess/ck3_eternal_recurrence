cmake_minimum_required(VERSION 3.24)
project(xar_focused_confirm_outcome LANGUAGES CXX)
if(NOT MSVC OR NOT CMAKE_GENERATOR STREQUAL "Ninja")
  message(FATAL_ERROR "This finite fixture imports the qualified MSVC/Ninja production object layout")
endif()
get_filename_component(_native_default "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(NATIVE_SOURCE "${_native_default}" CACHE PATH "Current production native source")
set(PRODUCTION_BUILD "" CACHE PATH "Completed same-source production build containing actual DLL objects/runtime/protocol")
if(NOT EXISTS "${PRODUCTION_BUILD}/compile_commands.json")
  message(FATAL_ERROR "Build the production DLL first with compile_commands enabled")
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_LIST_DIR}/extract_production_defines.py"
  "${PRODUCTION_BUILD}/compile_commands.json" "${NATIVE_SOURCE}" "${CMAKE_CURRENT_BINARY_DIR}/production-defines.cmake"
  RESULT_VARIABLE _definitions_result)
if(NOT _definitions_result EQUAL 0)
  message(FATAL_ERROR "Actual production source/definition binding rejected")
endif()
set(_target xar_ck3_ingame_decision_confirm_outcome12004_fixture)
set(_core_sources)
foreach(_unit IN ITEMS ingame_decision_item_v1 game_adapter zhongguo_case_snapshot_v1 zhongguo_scoreboard_action_v1 zhongguo_scoreboard_state_v1)
  list(APPEND _core_sources "${NATIVE_SOURCE}/src/${_unit}.cpp")
endforeach()
add_executable(${_target} EXCLUDE_FROM_ALL "${NATIVE_SOURCE}/tests/ingame_decision_confirm_outcome12004_fixture.cpp" ${_core_sources})
target_include_directories(${_target} PRIVATE "${NATIVE_SOURCE}/include" "${NATIVE_SOURCE}/src" "${NATIVE_SOURCE}/research")
target_compile_features(${_target} PRIVATE cxx_std_20)
target_compile_definitions(${_target} PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
target_compile_options(${_target} PRIVATE /W4 /WX /permissive- /EHsc /utf-8 /Gy)
set_source_files_properties("${NATIVE_SOURCE}/tests/ingame_decision_confirm_outcome12004_fixture.cpp" PROPERTIES COMPILE_OPTIONS /UNDEBUG)
include("${CMAKE_CURRENT_BINARY_DIR}/production-defines.cmake")
include("${CMAKE_CURRENT_LIST_DIR}/production-objects.cmake")
target_link_options(${_target} PRIVATE /OPT:REF /OPT:ICF /INCREMENTAL:NO)
target_link_libraries(${_target} PRIVATE "${PRODUCTION_BUILD}/xar_ck3_12002_runtime.lib" "${PRODUCTION_BUILD}/xar_bridge_protocol.lib" bcrypt user32 advapi32 shell32)
