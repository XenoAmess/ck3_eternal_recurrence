cmake_minimum_required(VERSION 3.24)
project(refill_add_cold_a03 LANGUAGES CXX ASM_MASM)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(refill_cold_target cold_target.cpp synthetic_add.asm)
add_executable(refill_cold_observer cold_observer.cpp)
foreach(target refill_cold_target refill_cold_observer)
  target_compile_options(${target} PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/EHsc> $<$<COMPILE_LANGUAGE:CXX>:/W4>)
  target_link_options(${target} PRIVATE /INCREMENTAL:NO)
endforeach()
