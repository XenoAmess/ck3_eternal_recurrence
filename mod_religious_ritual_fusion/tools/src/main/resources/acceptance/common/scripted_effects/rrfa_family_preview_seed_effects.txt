rrfa_seed_family_preview_child_effect = {
    clear_saved_scope = rrfa_family_seed_child
    create_character = {
        age = 20
        name = $NAME$
        gender_female_chance = 0
        random_traits = no
        faith = scope:rrfa_seed_actor.faith
        culture = scope:rrfa_seed_actor.culture
        location = scope:rrfa_seed_actor.location
        employer = scope:rrfa_seed_actor
        save_scope_as = rrfa_family_seed_child
    }
    if = {
        limit = { exists = scope:rrfa_family_seed_child }
        scope:rrfa_family_seed_child = {
            set_father = scope:rrfa_seed_actor
            set_character_rite = scope:rrfa_a
        }
        if = {
            limit = {
                scope:rrfa_family_seed_child = {
                    is_child_of = scope:rrfa_seed_actor
                    is_alive = yes
                    is_ai = yes
                    is_adult = yes
                    is_landed = no
                    faith = scope:rrfa_seed_actor.faith
                    rite = scope:rrfa_a
                    NOR = {
                        has_character_flag = ai_will_not_convert
                        this = faith.religious_head
                        rite.head_of_rite ?= this
                    }
                }
            }
            add_to_variable_list = { name = rrfa_family_seed_children target = scope:rrfa_family_seed_child }
            scope:rrfa_family_seed_child = {
                debug_log = "RRFAT: DIAG FAMILY_PREVIEW_INPUT_CHILD"
                debug_log_scopes = yes
            }
        }
        else = { debug_log = "RRFAT: TEST FAIL family_preview_input_child_ineligible" }
    }
    else = { debug_log = "RRFAT: TEST FAIL family_preview_input_child_missing" }
    clear_saved_scope = rrfa_family_seed_child
}
