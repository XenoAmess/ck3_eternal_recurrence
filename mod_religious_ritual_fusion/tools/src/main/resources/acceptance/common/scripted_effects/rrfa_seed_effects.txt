# PREPARATION ONLY. Never calls an RRF action, writes product state, or retries a partial seed.
rrfa_seed_inputs_effect = {
    save_scope_as = rrfa_seed_actor
    set_variable = { name = rrfa_original_rite value = rite }
    set_variable = { name = rrfa_original_faith value = faith }
    create_dynamic_rite = {
        save_scope_as = rrfa_a
        convert = no
        target_divergence = 0
        core_tenet_override_entry = tenet:tenet_esotericism
        core_tenet_override_entry = tenet:tenet_legalism
        core_tenet_override_entry = tenet:tenet_astrology
    }
    if = {
        limit = { exists = scope:rrfa_a }
        scope:rrfa_a = { save_scope_as = rrfa_seed_target }
        set_variable = { name = rrfa_seed_third value = tenet:tenet_astrology }
        debug_log = "RRFAT: DIAG SOURCE_A_CORE_PREPARATION_BEGIN"
        rrfa_align_seed_core_effect = yes
        debug_log = "RRFAT: DIAG SOURCE_A_CORE_PREPARATION_END"
        scope:rrfa_a = { change_rite_doctrine = doctrine:doctrine_divorce_disallowed }
        set_character_rite = scope:rrfa_a
        set_variable = { name = rrfa_source_a value = scope:rrfa_a }
        set_variable = { name = rrfa_source_a_faith value = scope:rrfa_a.faith }
        create_dynamic_rite = {
            save_scope_as = rrfa_b
            convert = no
            target_divergence = 0
            core_tenet_override_entry = tenet:tenet_esotericism
            core_tenet_override_entry = tenet:tenet_legalism
            core_tenet_override_entry = tenet:tenet_communal_identity
        }
        if = {
            limit = { exists = scope:rrfa_b }
            scope:rrfa_b = { save_scope_as = rrfa_seed_target }
            set_variable = { name = rrfa_seed_third value = tenet:tenet_communal_identity }
            debug_log = "RRFAT: DIAG SOURCE_B_RITE_CORE_PREPARATION_BEGIN"
            rrfa_align_seed_core_effect = yes
            debug_log = "RRFAT: DIAG SOURCE_B_RITE_CORE_PREPARATION_END"
            scope:rrfa_b = { change_rite_doctrine = doctrine:doctrine_divorce_allowed }
            set_variable = { name = rrfa_source_b_rite value = scope:rrfa_b }
            create_dynamic_rite = {
                save_scope_as = rrfa_b_faith_rite
                convert = no
                target_divergence = 0
                core_tenet_override_entry = tenet:tenet_esotericism
                core_tenet_override_entry = tenet:tenet_legalism
                core_tenet_override_entry = tenet:tenet_communal_identity
            }
            if = {
                limit = { exists = scope:rrfa_b_faith_rite }
                scope:rrfa_b_faith_rite = { change_rite_doctrine = doctrine:doctrine_divorce_allowed }
                detach_rite_to_new_faith = { target = scope:rrfa_b_faith_rite save_scope_as = rrfa_b_faith }
                if = {
                    limit = { exists = scope:rrfa_b_faith }
                    # Align after native detachment as well as native creation.
                    scope:rrfa_b_faith_rite = { save_scope_as = rrfa_seed_target }
                    set_variable = { name = rrfa_seed_third value = tenet:tenet_communal_identity }
                    debug_log = "RRFAT: DIAG SOURCE_B_FAITH_CORE_PREPARATION_BEGIN"
                    rrfa_align_seed_core_effect = yes
                    debug_log = "RRFAT: DIAG SOURCE_B_FAITH_CORE_PREPARATION_END"
                    set_variable = { name = rrfa_source_b_faith_rite value = scope:rrfa_b_faith_rite }
                    set_variable = { name = rrfa_source_b_faith value = scope:rrfa_b_faith }
                    set_variable = { name = rrfa_expected_b value = var:${EXPECTED_B} }
                    set_variable = { name = rrfa_mode value = flag:${MODE} }
                    add_known_tenet = tenet_esotericism
                    add_known_tenet = tenet_legalism
                    add_known_tenet = tenet_astrology
                    add_known_tenet = tenet_communal_identity
                    add_known_doctrine = doctrine_divorce_disallowed
                    add_known_doctrine = doctrine_divorce_allowed
                    add_piety = { value = ${PIETY} subtract = piety }
                    set_variable = { name = rrfa_piety_before value = piety }
                    add_character_flag = rrfa_seed_done
                    rrfa_seed_readiness_diagnostics_effect = yes
                    if = {
                        limit = { rrfa_inputs_intact_trigger = yes rite = var:rrfa_source_a knows_tenet = tenet_communal_identity knows_doctrine = doctrine_divorce_allowed var:rrfa_expected_b = { can_convert_to_rite = yes } }
                        debug_log = "RRFAT: TEST PASS fixture_ready"
                    }
                    else = { debug_log = "RRFAT: TEST FAIL fixture_ready" }
                }
                else = { debug_log = "RRFAT: TEST FAIL seed_detach_faith" }
            }
            else = { debug_log = "RRFAT: TEST FAIL seed_faith_rite" }
        }
        else = { debug_log = "RRFAT: TEST FAIL seed_same_faith_rite" }
    }
    else = { debug_log = "RRFAT: TEST FAIL seed_source_a" }
}

# Preparation mutates only the three just-created source Rites. Native override
# entries are seeds, not an assertion that another mod left exactly three cores.
# BEGIN/END delimit each source; counting the native iterator entry markers gives
# its actual before/after cardinality. Obtain identities independently through
# native queries or saved-state inspection; avoid full saved-target scope dumps.
rrfa_align_seed_core_effect = {
    every_in_list = { list = rrfa_seed_cores_to_remove remove_from_list = rrfa_seed_cores_to_remove }
    scope:rrfa_seed_target = {
        every_rite_tenet = {
            debug_log = "RRFAT: DIAG CORE_BEFORE_ENTRY"
            save_scope_as = rrfa_seed_item
            if = {
                limit = {
                    NOR = {
                        this = tenet:tenet_esotericism
                        this = tenet:tenet_legalism
                        this = scope:rrfa_seed_actor.var:rrfa_seed_third
                    }
                }
                add_to_list = rrfa_seed_cores_to_remove
            }
        }
    }
    # Snapshot first; never mutate the native core vector while enumerating it.
    every_in_list = {
        list = rrfa_seed_cores_to_remove
        save_scope_as = rrfa_seed_item
        scope:rrfa_seed_target = { remove_rite_tenet = scope:rrfa_seed_item }
    }
    every_in_list = { list = rrfa_seed_cores_to_remove remove_from_list = rrfa_seed_cores_to_remove }
    scope:rrfa_seed_target = {
        if = {
            limit = { NOT = { any_rite_tenet = { this = tenet:tenet_esotericism } } }
            add_rite_tenet = tenet:tenet_esotericism
        }
        if = {
            limit = { NOT = { any_rite_tenet = { this = tenet:tenet_legalism } } }
            add_rite_tenet = tenet:tenet_legalism
        }
        if = {
            limit = { NOT = { any_rite_tenet = { this = scope:rrfa_seed_actor.var:rrfa_seed_third } } }
            add_rite_tenet = scope:rrfa_seed_actor.var:rrfa_seed_third
        }
        every_rite_tenet = {
            debug_log = "RRFAT: DIAG CORE_AFTER_ENTRY"
        }
    }
    remove_variable = rrfa_seed_third
    clear_saved_scope = rrfa_seed_target
    clear_saved_scope = rrfa_seed_item
}
