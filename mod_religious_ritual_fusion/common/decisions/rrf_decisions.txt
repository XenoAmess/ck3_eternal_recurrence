	picture = { reference = "gfx/interface/illustrations/decisions/decision_personal_religious.dds" }
	decision_group_type = rrf_fusion
	sort_order = 220
	desc = rrf_begin_rite_fusion_decision_desc
	selection_tooltip = rrf_begin_rite_fusion_decision_tooltip
	confirm_text = rrf_begin_rite_fusion_decision_confirm
	is_shown = { is_ai = no is_landed = yes }
	is_valid = {
		rrf_actor_eligible_trigger = yes
		custom_description = {
			text = RRF_REQUIRE_SAME_FAITH_SOURCE
			faith = {
				any_faith_rite = {
					this != root.rite
					can_convert_to_rite = yes
				}
			}
		}
	}
	widget = {
		gui = "decision_view_widget_select_rite"
		controller = select_scope_object
		show_from_start = yes
		decision_to_second_step_button = "RRF_SELECT_SOURCE"
		setup_items = {
			scope:actor.faith = {
				every_faith_rite = {
					limit = {
						this != scope:actor.rite
						can_convert_to_rite = yes
					}
					add_to_list = item_list
				}
			}
		}
		default_item = {
			scope:actor.faith = {
				ordered_faith_rite = {
					limit = {
						this != scope:actor.rite
						can_convert_to_rite = yes
					}
					order_by = num_rite_county_followers
					position = 0
					save_scope_as = selected_item
				}
			}
		}
		# The decision list also evaluates the controller without opening its UI.
		# Supply the same legal candidate search as vanilla rite pickers. This
		# only selects a scope; the decision's AI execution gates remain closed.
		ai_select_item = {
			scope:actor.faith = {
				ordered_faith_rite = {
					limit = {
						this != scope:actor.rite
						can_convert_to_rite = yes
					}
					order_by = num_rite_county_followers
					position = 0
					save_scope_as = selected_item
				}
			}
		}
		is_item_valid = {
			save_temporary_scope_as = selected_item
			scope:actor = { rrf_target_rite_trigger = yes }
		}
	}
	effect = {
		custom_tooltip = RRF_OPEN_FUSION_DRAFT
		hidden_effect = { rrf_open_rite_effect = yes }
	}
	ai_potential = { always = no }
	ai_will_do = { base = 0 }
}
