# Every operation receives the actor Character as root. Item operations receive
# a typed Tenet/Doctrine scope as rrf_item; effects revalidate through shared code.
rrf_window_gui = {
	scope = character
	is_shown = {
		has_variable = rrf_state
		NOT = { has_character_flag = rrf_window_dismissed }
		OR = {
			var:rrf_state = flag:editing
			var:rrf_state = flag:committing
			var:rrf_state = flag:unresolved
		}
	}
}

rrf_editing_gui = {
	scope = character
	is_valid = { has_variable = rrf_state var:rrf_state = flag:editing }
}
rrf_committing_gui = {
	scope = character
	is_valid = { has_variable = rrf_state var:rrf_state = flag:committing }
}
rrf_unresolved_gui = {
	scope = character
	is_valid = { has_variable = rrf_state var:rrf_state = flag:unresolved }
}
rrf_rite_mode_gui = {
	scope = character
	is_valid = { has_variable = rrf_mode var:rrf_mode = flag:rite }
}

rrf_toggle_tenet_gui = {

rrf_dismiss_gui = {
	scope = character
	is_valid = {
		OR = {
			var:rrf_state ?= flag:editing

		}
	}
	effect = { rrf_dismiss_effect = yes }
}
