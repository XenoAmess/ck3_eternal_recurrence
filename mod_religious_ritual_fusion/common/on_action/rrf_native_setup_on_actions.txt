# on_actions lists are additive; do not replace stock/DES/MTS effect blocks.
on_rite_updated = { on_actions = { rrf_native_setup_applied } }
rrf_native_setup_applied = { effect = { rrf_on_native_setup_applied_effect = yes } }

# Initialize the neutral registry on new games and before a paused creation.
# Never clear a nonempty registry or invent a Doctrine when no provider is loaded.
on_game_start_after_lobby = { on_actions = { xar_rite_creation_ensure_optional_defaults } }
xar_rite_creation_collect_optional_default_doctrines = {
    on_actions = { xar_rite_creation_ensure_optional_defaults }
}
xar_rite_creation_ensure_optional_defaults = {
    effect = {
        if = {
            limit = { NOT = { has_global_variable_list = xar_rite_creation_optional_default_doctrines } }
            clear_global_variable_list = xar_rite_creation_optional_default_doctrines
        }
    }
}
